#Separatism Tatarstan
country_event = {
	id = separatism.1
	title = separatism.1.t
	desc = separatism.1.d
	timeout_days = 2
	picture = GFX_generic_event_separatism
	is_triggered_only = yes
	immediate = { clear_variable = TAT.separatism }
	option = {
		name = separatism.1.a
		log = "[GetDateText]: [This.GetName]: separatism.1.a executed"
		TAT = {
			set_country_flag = GENERIC_autonomy_federation_flag
		}
		release = TAT
		set_autonomy = {
			target = TAT
			autonomy_state = autonomy_puppet_federation
		}
		TAT = { 
			set_country_flag = GENERIC_separatism_rebeliion_flag
			hidden_effect = {
				news_event = {
					hours = 6
					id = separatism_news.1
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 2
				has_war_support < 0.40
			}
			modifier = {
				factor = 2
				has_war_support < 0.20
			}
			modifier = {
				factor = 0
				OR = {
					has_government = communism
					has_government = nationalist
					has_government = fascism
				}
			}
		}
	}
	option = {
		name = separatism.1.b
		log = "[GetDateText]: [This.GetName]: separatism.1.b executed"
		hidden_effect = {
			release = TAT
			set_autonomy = {
				target = TAT
				autonomy_state = autonomy_free
			}
			declare_war_on = {
				target = TAT
				type = annex_everything
			}
			if = {
				limit = { is_ai = yes }
				set_country_flag = annex_on_peace_conference@TAT
			}
			TAT = {
				set_country_flag = GENERIC_separatism_rebeliion_flag
				division_template = {
					name = "National Militia"
					priority = 0
					regiments = {
						L_Inf_Bat = { x = 0 y = 0 }
						L_Inf_Bat = { x = 0 y = 1 }
						L_Inf_Bat = { x = 0 y = 2 }
						L_Inf_Bat = { x = 0 y = 3 }
						Arty_Bat = { x = 1 y = 0 }
					}
					support = {
						L_Recce_Comp = { x = 0 y = 0 }
						L_Engi_Comp = { x = 0 y = 1 }
					}
				}
				random_owned_controlled_state = {
					create_unit = {
						division = "name = \"Militia Battalion\" division_template = \"National Militia\" start_experience_factor = 0.1"
						owner = TAT
					}
				}
				random_owned_controlled_state = {
					create_unit = {
						division = "name = \"Militia Battalion\" division_template = \"National Militia\" start_experience_factor = 0.1"
						owner = TAT
					}
				}
				random_owned_controlled_state = {
					create_unit = {
						division = "name = \"Militia Battalion\" division_template = \"National Militia\" start_experience_factor = 0.1"
						owner = TAT
					}
				}
				random_owned_controlled_state = {
					create_unit = {
						division = "name = \"Militia Battalion\" division_template = \"National Militia\" start_experience_factor = 0.1"
						owner = TAT
					}
				}
				random_owned_controlled_state = {
					create_unit = {
						division = "name = \"Militia Battalion\" division_template = \"National Militia\" start_experience_factor = 0.1"
						owner = TAT
					}
				}
			}
		}
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = separatism.1.c
		log = "[GetDateText]: [This.GetName]: separatism.1.c executed"
		add_political_power = -300
		TAT = {
			set_temp_variable = { modify_separatism = -20 }
			modify_separatism_support = yes
		}
		ai_chance = {
			base = 100
			modifier = {
				has_political_power < 150
				factor = 0
			}
		}
	}
