country_event = {
	id = sanctions.5
	title = sanctions.5.t
	desc = sanctions.5.d

	picture = GFX_gas_tanker
    
	is_triggered_only = yes

	immediate = {
		FROM = {
			set_temp_variable = { treasury_change = 1 }
			modify_treasury_effect = yes
			add_named_threat = {
				name = shadow_fleet_snatch_treat
				threat = 1
			}
			if = {
				limit = { tag = SOV }
				random_list = {
					99 = {
						country_event = { id = sanctions.7 }
					}
					1 = {
						country_event = { id = sanctions.9 }
					}
				}
			}
		}
		send_equipment = {
			equipment = convoy
			amount = 1
			target = FROM
		}
		set_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
	}

	option = {
		name = sanctions.5.a
		log = "[GetDateText]: [This.GetName]: sanctions.5.a executed"
		effect_tooltip = {
			FROM = {
				set_temp_variable = { treasury_change = 1 }
				modify_treasury_effect = yes
				add_named_threat = {
					name = shadow_fleet_snatch_treat
					threat = 1
				}
			}
			send_equipment = {
				equipment = convoy
				amount = 1
				target = FROM
			}
			set_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
		}
		set_country_flag = shadow_fleet_pirate_happened
	}

	option = {
		name = sanctions.5.b
		log = "[GetDateText]: [This.GetName]: sanctions.5.b executed"
		effect_tooltip = {
			FROM = {
				set_temp_variable = { treasury_change = 1 }
				modify_treasury_effect = yes
				add_named_threat = {
					name = shadow_fleet_snatch_treat
					threat = 1
				}
			}
			send_equipment = {
				equipment = convoy
				amount = 1
				target = FROM
			}
			set_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
		}
		create_wargoal = {
    		type = puppet_wargoal_focus
    		target = FROM
			expire = 120
		}
		set_country_flag = shadow_fleet_pirate_happened
		ai_chance = {
			base = 100
		}
	}
}
