country_event = {
	id = energy_selling.1
	title = energy_selling.1.t
	desc = { trigger = { has_country_flag = eon_energy_offer_cancelled } text = eon_energy_offer_cancelled_desc }
	desc = {
		trigger = { NOT = { has_country_flag = eon_energy_offer_cancelled } check_variable = { eon_energy_offer_counter_depth > 0 } }
		text = eon_energy_counter_pending_import_desc
	}
	desc = {
		trigger = { NOT = { has_country_flag = eon_energy_offer_cancelled } NOT = { check_variable = { eon_energy_offer_counter_depth > 0 } } }
		text = eon_energy_offer_import_desc
	}
	picture = GFX_energy_sell_buy
	is_triggered_only = yes
	immediate = { eon_energy_evaluate_offer = yes eon_energy_prepare_ai_counter = yes }
	option = {
		name = energy_selling.1.a
		log = "[GetDateText]: [This.GetName]: energy_selling.1.a executed"
		custom_effect_tooltip = eon_energy_accept_import_tt
		eon_energy_accept_offer = yes
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					NOT = { check_variable = { eon_energy_ai_accept_chance > 0 } }
					NOT = { eon_energy_normal_response_current = yes }
					NOT = { eon_energy_current_offer_capacity_available = yes }
				}
			}
		}
	}
	option = {
		name = energy_selling.1.b
		log = "[GetDateText]: [This.GetName]: energy_selling.1.b executed"
		custom_effect_tooltip = eon_energy_refuse_tt
		eon_energy_refuse_offer = yes
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					AND = { check_variable = { eon_energy_counter_ready = 1 } eon_energy_normal_response_current = yes }
					AND = { check_variable = { eon_energy_ai_accept_chance > 0 } eon_energy_normal_response_current = yes eon_energy_current_offer_capacity_available = yes }
				}
			}
		}
	}
	option = {
		name = eon_energy_counter_offer
		trigger = { eon_energy_normal_response_current = yes OR = { is_ai = no check_variable = { eon_energy_counter_ready = 1 } } }
		custom_effect_tooltip = eon_energy_counter_prepare_tt
		if = { limit = { is_ai = yes } eon_energy_send_ai_counter = yes }
		else = { eon_energy_prepare_counter_draft = yes }
		ai_chance = { base = 100 modifier = { factor = 0 OR = { is_ai = no NOT = { check_variable = { eon_energy_counter_ready = 1 } } } } }
	}
}
