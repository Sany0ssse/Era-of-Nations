country_event = {
	id = energy.1
	title = energy.1.t
	desc = energy.1.d
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	immediate = {
		set_variable = { ai_accept_chance = 25 }
		if = {
			limit = { check_variable = { nuclear_reactor_fuel_production > nuclear_fuel_consumption } }
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				set_temp_variable = { nuclear_fuel_consumption_tmp = nuclear_fuel_consumption }
				multiply_temp_variable = { nuclear_fuel_consumption_tmp = 10 }
				set_temp_variable = { var_reactor_material_stockpile_tmp = var_reactor_material_stockpile }
				subtract_from_temp_variable = { var_reactor_material_stockpile_tmp = eon_nuclear_fuel_trade_quantity }
				check_variable = { var_reactor_material_stockpile_tmp < nuclear_fuel_consumption_tmp }
			}
			add_to_variable = { ai_accept_chance = -50 }
		}
		if = {
			limit = {
				has_opinion = {
					target = FROM
					value > 100
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_opinion = {
					target = FROM
					value > 150
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_opinion = {
					target = FROM
					value > 200
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_war_with = FROM
			}
			add_to_variable = { ai_accept_chance = -1000 }
		}
		if = {
			limit = { check_variable = { display_income < display_expense } }
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				original_tag = CHI
				FROM = {
					original_tag = TAI
					NOT = { is_subject_of = CHI }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = USA
				FROM = {
					original_tag = CHI
					NOT = { is_subject_of = USA }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = SOV
				NOT = { has_idea = NATO_member }
				FROM = {
					has_idea = NATO_member
					NOT = { is_subject_of = SOV }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = FRA
				FROM = {
					original_tag = SOV
					NOT = { is_subject_of = FRA }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				check_variable = { eon_nuclear_fuel_trade_price = 0 }
			}
			add_to_variable = { ai_accept_chance = -1000 }
		}

		set_temp_variable = { price_adjust = eon_nuclear_fuel_trade_price }
		add_to_temp_variable = { price_adjust = -0.5 }
		multiply_temp_variable = { price_adjust = 100 }
		add_to_variable = { ai_accept_chance = price_adjust }
		set_temp_variable = { ammount_adjuster = eon_nuclear_fuel_trade_quantity }
		multiply_temp_variable = { ammount_adjuster = -0.001 }
		add_to_variable = { ai_accept_chance = ammount_adjuster }
	}

	option = {
		name = energy.1.a
		trigger = { set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 1 } eon_nuclear_fuel_trade_response_ready = yes }
		log = "[GetDateText]: [This.GetName]: energy.1.a executed"
		set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 1 }
		eon_nuclear_fuel_trade_accept = yes
		clear_variable = ai_accept_chance
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				check_variable = {
    				var = ai_accept_chance
    				value = 0
    				compare = less_than_or_equals
				}
			}

			modifier = { factor = 0 set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 1 } NOT = { eon_nuclear_fuel_trade_response_ready = yes } }
		}
	}
	option = {
		name = energy.1.b
		log = "[GetDateText]: [This.GetName]: energy.1.b executed"
		set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 1 }
		eon_nuclear_fuel_trade_refuse = yes
		clear_variable = ai_accept_chance
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 1 }
				eon_nuclear_fuel_trade_response_ready = yes
				check_variable = { ai_accept_chance > 0 }
			}
		}
	}
}

# Sell Event Chains
country_event = {
	id = energy.10
	title = energy.10.t
	desc = energy.10.d
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	immediate = {
		set_variable = { ai_accept_chance = 0 }
		# Preserve the legacy fuel duration after converting stock units to kg.
		if = {
			limit = { check_variable = { var_reactor_material_stockpile < 173100 } }
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = { FROM = { is_in_faction_with = ROOT } }
			add_to_variable = { ai_accept_chance = 50 }
		}
		if = {
			limit = {
				OR = {
					AND = {
						ROOT = { original_tag = USA }
						FROM = { original_tag = SOV }
					}
					AND = {
						ROOT = { original_tag = SOV }
						FROM = { original_tag = USA }
					}
					AND = {
						ROOT = { original_tag = CHI }
						FROM = { original_tag = USA }
					}
					AND = {
						ROOT = { original_tag = USA }
						FROM = { original_tag = CHI }
					}
					AND = {
						ROOT = { original_tag = CHI }
						FROM = { original_tag = RAJ }
					}
					AND = {
						ROOT = { original_tag = RAJ }
						FROM = { original_tag = CHI }
					}
					AND = {
						ROOT = { original_tag = TAI }
						FROM = { original_tag = CHI }
					}
					AND = {
						ROOT = { original_tag = CHI }
						FROM = { original_tag = TAI }
					}
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				has_opinion = {
					value > 100
					target = FROM
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_opinion = {
					target = FROM
					value < 75
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				set_temp_variable = { nuclear_fuel_consumption_tmp = nuclear_fuel_consumption }
				multiply_temp_variable = { nuclear_fuel_consumption_tmp = 10 }
				check_variable = { var_reactor_material_stockpile > nuclear_fuel_consumption_tmp }
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				NOT = {
					any_other_country = {
						is_in_array = { global.enrichment_countries = THIS.id }
						check_variable = { nuclear_reactor_fuel_production > nuclear_fuel_consumption }
						# Preserve the legacy fuel duration after converting stock units to kg.
						check_variable = { var_reactor_material_stockpile > 86550 }
					}
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		set_temp_variable = { price_adjust = eon_nuclear_fuel_trade_price }
		add_to_temp_variable = { price_adjust = -0.5 }
		multiply_temp_variable = { price_adjust = -100 }
		add_to_variable = { ai_accept_chance = price_adjust }
		# Benefit follows a six-month reserve shortfall, not the absolute kg lot.
		# A huge unwanted shipment cannot buy acceptance through an AI bonus.
		set_temp_variable = { eon_fuel_ai_reserve_target = nuclear_fuel_consumption }
		multiply_temp_variable = { eon_fuel_ai_reserve_target = 26 }
		clamp_temp_variable = { var = eon_fuel_ai_reserve_target min = 0 max = 20000000 }
		set_temp_variable = { eon_fuel_ai_reserve_gap = eon_fuel_ai_reserve_target }
		subtract_from_temp_variable = { eon_fuel_ai_reserve_gap = var_reactor_material_stockpile }
		clamp_temp_variable = { var = eon_fuel_ai_reserve_gap min = 0 }
		set_temp_variable = { ammount_adjuster = 0 }
		if = {
			limit = { check_variable = { eon_fuel_ai_reserve_gap > 0 } }
			set_temp_variable = { ammount_adjuster = eon_nuclear_fuel_trade_quantity }
			clamp_temp_variable = { var = ammount_adjuster min = 0 max = eon_fuel_ai_reserve_gap }
			divide_temp_variable = { ammount_adjuster = eon_fuel_ai_reserve_gap }
			multiply_temp_variable = { ammount_adjuster = 50 }
			clamp_temp_variable = { var = ammount_adjuster min = 0 max = 50 }
			add_to_variable = { ai_accept_chance = ammount_adjuster }
		}
		else = { clamp_variable = { var = ai_accept_chance max = 0 } }
		set_temp_variable = { eon_fuel_ai_max_useful_lot = eon_fuel_ai_reserve_gap }
		# One editor step accommodates procurement lot rounding near the target.
		add_to_temp_variable = { eon_fuel_ai_max_useful_lot = 500 }
		if = {
			limit = { check_variable = { eon_nuclear_fuel_trade_quantity > eon_fuel_ai_max_useful_lot } }
			clamp_variable = { var = ai_accept_chance max = 0 }
		}
	}

	option = {
		name = energy.10.a
		trigger = { set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 2 } eon_nuclear_fuel_trade_response_ready = yes }
		log = "[GetDateText]: [This.GetName]: energy.10.a executed"
		set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 2 }
		eon_nuclear_fuel_trade_accept = yes
		clear_variable = ai_accept_chance
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				check_variable = {
    				var = ai_accept_chance
    				value = 0
    				compare = less_than_or_equals
				}
			}

			modifier = { factor = 0 set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 2 } NOT = { eon_nuclear_fuel_trade_response_ready = yes } }
		}
	}
	option = {
		name = energy.10.b
		log = "[GetDateText]: [This.GetName]: energy.10.b executed"
		set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 2 }
		eon_nuclear_fuel_trade_refuse = yes
		clear_variable = ai_accept_chance
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				set_temp_variable = { eon_nuclear_fuel_trade_response_kind = 2 }
				eon_nuclear_fuel_trade_response_ready = yes
				check_variable = { ai_accept_chance > 0 }
			}
		}
	}
}
