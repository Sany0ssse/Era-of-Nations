# Purchase Event Chains
country_event = {
	id = energy.1
	title = energy.1.t
	desc = energy.1.d
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	immediate = {
		set_variable = { ai_accept_chance = 25 }
		if = {
			limit = { check_variable = { nuclear_reactor_fuel_production > nuclear_fuel_consumption } }
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				set_temp_variable = { nuclear_fuel_consumption_tmp = nuclear_fuel_consumption }
				multiply_temp_variable = { nuclear_fuel_consumption_tmp = 10 }
				set_temp_variable = { var_reactor_material_stockpile_tmp = var_reactor_material_stockpile }
				subtract_from_temp_variable = { var_reactor_material_stockpile_tmp = FROM.temp_nuclear_fuel_ammount }
				check_variable = { var_reactor_material_stockpile < nuclear_fuel_consumption_tmp }
			}
			add_to_variable = { ai_accept_chance = -50 }
		}
		if = {
			limit = {
				has_opinion = {
					target = PREV
					value > 100
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_opinion = {
					target = PREV
					value > 150
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_opinion = {
					target = PREV
					value > 200
				}
			}
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				has_war_with = PREV
			}
			add_to_variable = { ai_accept_chance = -1000 }
		}
		if = {
			limit = { check_variable = { display_income < display_expense } }
			add_to_variable = { ai_accept_chance = 100 }
		}
		if = {
			limit = {
				original_tag = CHI
				FROM = {
					original_tag = TAI
					NOT = { is_subject_of = CHI }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = USA
				FROM = {
					original_tag = CHI
					NOT = { is_subject_of = USA }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = SOV
				NOT = { has_idea = NATO_member }
				FROM = {
					has_idea = NATO_member
					NOT = { is_subject_of = SOV }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				original_tag = FRA
				FROM = {
					original_tag = SOV
					NOT = { is_subject_of = FRA }
				}
			}
			add_to_variable = { ai_accept_chance = -100 }
		}
		if = {
			limit = {
				check_variable = { FROM.temp_nuclear_fuel_price = 0 }
			}
			add_to_variable = { ai_accept_chance = -1000 }
		}
		if = {
			limit = {
				check_variable = { nuclear_reactor_fuel_production = 0 }
			}
			add_to_variable = { ai_accept_chance = -250 }
		}
		set_temp_variable = { price_adjust = FROM.temp_nuclear_fuel_price }
		add_to_temp_variable = { price_adjust = -0.05 }
		multiply_temp_variable = { price_adjust = 10000 }
		add_to_variable = { ai_accept_chance = price_adjust }
		set_temp_variable = { ammount_adjuster = FROM.temp_nuclear_fuel_ammount }
		multiply_temp_variable = { ammount_adjuster = -0.001 }
		add_to_variable = { ai_accept_chance = ammount_adjuster }
	}

	option = { #accept to sell
		name = energy.1.a
		log = "[GetDateText]: [This.GetName]: energy.1.a executed"
		if = {
			limit = {
				FROM = { is_ai = yes }
			}
			set_temp_variable = { change_resource = -500 }
			change_reactor_grade_material_effect = yes
			set_temp_variable = { treasury_change = 0.25 }
			modify_treasury_effect = yes
			FROM = {
				set_temp_variable = { change_resource = 500 }
				change_reactor_grade_material_effect = yes
				set_temp_variable = { treasury_change = -0.25 }
				modify_treasury_effect = yes
				country_event = energy.2
			}
		}
		else = {
			set_temp_variable = { change_resource = FROM.temp_nuclear_fuel_ammount }
			multiply_temp_variable = { change_resource = -1 }
			change_reactor_grade_material_effect = yes
			set_temp_variable = { purchase_price =  FROM.temp_nuclear_fuel_price }
			multiply_temp_variable = { purchase_price = FROM.temp_nuclear_fuel_ammount }
			multiply_temp_variable = { purchase_price = 0.01 }
			set_temp_variable = { treasury_change = purchase_price }
			modify_treasury_effect = yes
			FROM = {
				set_temp_variable = { change_resource = FROM.temp_nuclear_fuel_ammount }
				change_reactor_grade_material_effect = yes
				set_temp_variable = { purchase_price =  FROM.temp_nuclear_fuel_price }
				multiply_temp_variable = { purchase_price = FROM.temp_nuclear_fuel_ammount }
				multiply_temp_variable = { purchase_price = -0.01 }
				set_temp_variable = { treasury_change = purchase_price }
				modify_treasury_effect = yes
				country_event = energy.2
			}
		}
		
		clear_variable = ai_accept_chance

		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				check_variable = {
    				var = ai_accept_chance
    				value = 0
    				compare = less_than_or_equals
				}
			}
		}
	}
	option = { #refuse to sell
		name = energy.1.b
		log = "[GetDateText]: [This.GetName]: energy.1.b executed"
		FROM = {
			country_event = energy.3
		}

		clear_variable = ai_accept_chance

		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				check_variable = { ai_accept_chance > 0 }
			}
		}
	}
}
