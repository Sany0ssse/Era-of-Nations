add_namespace = nuclear_ai_raid

country_event = {
	id = nuclear_ai_raid.1
	title = nuclear_ai_raid.1.t
	desc = nuclear_ai_raid.1.d
	picture = GFX_full_nuke_strike
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			every_country = {
				limit = {
					has_country_flag = sugomusus
					exists = yes
					has_capitulated = no
				}
				clr_country_flag = sugomusus
				for_each_loop = {
					array = enemies
					value = v
					if = {
						limit = { exists = yes has_capitulated = no }
					add_named_threat = {
						threat = 100
						name = strat_nuclear_strike_wt_tt
						}

					if = {
						limit = {
							     var:v = { has_country_flag = perimetr_system eon_nuclear_arsenal_ready = yes OR = { check_variable = { num_equipment@nuclear_missile_equipment > 9 } check_variable = { num_equipment@nuclear_ballistic_missile_equipment > 9 } } }
					            }
						set_temp_variable = { eon_nuclear_ai_retaliation_previous_level = global.temp_arctic_nuclear_level }
						set_variable = { global.temp_arctic_nuclear_level = var:v.arctic_nuclear_level }
						set_temp_variable = { eon_nuclear_ai_retaliation_previous_shooter = global.nuke_striker }
						set_variable = { global.nuke_striker = var:v }
						launch_nuclear_ai_country_strike = yes
						set_variable = { global.nuke_striker = eon_nuclear_ai_retaliation_previous_shooter }
						set_variable = { global.temp_arctic_nuclear_level = eon_nuclear_ai_retaliation_previous_level }
                        if = { limit = { has_country_flag = eon_nuclear_scripted_launch_succeeded }
                        var:v = {
							add_named_threat = {
								threat = 100
								name = strat_nuclear_strike_wt_tt
								}

						}
                        }
					}
				}
				}
			}
		}
	}
	option = {
		name = nuclear_ai_raid.1.a
		hidden_effect = {
			every_country = {
				set_variable = { nukes_strikes_count = 0 }
			}
			clear_variable = global.temp_arctic_nuclear_level
			clear_variable = global.nuke_striker
		}
	}
}
