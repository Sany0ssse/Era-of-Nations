country_event = {
	id = formable_nations.1
	title = formable_nations.1.t
	desc = formable_nations.1.d
	is_triggered_only = yes
	picture = GFX_election_rally
	option = {
		name = formable_nations.1.a
		log = "[GetDateText]: [This.GetName]: formable_nations.1.a executed"
		effect_tooltip = {
			for_each_scope_loop = {
				array = core_states
				add_core_of = FROM
			}
			every_unit_leader = { set_nationality = FROM }
			FROM = {
			add_to_variable = { treasury = ROOT.treasury }
			clamp_variable = { var = treasury min = -1000000 max = 1000000 }
			add_to_variable = { debt = ROOT.debt }
	    	clamp_variable = { var = debt min = 0 max = 1000000 }
			add_to_variable = { int_investments = ROOT.int_investments }
    		clamp_variable = { var = int_investments min = 0 max = 1000000 }
			annex_country = {
					target = ROOT
					transfer_troops = yes
				}
			}
		}
		FROM = { country_event = formable_nations.2 }
	}
	option = {
		name = formable_nations.1.b
		log = "[GetDateText]: [This.GetName]: formable_nations.1.b executed"
		effect_tooltip = {
			set_temp_variable = { percent_change = 10 }
			set_temp_variable = { tag_index = FROM.id }
			set_temp_variable = { influence_target = ROOT.id }
			change_influence_percentage = yes
			FROM = {
				set_autonomy = {
					target = ROOT
					autonomous_state = autonomy_satellite_state
					end_wars = no
					end_civil_wars = no
				}
			}
			set_temp_variable = { treasury_change = 6 }
			modify_treasury_effect = yes
		}
		FROM = { country_event = formable_nations.3 }
	}
	option = {
		name = formable_nations.1.c
		log = "[GetDateText]: [This.GetName]: formable_nations.1.c executed"
		effect_tooltip = {
			set_temp_variable = { percent_change = -15 }
			set_temp_variable = { tag_index = FROM.id }
			set_temp_variable = { influence_target = ROOT.id }
			change_influence_percentage = yes
			FROM = {
				add_opinion_modifier = {
					target = ROOT
					modifier = major_breach_of_trust
				}
			}
		}
		FROM = { country_event = formable_nations.4 }
		ai_chance = {
			base = 0
			modifier = {
				add = 300
				OR = {
					has_government = nationalist
					has_government = fascism
					has_government = communism
				}
			}
		}
	}
