	desc = { trigger = { set_temp_variable = { eon_mediation_response_topic = 1 } eon_mediation_mediator_response_valid = yes NOT = { eon_mediation_cancelled_current = yes } } text = eon_mediation_mediator_deescalation_desc }
	desc = { trigger = { set_temp_variable = { eon_mediation_response_topic = 1 } NOT = { eon_mediation_mediator_response_valid = yes } NOT = { eon_mediation_cancelled_current = yes } } text = eon_mediation_stale_desc }
	picture = GFX_handshake_black
	is_triggered_only = yes
	option = {
		name = eon_mediation_accept
		trigger = { set_temp_variable = { eon_mediation_response_topic = 1 } eon_mediation_mediator_response_valid = yes }

		eon_mediation_accept_mediator = yes
		ai_chance = {
			factor = 2
			modifier = {
				factor = 1.5
				has_opinion = { target = FROM value > 50 }
				var:eon_mediation_party_b = { ROOT = { has_opinion = { target = PREV value > 50 } } }
			}
			modifier = {
				factor = 0.25
				OR = {
					has_opinion = { target = FROM value < -25 }
					var:eon_mediation_party_b = { ROOT = { has_opinion = { target = PREV value < -25 } } }
				}
			}
			modifier = { factor = 0 set_temp_variable = { eon_mediation_response_topic = 1 } NOT = { eon_mediation_mediator_response_valid = yes } }
		}
	}
	option = {
		name = eon_mediation_decline
		trigger = { set_temp_variable = { eon_mediation_response_topic = 1 } always = yes }
		set_temp_variable = { eon_mediation_response_topic = 1 }
		eon_mediation_refuse_mediator = yes
		ai_chance = { factor = 1 }
	}
}
