	desc = AB_mobilization.1.desc

	is_triggered_only = yes

	option = {
		name = AB_mobilization.1.a
		log = "[GetDateText]: [This.GetName]: AB_mobilization.1.a executed"
		add_ideas = AB_partial_mobilization_idea
		set_temp_variable = { population_for_mobilize = max_manpower_k }
		multiply_temp_variable = { population_for_mobilize = 1.3 }
		add_manpower = population_for_mobilize
		hidden_effect = {
			country_event = AB_mobilization.9
		}

	}
}
