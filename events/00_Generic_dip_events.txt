	id = mdc4.1
	title = mdc4.1.t
	desc = mdc4.1.d
	picture = GFX_negotiations

	immediate = {
		hidden_effect = {
			save_event_target_as = fta_applicant
			FROM = {
				save_event_target_as = fta_inviter
			}
		}
	}

	is_triggered_only = yes

	option = { #Agree
		name = mdc4.1.a
		log = "[GetDateText]: [This.GetName]: mdc4.1.a executed"

			base = 100
		}
		FROM = {
			add_opinion_modifier = {
				target = ROOT
				modifier = free_trade_agreement
			}
			country_event = { id = mdc4.2 hours = 6 }
		}
		add_opinion_modifier = {
			target = FROM
			modifier = free_trade_agreement
		}
		set_temp_variable = { receiver_nation = ROOT.id }
		set_temp_variable = { sender_nation = FROM.id }
		set_improved_trade_agreement = yes
		newline = yes
		set_mutual_investment_treaty = yes
		hidden_effect = {
			set_country_flag = fta_applicant@ROOT
			FROM = { set_country_flag = fta_inviter@ROOT }
			#news_event = { id = news.1 hours = 12 }
		}
	}

	option = { #Refuse
		name = mdc4.1.b
		log = "[GetDateText]: [This.GetName]: mdc4.1.b executed"
		ai_chance = { factor = 0 }
		FROM = {
			country_event = { id = mdc4.3 hours = 6 }
			add_opinion_modifier = {
				target = ROOT
				modifier = insult
			}
		}
	}
}
