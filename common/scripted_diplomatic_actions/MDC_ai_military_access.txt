scripted_diplomatic_actions = {
	aisend_military_access = {
		allowed = { always = yes }

		visible = {
			ROOT = {
				is_ai = yes
				NOT = { has_country_flag = ai_military_access_cooldown }
			}
			NOT = { has_capitulated = yes }
			NOT = { gives_military_access_to = ROOT }
			NOT = { root_supports_an_enemy_of_this_country = yes }
			NOT = { is_subject_of = ROOT }
		}

		selectable = {
			ROOT = { has_opinion = { target = PREV value > 0 } }
			has_opinion = { target = ROOT value > 0 }
		}

		cost = 25

		requires_acceptance = yes
		show_acceptance_on_action_button = yes

		icon = 2

		receive_description = AISEND_MILITARY_ACCESS_RECIEVER_LOC

		on_sent_effect = {
			ROOT = { set_variable = { pending_ai_military_access_country = PREV.id } }
		}
		complete_effect = {
            give_military_access = ROOT
			ROOT = {
				clear_variable = pending_ai_military_access_country
				set_country_flag = {
					flag = airecallcooldown
					days = 30
					value = 1
				}
				if = { limit = { is_major = yes }
					set_country_flag = { flag = ai_military_access_cooldown days = 28 value = 1 }
				}
				else = {
					set_country_flag = { flag = ai_military_access_cooldown days = 160 value = 1 }
				}
			}
		}
		reject_effect = {
			ROOT = {
				clear_variable = pending_ai_military_access_country
				set_country_flag = { flag = rejected_ai_military_access_@PREV days = 180 value = 1 }
			}
		}

		ai_desire = {
			base = 0
			modifier = {
				add = 25
				has_volunteers_amount_from = {
					tag = ROOT
					count > 0
				}
			}
			modifier = {
				add = 75
                is_guaranteed_by = ROOT
			}
			modifier = {
				add = 50
				is_top_influencer = yes
			}
			modifier = {
				add = 500
                any_neighbor_country = {
                    has_war_with = ROOT
                }
				is_directly_reachable = yes
			}
			modifier = {
				factor = 0
				ROOT = { has_variable = pending_ai_military_access_country }
			}
		}

		ai_acceptance = {
            base = 100
		}
	}
}
