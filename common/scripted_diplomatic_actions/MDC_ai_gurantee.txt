scripted_diplomatic_actions = {
	aisend_guarantee = {

		allowed = {
			always = yes
		}

		visible = {
			threat > 0.2
			ROOT = {
				is_ai = yes
				is_major = yes
				is_subject = no
				has_war = no
				this_concretely_threatened = no
				strength_ratio = {
    				tag = PREV
    				ratio > 1
				}
				NOT = { has_country_flag = aiguranteecooldown }
			}
			NOT = { is_guaranteed_by = ROOT }
		}

		selectable = {
			is_major = no
			is_subject = no
			is_in_faction = no
			this_concretely_threatened = yes
		}

		cost = 25

		requires_acceptance = no
		show_acceptance_on_action_button = yes

		icon = 1

		receive_description = AISEND_GUARANTEE_RECIEVER_LOC
		accept_title = AISEND_GUARANTEE_ACCEPTED_TITLE
		accept_description = AISEND_GUARANTEE_ACCEPTED_DESC

		complete_effect = {
			ROOT = {
                diplomatic_relation = {
                    country = PREV
                    relation = guarantee
                    active = yes
                }
				if = { limit = { is_major = yes }
					set_country_flag = { flag = aiguranteecooldown days = 28 value = 1 }
				}
				else = {
					set_country_flag = { flag = aiguranteecooldown days = 160 value = 1 }
				}
			}
		}

		ai_desire = {
			base = 5
			modifier = {
				add = 25
				is_directly_reachable = yes
			}
			modifier = {
				factor = 2
				FROM = { is_neighbor_of = ROOT }
			}
			modifier = {
				factor = 4
				is_top_influencer = yes
			}
			modifier = {
				factor = 0
				ROOT = {
					has_opinion = {
    					target = PREV
    					value < 75
					}
				}
			}
			modifier = {
				factor = 0
				ROOT = {
					NOT = {
						any_other_country = {
							OR = {
								has_wargoal_against = PREV.PREV
								is_justifying_wargoal_against = PREV.PREV
							}
							ROOT = {
								has_opinion = {
    								target = PREV
    								value < 75
								}
							}
						}
					}
				}
			}
		}

		ai_acceptance = {
			base = 100
		}
	}
}
