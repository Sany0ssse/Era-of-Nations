@Ammo_send_qty = 100000

scripted_diplomatic_actions = {

	Send_ammo = { # will be used in most localizations
        
        # ROOT - initiator
        # THIS - target country
		allowed = { # Disables send button and hides AI acceptance icon if false
            always = yes
		}
        
		visible = {
			always = yes
		}
		selectable = {
			OR = { eon_ammo_response_pending = yes eon_ammo_gui_ready = yes }
		}
		can_be_accepted = { eon_ammo_response_ready = yes }
		can_be_sent = { eon_ammo_gui_ready = yes }
        
        # where_can_i_put_this? = {
            # allowed     -> Confirmation window vanishes if the recipient tries to interact
            # visible     -> Recipient can view confirmation window but neither the reject nor complete effects execute
            # selectable  -> Recipient can view confirmation, effects don't run
            # can_be_sent -> Effects work but doesn't seem to have any impact on blocking sender (bugged in 1.13+?)
            # can_be_accepted -> Non-starter
            #
            # ROOT = {
                # NOT = {
                    # has_country_flag = JAM_diplo_cooldown
                # }
            # }
        # }

		requires_acceptance = yes # if true the action will require acceptance of target country

		show_acceptance_on_action_button = yes # set to no if acceptance shouldn't be displayed on the action button

		icon = 9 # icon frame to use in notification
        
        # ROOT - Sender
        # THIS - Receiver
        # runs when the action is sent
		on_sent_effect = { eon_ammo_send_offer = yes }
        
        # runs when the action is properly sent and rejected
		reject_effect = { eon_ammo_reject_offer = yes }
        
		# runs when the action is properly sent and accepted
		complete_effect = { eon_ammo_accept_offer = yes }
        
		send_description = Sda_send_ammo_send_desc # if there is no scripted gui, this desc will be used in send yes/no pop up

		receive_description = Sda_send_ammo_receive_desc # if there is no scripted gui, this desc will be used in send yes/no pop up

		accept_title = Sda_send_ammo_accept_title #can be overridden to customize accepted/rejected pop ups for the sender
		accept_description = Sda_send_ammo_accept_desc
		reject_title = Sda_send_ammo_reject_title
		reject_description = Sda_send_ammo_reject_desc
		
		# a list of ai_will_do entries that will be used to decide if AI should accept or reject a request
		ai_acceptance = {
			Sda_send_ammo_base = { # name is used for loc
				base = 100
			}
            
            Sda_send_ammo_full = {
                base = 0
                
                modifier = {
                    add  = 10
                    THIS = {
                        NOT = {
                            AND = {
                                check_variable = { ammo_production_rating > 0 }
                                check_variable = { ammo_stock_days  < 8 }
                            }
                        }
                    }
                }
                modifier = {
                    add  = -200
                    THIS = {
                        check_variable = { ammo_production_rating > 0 }
                        check_variable = { ammo_stock_days  < 8 }
                    }
                }
			}
		}

		# ai_will_do to decide if AI should send this action or not
		ai_desire = {
			base = -1
			# Lifecycle safety only: retain every inherited political preference.
			modifier = {
				add = -20000
				NOT = { eon_ammo_gui_ready = yes }
			}
            
            # Don't bother players if they rejected multiple requests recently unless if they're in the crisis zone
            modifier = {
                add = -20000
                
                is_ai = no
                has_country_flag = {
                    flag  = Ammo_diplo_rejections
                    value > 4
                }
                OR = {
                    check_variable = { ammo_production_rating > 0 }
                    check_variable = { ammo_stock_months > 1 }
                }
            }
            
            # Only send one at a time to players
            modifier = {
                add = -20000
                
                is_ai = no
                ROOT = {
                    has_country_flag = Ammo_diplo_AI_debounce
                }
            }
            
            # Enemies
			modifier = {
				add = -9000
                
                ROOT = {
                    OR = {
                        has_war_with = PREV
                        any_allied_country = {
                            has_war_with = PREV.PREV
                        }
                        any_subject_country = {
                            has_war_with = PREV.PREV
                        }
                        PREV = {
                            has_same_ideology = no
                        }
                    }
                }
			}
            
            # No ammo to give
            # AI will try to keep:
            #     - 1 day's worth of ammo on hand at all times
            #     - At least 5 years' ammo on hand if they have a net loss of ammo at the moment
            # AI will also avoid donating ammo if it will take at least 1 year to fill their reserves (they only want to
            # donate ammo they can safely give away)
            modifier = {
				add = -1000
                
                ROOT = {
                    OR = {
                        check_variable = { ammo_stock < maximum_ammo_consumption }
                        AND = {
                            check_variable = { ammo_production_rating < 0 }
                            check_variable = { ammo_stock_years < 5 }
                        }
                        AND = {
                            check_variable = { ammo_production_rating > 0 }
                            check_variable = { ammo_stock_years > 1 }
                        }
                    }
                }
			}
            
            # Little extra priority to allied players in need
            modifier = {
                add = 250
                
                THIS = {
                    is_ai = no
                    OR = {
                        is_ally_with = ROOT
                        is_subject_of = ROOT
                        is_guaranteed_by = ROOT
                        is_in_faction_with = ROOT
                        any_country = {
                            has_same_ideology = no
                            has_war_with = PREV
                        }
                    }
                    
                    check_variable = { ammo_production_rating < 0 }
                    check_variable = { ammo_stock_years < 1 }
                }
            }
            
            # Give to allies in need
            modifier = {
				add = 500
                
                THIS = {
                    OR = {
                        is_ally_with = ROOT
                        is_subject_of = ROOT
                        is_guaranteed_by = ROOT
                        is_in_faction_with = ROOT
                        any_country = {
                            has_same_ideology = no
                            has_war_with = PREV
                        }
                    }
                    
                    OR = {
                        check_variable = { ammo_stock < maximum_ammo_consumption }
                        AND = {
                            check_variable = { ammo_production_rating < 0 }
                            check_variable = { ammo_stock_years < 1 }
                        }
                    }
                }
			}
            
            # Get rid of excess
            modifier = {
                add = 500
                
                ROOT = {
                    check_variable = { ammo_production_rating > 0 }
                    check_variable = { ammo_stock_days  < 8 }
                }
                THIS = {
                    OR = {
                        is_ally_with = ROOT
                        is_subject_of = ROOT
                        is_guaranteed_by = ROOT
                        is_in_faction_with = ROOT
                        any_country = {
                            has_same_ideology = no
                            has_war_with = PREV
                        }
                    }
                    
                    # Don't donate to someone who also has excess
                    NOT = {
                        AND = {
                            check_variable = { ammo_production_rating > 0 }
                            check_variable = { ammo_stock_days  < 8 }
                        }
                    }
                }
            }
		}
	}
}
