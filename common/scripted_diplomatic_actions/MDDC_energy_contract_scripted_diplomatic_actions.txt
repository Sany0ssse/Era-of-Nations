scripted_diplomatic_actions = {
	energy_contract_agreement = {
		allowed = { always = yes }
		visible = {
			is_neighbor_of = ROOT
			NOT = {
				THIS = { has_country_flag = energy_agreement@ROOT }
			}
		}
		selectable = {
			custom_trigger_tooltip = {
				tooltip = eon_diplo_no_pending_energy_framework_tt
				eon_energy_framework_send_ready = yes
			}
		}
		can_be_sent = {
			eon_energy_framework_send_ready = yes
		}
		can_be_accepted = {
			custom_trigger_tooltip = {
				tooltip = eon_diplo_energy_framework_response_valid_tt
				eon_energy_framework_terms_available = yes
				eon_energy_framework_response_current = yes
				NOT = { has_country_flag = eon_energy_framework_withdrawn }
				ROOT = { NOT = { has_country_flag = eon_energy_framework_withdrawn } }
			}
		}
		cost = 75
		requires_acceptance = yes
		show_acceptance_on_action_button = yes
		icon = 17
		send_description = PROPOSE_ENERGY_AGREEMENT_DESC

		on_sent_effect = {
			log = "[GetDateText]: [Root.GetName]: diplomatic action on_sent_effect propose_improved_energy_agreement"
			set_variable = { eon_energy_framework_pending_sender = ROOT.id }
			ROOT = { set_variable = { pending_energy_offer_country = PREV.id } }
		}
		complete_effect = {
			if = {
				limit = {
					eon_energy_framework_terms_available = yes
					eon_energy_framework_response_current = yes
					NOT = { has_country_flag = eon_energy_framework_withdrawn }
					ROOT = { NOT = { has_country_flag = eon_energy_framework_withdrawn } }
				}
				log = "[GetDateText]: [Root.GetName]: diplomatic action complete_effect propose_improved_energy_agreement"
				set_country_flag = energy_agreement@ROOT
				ROOT = {
					set_country_flag = energy_agreement@PREV
				}
				THIS = {
					add_opinion_modifier = { target = ROOT modifier = energy_agreement_accept }
					reverse_add_opinion_modifier = { target = ROOT modifier = energy_agreement_accept }
					set_temp_variable = { percent_change = 1 }
					set_temp_variable = { tag_index = ROOT.id }
					set_temp_variable = { influence_target = THIS.id }
					change_influence_percentage = yes
					ROOT = {
						set_temp_variable = { percent_change = 1 }
						set_temp_variable = { tag_index = PREV.id }
						set_temp_variable = { influence_target = THIS.id }
						change_influence_percentage = yes
					}
				}

			}
			eon_energy_framework_finish_response = yes
		}
		reject_effect = {
			if = {
				limit = {
					eon_energy_framework_response_current = yes
					NOT = { has_country_flag = eon_energy_framework_withdrawn }
					ROOT = { NOT = { has_country_flag = eon_energy_framework_withdrawn } }
				}
				log = "[GetDateText]: [Root.GetName]: diplomatic action complete_effect propose_improved_energy_agreement"
				THIS = {
					add_opinion_modifier = { target = ROOT modifier = energy_agreement_reject }
					reverse_add_opinion_modifier = { target = ROOT modifier = energy_agreement_reject }
				}

			}
			eon_energy_framework_finish_response = yes
		}

		accept_title = PROPOSE_ENERGY_AGREEMENT_ACCEPT_TT
		accept_description = PROPOSE_ENERGY_AGREEMENT_ACCEPT_DESC
		reject_title = PROPOSE_ENERGY_AGREEMENT_REJECT_TT
		reject_description = PROPOSE_ENERGY_AGREEMENT_REJECT_DESC

		ai_acceptance = {
			base_energy_condition = {
				base = -25
			}
			condition_three = {
				base = 0
				modifier = {
					set_temp_variable = { opinion_calculator = opinion@ROOT }
					multiply_temp_variable = { opinion_calculator = 0.05 }
					check_variable = { opinion_calculator > -60 }
					add = opinion_calculator
				}
			}
			condition_six = {
				base = 0
				modifier = {
					add = 25
					ROOT = { is_in_faction_with = PREV }
				}
			}
			condition_energy_five = {
				base = 0
				modifier = {
					in_same_economic_faction = yes
					add = 15
				}
			}
			condition_energy_six = {
				base = 0
				modifier = {
					add = 10
					is_neighbor_of = ROOT
				}
			}
			condition_seven = {
				base = 0
				modifier = {
					add = 10
					is_subject_of = ROOT
				}
			}
			condition_eight = {
				base = 0
				modifier = {
					add = 5
					is_influencer = yes
				}
			}
		}
		ai_desire = {
			factor = 10
			modifier = {
				is_neighbor_of = ROOT
				factor = 2
			}
			modifier = {
				in_same_economic_faction = yes
				factor = 4
			}
			modifier = {
				is_in_faction_with = ROOT
				factor = 5
			}
			modifier = {
				is_subject_of = ROOT
				factor = 999
			}
			modifier = {
				set_temp_variable = { opinion_calculator = opinion@ROOT }
				multiply_temp_variable = { opinion_calculator = 0.05 }
				check_variable = { opinion_calculator > -60 }
				factor = opinion_calculator
			}
			modifier = {
				factor = 0
				ROOT = { NOT = { check_variable = { pending_energy_offer_country = 0 } } }
			}
		}
	}

	cancel_energy_agreement = {
		allowed = { always = yes }
		visible = { THIS = { has_country_flag = energy_agreement@ROOT } }
		selectable = {
			custom_trigger_tooltip = {
				tooltip = CANCEL_ENERGY_AGREEMENT_TOOLTIP
				THIS = { has_country_flag = energy_agreement@ROOT }
			}
		}
		cost = 75
		can_be_sent = { always = yes }
		show_acceptance_on_action_button = no
		requires_acceptance = no
		icon = 17
		send_description = CANCEL_ENERGY_AGREEMENT_DESC

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: diplomatic action cancel_energy_agreement"
			if = {
				limit = {
					OR = {
						has_country_flag = energy_agreement@ROOT
						ROOT = { has_country_flag = energy_agreement@PREV }
					}
				}
				clr_country_flag = energy_agreement@ROOT
				remove_opinion_modifier = { target = ROOT modifier = energy_agreement_accept }
				ROOT = {
					clr_country_flag = energy_agreement@PREV
					remove_opinion_modifier = { target = PREV modifier = energy_agreement_accept }
				}
				ROOT = {
					set_temp_variable = { eon_energy_pair_partner = PREV }
					eon_energy_end_pair = yes
				}
			}
			# Withdraw pending invitations of this pair; their old response drains the locks.
			set_temp_variable = { eon_energy_pair_partner = ROOT }
			eon_energy_framework_withdraw_pair = yes
			ROOT = {
				set_temp_variable = { eon_energy_pair_partner = PREV }
				eon_energy_framework_withdraw_pair = yes
			}
		}

		ai_desire = {
			base = 5
			modifier = {
				has_war_with = ROOT
				factor = 100
			}
			modifier = {
				set_temp_variable = { opinion_calculator = opinion@ROOT }
				multiply_temp_variable = { opinion_calculator = 0.05 }
				check_variable = { opinion_calculator < -10 }
				factor = opinion_calculator
			}
		}
	}
}
