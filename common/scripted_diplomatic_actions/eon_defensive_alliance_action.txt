scripted_diplomatic_actions = {
	eon_propose_defensive_alliance = {
		# First package: AI may answer, but does not initiate unsolicited factions.
		allowed = { ROOT = { is_ai = no } }
		visible = {
			exists = yes
			ROOT = { exists = yes }
			NOT = { tag = ROOT }
		}
		selectable = {
			custom_trigger_tooltip = {
				tooltip = eon_defensive_alliance_offer_terms_available_tt
				eon_defensive_alliance_offer_terms_available = yes
			}
			custom_trigger_tooltip = {
				tooltip = eon_defensive_alliance_offer_no_pending_tt
				eon_defensive_alliance_offer_no_pending = yes
			}
		}
		can_be_sent = { eon_defensive_alliance_offer_send_ready = yes }
		can_be_accepted = {
			custom_trigger_tooltip = {
				tooltip = eon_defensive_alliance_offer_response_valid_tt
				eon_defensive_alliance_offer_response_valid = yes
			}
		}
		cost = 0
		requires_acceptance = yes
		show_acceptance_on_action_button = yes
		icon = 21
		send_description = eon_propose_defensive_alliance_send_desc
		on_sent_effect = {
			if = {
				limit = { eon_defensive_alliance_offer_send_ready = yes }
				eon_defensive_alliance_offer_start = yes
			}
		}
		complete_effect = {
			if = {
				limit = { eon_defensive_alliance_offer_response_valid = yes }
				eon_defensive_alliance_offer_establish = yes
			}
			eon_defensive_alliance_offer_finish_response = yes
		}
		reject_effect = { eon_defensive_alliance_offer_finish_response = yes }
		ai_acceptance = {
			# Strong recipient opinion can overcome caution across outlooks; outlook is only a bonus.
			eon_defensive_alliance_ai_base = { base = -50 }
			eon_defensive_alliance_ai_opinion = {
				base = 0
				modifier = {
					add = 25
					has_opinion = { target = ROOT value > 50 }
				}
				modifier = {
					add = 30
					has_opinion = { target = ROOT value > 100 }
				}
			}
			eon_defensive_alliance_ai_outlook = {
				base = 0
				modifier = {
					add = 10
					is_same_government_THIS_ROOT = yes
				}
			}
			eon_defensive_alliance_ai_policy = {
				base = 0
				modifier = {
					set_temp_variable = {
						eon_defensive_alliance_ai_policy = modifier@ai_get_ally_desire_factor
					}
					add = var:eon_defensive_alliance_ai_policy
				}
			}
		}
	}
}
