				check_variable = { treasury > 4.9 }

			visegrad_invest_5_button_click = {
				ai_will_do = {
					base = 0

						check_variable = { treasury > 4.9 }

					}

			}

				set_country_flag = { flag = temp_vishegrad_invest value = 1 days = 30 }

				set_temp_variable = { treasury_change = -5.00 }
				modify_treasury_effect = yes
				ingame_update_setup = yes
