defined_text = {
    name = Ammo_status

    text = { # Full
        trigger = {
            check_variable = { ammo_production_rating > 0 }
            OR = {
                check_variable = { ammo_stock_days < 1 }
                check_variable = { ammo_stock = max_ammo_limit }
            }
        }
        localization_key = Ammo_full_txt
    }
    text = { # Empty
        trigger = {
            check_variable = { ammo_production_rating < 0 }
            OR = {
                check_variable = { ammo_stock_days < 1 }
                check_variable = { ammo_stock = 0 }
            }
        }
        localization_key = Ammo_empty_txt
    }
    text = { # Exactly 0 net or gaining, >5yr
        trigger = {
            OR = {
                check_variable = { ammo_production_rating = 0 }
                AND = {
                    check_variable = { ammo_production_rating > 0 }
                    check_variable = { ammo_stock_days > 1825 }
                }
            }
        }
        localization_key = Ammo_p_gt_5_yrs_txt
    }
    text = { # Gaining, <=5yr and >=1yr
        trigger = {
            check_variable = { ammo_production_rating > 0 }
            check_variable = {
                var     = ammo_stock_days
                value   = 1825
                compare = less_than_or_equals
            }
            check_variable = {
                var     = ammo_stock_days
                value   = 365
                compare = greater_than_or_equals
            }
        }
        localization_key = Ammo_p_yrs_txt
    }
    text = { # Gaining, <1yr
        trigger = {
            check_variable = { ammo_production_rating > 0 }
            check_variable = { ammo_stock_days < 365 }
        }
        localization_key = Ammo_p_days_txt
    }
    text = { # Draining, <1yr
        trigger = {
            check_variable = { ammo_production_rating < 0 }
            check_variable = { ammo_stock_days < 365 }
        }
        localization_key = Ammo_n_days_txt
    }
    text = { # Draining, <=5yr and >=1yr
        trigger = {
            check_variable = { ammo_production_rating < 0 }
            check_variable = {
                var     = ammo_stock_days
                value   = 1825
                compare = less_than_or_equals
            }
            check_variable = {
                var     = ammo_stock_days
                value   = 365
                compare = greater_than_or_equals
            }
        }
        localization_key = Ammo_n_years_txt
    }
    text = { # Draining, >5yr
        trigger = {
            check_variable = { ammo_production_rating < 0 }
            check_variable = { ammo_stock_days > 1825 }
        }
        localization_key = Ammo_n_gt_5_yrs_txt
    }
}
