# Native action ROOT requester / THIS provider. Menu ROOT provider / FROM
# requester. The requester is explicit so provider ROOT cannot reverse policy.
eon_support_request_policy_allowed = {
 exists = yes
 OR = { has_idea = large_power has_idea = great_power has_idea = superpower }
 check_variable = { eon_support_request_policy_requester > 0 }
 var:eon_support_request_policy_requester = {
  exists = yes
  NOT = { tag = PREV }
  check_variable = { THIS.id = PREV.eon_support_request_policy_requester }
  has_defensive_war = yes
  OR = { has_idea = non_power has_idea = minor_power has_idea = regional_power }
  NOT = { has_war_with = PREV }
  PREV = { has_opinion = { target = PREV value > 49 } }
 }
}

  NOT = { tag = PREV }

}
