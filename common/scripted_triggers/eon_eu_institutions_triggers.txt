eon_eu_office_available = {
 OR = {
  AND = { check_variable = { eon_eu_office_requested_id = 1 } }
  AND = { check_variable = { eon_eu_office_requested_id = 2 } }
  AND = { check_variable = { eon_eu_office_requested_id = 3 } }
  AND = { check_variable = { eon_eu_office_requested_id = 4 } has_global_flag = treaty_of_lisbon }
  AND = { check_variable = { eon_eu_office_requested_id = 5 } }
  AND = { check_variable = { eon_eu_office_requested_id = 6 } has_global_flag = frontex_ExDir }
  AND = { check_variable = { eon_eu_office_requested_id = 7 } has_global_flag = EU_finance_minister }
  AND = { check_variable = { eon_eu_office_requested_id = 8 } has_global_flag = euro_army }
  AND = { check_variable = { eon_eu_office_requested_id = 9 } has_global_flag = euro_navy }
  AND = { check_variable = { eon_eu_office_requested_id = 10 } has_global_flag = EU_UNSC }
 }
}
eon_eu_office_nominee_eligible = {
 eon_eu_is_active_member = yes
 eon_eu_office_available = yes
 NOT = { has_country_flag = eon_eu_office_nomination_cooldown }
 # Council-government voting suspension does not disqualify an independent EP nominee by origin.
 OR = { check_variable = { eon_eu_office_requested_id = 2 } NOT = { has_idea = EUU_voting_rights_suspended } }
 OR = {
  NOT = { check_variable = { eon_eu_office_requested_id = 3 } }
  AND = {
   has_idea = the_euro
   OR = { check_variable = { eon_eu_ecb_appointed_generation < 1 } NOT = { check_variable = { var = eon_eu_ecb_appointed_generation value = eon_eu_office_nominee_generation compare = equals } } }
  }
 }
 OR = {
  NOT = { check_variable = { eon_eu_office_requested_id = 4 } }
  NOT = { check_variable = { var = eon_eu_council_served_generation value = eon_eu_office_nominee_generation compare = equals } }
  check_variable = { eon_eu_council_nominee_terms < 2 }
 }
 OR = {
  NOT = { check_variable = { eon_eu_office_requested_id = 6 } }
  NOT = { check_variable = { var = eon_eu_frontex_served_generation value = eon_eu_office_nominee_generation compare = equals } }
  check_variable = { eon_eu_frontex_nominee_terms < 2 }
 }
 OR = {
  NOT = { check_variable = { eon_eu_office_requested_id = 9 } }
  any_owned_state = { is_coastal = yes }
 }
}

eon_eu_can_cast_office_ballot = {
 custom_trigger_tooltip = {
 tooltip = eon_eu_office_current_ballot_tt
 eon_eu_is_active_member = yes
 check_variable = { global.eon_eu_office_id > 0 }
 has_global_flag = eon_eu_office_ballot_deadline
 # Suspended Council voting rights do not silence the citizens' Parliament seats.
 OR = {
  check_variable = { global.eon_eu_office_id = 2 }
  AND = { check_variable = { global.eon_eu_office_id = 1 } check_variable = { global.eon_eu_office_phase = 2 } }
  NOT = { has_idea = EUU_voting_rights_suspended }
 }
 OR = { NOT = { check_variable = { global.eon_eu_office_id = 3 } } has_idea = the_euro }
 }
}
