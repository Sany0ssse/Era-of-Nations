# Scripted diplomacy context: ROOT is sender; current country is receiver.
# Contact is deliberately open to defenders, minor members and early wars.
eon_peace_contact_ready = {
	exists = yes
	ROOT = { exists = yes }
	NOT = { tag = ROOT }
	has_war_with = ROOT
	ROOT = { has_war_with = PREV }
}
# The settlement affects the selected two countries; no government may bind a subject silently.
eon_peace_settlement_ready = {
	eon_peace_contact_ready = yes
	is_subject = no
	has_capitulated = no
	NOT = { is_in_peace_conference = yes }
	NOT = { civilwar_target = ROOT }
	NOT = { any_subject_country = { has_war_together_with = PREV has_war_with = ROOT } }
	ROOT = {
		is_subject = no
		has_capitulated = no
		NOT = { is_in_peace_conference = yes }
		NOT = { civilwar_target = PREV }
		NOT = { any_subject_country = { has_war_together_with = PREV has_war_with = PREV.PREV } }
	}
}
# Conference can involve subjects/factions implicitly: require a verified isolated conflict.
eon_peace_conference_scope_ready = {
	eon_peace_settlement_ready = yes
	NOT = { any_subject_country = { exists = yes } }
	ROOT = { NOT = { any_subject_country = { exists = yes } } }
	NOT = { any_enemy_country = { NOT = { tag = ROOT } } }
	ROOT = { NOT = { any_enemy_country = { NOT = { tag = PREV.PREV } } } }
	NOT = {
		any_country = {
			NOT = { tag = ROOT }
			NOT = { tag = PREV }
			OR = { has_war_together_with = ROOT has_war_together_with = PREV }
		}
	}
}
# Native primitive r2 confirms pair-only execution; participating subjects remain excluded.
eon_peace_bilateral_engine_scope_ready = {
	eon_peace_settlement_ready = yes
}
