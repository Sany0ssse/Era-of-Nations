		AND = {
			check_variable = { influence_array^0 = SOV }
			check_variable = { influence_array_val^0 > 9.999 }
		}
		AND = {
			check_variable = { influence_array^1 = SOV }
			check_variable = { influence_array_val^1 > 9.999 }
		}
		AND = {
			check_variable = { influence_array^2 = SOV }
