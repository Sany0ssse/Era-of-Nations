# A cash repayment needs an active or overdue claim, regardless of loan rates.
eon_debt_default_payment_period_available = {
	OR = {
		has_active_mission = debt_default_main_mission
		has_country_flag = eon_debt_default_overdue
	}
}

eon_debt_default_payment_policy_available = {
	exists = yes
	eon_debt_default_payment_period_available = yes
	check_variable = { treasury > 0 }
	check_variable = { debt_default_left > 0 }
}

# The original positive total certifies a real mission, and is consumed once.
eon_debt_default_completed_available = {
	check_variable = { debt_default_left = 0 }
	check_variable = { debt_default_total > 0 }
	NOT = { has_country_flag = eon_debt_default_overdue }
}

eon_debt_default_timeout_available = {
	check_variable = { debt_default_left > 0 }
	check_variable = { debt_default_total > 0 }
	NOT = { has_country_flag = eon_debt_default_overdue }
}

# No existing mission or unpaid record may be overwritten by another default.
eon_debt_default_start_record_available = {
	exists = yes
	NOT = { has_active_mission = debt_default_main_mission }
	NOT = { has_country_flag = eon_debt_default_overdue }
	check_variable = { debt_default_left = 0 }
	check_variable = { debt_default_total = 0 }
	check_variable = { debt > 0 }
	NOT = { has_decision = debt_default_sell_civilian_factories }
	NOT = { has_decision = debt_default_dismantle_military_factories }
	NOT = { has_decision = debt_default_scrap_dockyard }
	NOT = { has_decision = debt_default_cut_down_government_services }
}

# Preserve the existing national/player interest and treasury policy on execution.
eon_debt_default_start_available = {
	eon_debt_default_start_record_available = yes
	check_variable = { interest_rate > 14.999 }
	if = {
		limit = { is_ai = no }
		custom_trigger_tooltip = {
			tooltip = treasury_less_than_5
			check_variable = { treasury < 5 }
		}
	}
}

# GDP exactly 100 belongs to the smaller payment; the two bands do not overlap.
eon_debt_default_pay_10_available = {
	eon_debt_default_payment_policy_available = yes
	check_variable = { var = gdp_total value = 100 compare = less_than_or_equals }
}

eon_debt_default_pay_50_available = {
	eon_debt_default_payment_policy_available = yes
	check_variable = { gdp_total > 100 }
}
