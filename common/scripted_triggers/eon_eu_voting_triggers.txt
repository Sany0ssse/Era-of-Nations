eon_eu_council_vote_passes = { if = { limit = { eon_eu_council_deadline_outcome_current = yes } check_variable = { global.eon_eu_council_outcome_yes = 1 } } else = {
	check_variable = { global.eon_eu_voters > 0 }
	check_variable = { global.current_active_vote_disp > 0 }
	OR = {
		AND = {
			eon_eu_agenda_requires_unanimity = yes
			check_variable = { global.EU_active_vote_no = 0 }
			# Abstention is permitted; an unanswered country has not consented.
			check_variable = { global.eon_eu_unanswered_members = 0 }
			check_variable = { global.EU_active_vote_yes > 0 }
		}
		AND = {
			eon_eu_agenda_requires_unanimity = no
			check_variable = { var = global.eon_eu_yes_fraction value = 0.55 compare = greater_than_or_equals }
			OR = {
				check_variable = { var = global.EU_pop_ratio_yes value = 0.65 compare = greater_than_or_equals }
				check_variable = { global.eon_eu_opposing_members < 4 }
			}
		}
	}

} }

eon_eu_council_sway_ready = {
 eon_eu_can_cast_council_vote = yes
 is_in_array = { array = global.EU_member value = THIS }
 ROOT = {
  eon_eu_can_cast_council_vote = yes
  is_in_array = { array = global.EU_member value = THIS }
 }
 is_ai = yes
 NOT = { has_country_flag = current_vote_influenced }
 EU_external_chi_influence_15 = no
 OR = { EU_external_sov_influence_15 = no SOV = { eon_eu_is_active_member = yes } }
 OR = {
  AND = {
   is_in_array = { array = global.EU_active_vote_supporters value = ROOT }
   is_in_array = { array = global.EU_active_vote_rejecters value = THIS }
  }
  AND = {
   is_in_array = { array = global.EU_active_vote_rejecters value = ROOT }
   is_in_array = { array = global.EU_active_vote_supporters value = THIS }
  }
 }
 # Native trigger arithmetic uses the same value as the debit. PP balancing is
 # a gameplay coefficient, never a claim that real votes can be bought.
 # UI reads the existing canonical census; the effect refreshes it before
 # rechecking this trigger. No effect-only country loop is placed in a trigger.
 set_temp_variable = { eon_eu_council_sway_total_k = global.eon_eu_voting_population_k }
 set_temp_variable = { eon_eu_council_sway_target_k = eon_eu_voting_population_k }
 check_variable = { eon_eu_council_sway_total_k > 0 }
 set_temp_variable = { eon_eu_council_sway_cost = eon_eu_council_sway_target_k }
 divide_temp_variable = { eon_eu_council_sway_cost = eon_eu_council_sway_total_k }
 multiply_temp_variable = { eon_eu_council_sway_cost = 7500 }
 clamp_temp_variable = { var = eon_eu_council_sway_cost min = 0 max = 1000 }
 set_temp_variable = { base_pp_cost = eon_eu_council_sway_cost }
 ROOT = { has_political_power > eon_eu_council_sway_cost }
}

  AND = { eon_eu_council_round_identity_current = yes has_global_flag = eon_eu_council_vote_deadline }

   any_country = {
    eon_eu_is_active_member = yes is_in_array = { array = global.EU_member value = THIS }
    OR = {
     AND = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result check_variable = { days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result > 0 } }
     AND = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result_timer check_variable = { days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result_timer > 0 } }
    }
   }
   NOT = { any_country = {
    eon_eu_is_active_member = yes is_in_array = { array = global.EU_member value = THIS }
    OR = {
     AND = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result check_variable = { var = days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result value = 0 compare = less_than_or_equals } }
