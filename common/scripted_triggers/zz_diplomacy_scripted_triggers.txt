is_major_conflict_t = {
	ROOT = {
		is_major = yes
	}
	THIS = {
		is_major = yes
	}
}

is_half_major_conflict_t = {
	OR = {
		ROOT = {
			is_major = yes
		}
		THIS = {
			is_major = yes
		}
	}
}

is_minor_conflict_t = {
	ROOT = {
		is_major = no
	}
	THIS = {
		is_major = no
	}
}	

is_reachable = {
	OR = {
		is_neighbor_of = ROOT
		any_allied_country = {
			is_neighbor_of = ROOT
		}
	}
	
	ROOT = {
		OR = {
			is_neighbor_of = PREV
			any_allied_country = {
				is_neighbor_of = PREV
			}
		}	
	}
}

is_reachable_continent = {

	OR = {
		any_owned_state = {
			is_on_same_continent_as = ROOT
		}
		any_controlled_state = {
			is_on_same_continent_as = ROOT
		}
		any_allied_country = {
			OR = {
				any_owned_state = {
					is_on_same_continent_as = ROOT
				}
				any_controlled_state = {
					is_on_same_continent_as = ROOT
				}
			}
		}	
	}
	ROOT = {
		OR = {
			any_owned_state = {
				is_on_same_continent_as = THIS
			}
			any_controlled_state = {
				is_on_same_continent_as = THIS
			}
			any_allied_country = {
				OR = {
					any_owned_state = {
						is_on_same_continent_as = THIS
					}
					any_controlled_state = {
						is_on_same_continent_as = THIS
					}
				}
			}	
		}
	}
}

is_directly_reachable = {
	OR = {
		is_neighbor_of = ROOT
		ROOT = {
			any_allied_country = {
				is_neighbor_of = PREV.PREV
			}
		}
		AND = {
			ROOT = {
				any_owned_state = {
					is_coastal = yes
				}
			}
			any_owned_state = {
				is_coastal = yes
			}
		}
	}
}

NonCoreControlledAlliesTHIS = {
	OR = {
		any_allied_country = {
			any_controlled_state = {
				NOT = {
					is_core_of = ROOT
				}
				is_owned_by = ROOT	
			}
		}	
		any_controlled_state = {
			NOT = {
				is_core_of = ROOT
			}
			is_owned_by = ROOT
		}	
	}
}

NonCoreControlledAlliesROOT = {
	OR = {
		ROOT = {
			any_allied_country = {
				any_controlled_state = {
					NOT = {
						is_core_of = PREV.PREV
					}
					is_owned_by = PREV.PREV
				}
			}
		}
		ROOT = {
			any_controlled_state = {
				NOT = {
					is_core_of = PREV.PREV
				}

			}
		}	
	}
}

ControlledAlliesROOT = {
	any_allied_country = {
		any_owned_state = {
			is_controlled_by_ROOT_or_ally = yes
		}
	}	
}

ControlledAlliesTHIS = {
	ROOT = {
		any_allied_country = {
			any_owned_state = {
				OR = {
					is_controlled_by = PREV.PREV
					controller = {
						is_ally_with = PREV.PREV
					}
				}
			}
		}
	}
