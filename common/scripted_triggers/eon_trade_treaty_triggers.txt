# Native diplomacy: ROOT = sender, THIS = receiver.
# The inherited Swiss/EU, allied-country and Eritrean policies are preserved below.
eon_trade_treaty_terms_available = {
	exists = yes
	ROOT = {
		exists = yes
		NOT = { has_country_flag = trade_agreement@PREV }
	}
	custom_trigger_tooltip = {
		tooltip = PROPOSE_TRADE_AGREEMENT_TOOLTIP
		THIS = {
			NOT = { has_country_flag = trade_agreement@ROOT }
		}
	}
	THIS = {
		if = {
			limit = {
				ROOT = { NOT = { has_idea = EU_member } }
				THIS = { tag = SWI }
			}
			NOT = { has_war_with = ROOT }
			has_opinion = { target = ROOT value > 50 }
		}
		else = {
			NOT = { has_war_with = ROOT }
			has_opinion = { target = ROOT value > 10 }
		}
		if = {
			limit = {
				is_in_faction = yes
				any_allied_country = {
					exists = yes
				}
			}
			# Do not ally your enemies friends
			any_allied_country = {
				NOT = { has_war_with = ROOT }
			}
		}
	}
	ERI_is_not_transitional_government = yes
}

eon_trade_treaty_no_pending = {
	check_variable = { pending_trade_offer_country = 0 }
	check_variable = { eon_trade_treaty_pending_sender = 0 }
	ROOT = {
		check_variable = { pending_trade_offer_country = 0 }
		check_variable = { eon_trade_treaty_pending_sender = 0 }
	}
}

eon_trade_treaty_send_ready = {
	eon_trade_treaty_terms_available = yes
	eon_trade_treaty_no_pending = yes
}

eon_trade_treaty_response_current = {
	check_variable = { eon_trade_treaty_pending_sender = ROOT.id }
	ROOT = { check_variable = { pending_trade_offer_country = PREV.id } }
}

eon_trade_treaty_response_valid = {
	eon_trade_treaty_terms_available = yes
	eon_trade_treaty_response_current = yes
	NOT = { has_country_flag = eon_trade_treaty_withdrawn }
	ROOT = { NOT = { has_country_flag = eon_trade_treaty_withdrawn } }
}

# The recovery action also recognizes a legacy one-sided treaty.
eon_trade_treaty_pair_exists = {
	OR = {
		has_country_flag = trade_agreement@ROOT
		ROOT = { has_country_flag = trade_agreement@PREV }
	}
}
