# Native scripted diplomacy: ROOT is the sender; current country is the receiver.
# Fixed terms remain the existing mutual investment treaty, without new prices.
eon_investment_treaty_terms_available = {
	exists = yes
	NOT = { has_country_flag = mutual_investment_treaty_@ROOT }
	NOT = { has_war_with = ROOT }
	has_opinion = { target = ROOT value > 25 }
	ROOT = {
		exists = yes
		NOT = { has_country_flag = mutual_investment_treaty_@PREV }
	}
	# Preserve the existing existential allied-country condition, including its scope.
	if = {
		limit = { is_in_faction = yes any_allied_country = { exists = yes } }
		any_allied_country = { NOT = { has_war_with = ROOT } }
	}
	ERI_is_not_transitional_government = yes
}

# One active treaty request per country, reserving both its sender and receiver.
eon_investment_treaty_send_ready = {
	eon_investment_treaty_terms_available = yes
	check_variable = { pending_mutual_investment_treaty_offer = 0 }
	check_variable = { eon_investment_treaty_pending_sender = 0 }
	ROOT = {
		check_variable = { pending_mutual_investment_treaty_offer = 0 }
		check_variable = { eon_investment_treaty_pending_sender = 0 }
	}
}

# Version 1 records the existing fixed terms, and excludes legacy outgoing-only offers.
eon_investment_treaty_response_current = {
	check_variable = { eon_investment_treaty_pending_sender = ROOT.id }
	check_variable = { eon_investment_treaty_pending_terms = 1 }
	ROOT = {
		check_variable = { pending_mutual_investment_treaty_offer = PREV.id }
		check_variable = { eon_investment_treaty_pending_terms = 1 }
	}
}

eon_investment_treaty_response_not_withdrawn = {
	NOT = { check_variable = { eon_investment_treaty_withdrawn_partner = ROOT.id } }
	ROOT = {
		NOT = { check_variable = { eon_investment_treaty_withdrawn_partner = PREV.id } }
	}
}

eon_investment_treaty_response_valid = {
	eon_investment_treaty_response_current = yes
	eon_investment_treaty_response_not_withdrawn = yes
	eon_investment_treaty_terms_available = yes
}

# A one-sided story-set treaty can still be cancelled and reconciled bilaterally.
eon_investment_treaty_pair_active = {
	OR = {
		has_country_flag = mutual_investment_treaty_@ROOT
		ROOT = { has_country_flag = mutual_investment_treaty_@PREV }
	}
}
