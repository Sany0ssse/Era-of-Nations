# Creating a faction and joining one are separate political rights.
eon_defensive_alliance_creator_eligible = {
	exists = yes
	is_subject = no
	is_in_faction = no
	has_rule = can_create_factions
	has_offensive_war = no
	NOT = {
		OR = {
			has_idea = CIW_member
			has_idea = European_Security_Council
			has_idea = EDU_major_non_EU_ally
			has_idea = BLR_neutrality_politic
		}
	}
}
eon_defensive_alliance_member_eligible = {
	exists = yes
	is_subject = no
	is_in_faction = no
	has_rule = can_join_factions
	has_offensive_war = no
	NOT = {
		has_idea = BLR_neutrality_politic
	}
	# Preserve the country's current rank/law/idea world-tension threshold.
	set_temp_variable = { eon_defensive_alliance_join_tension = modifier@join_faction_tension }
	check_variable = { var = global.threat value = eon_defensive_alliance_join_tension compare = greater_than_or_equals }
}
