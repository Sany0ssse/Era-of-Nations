# Entry policy follows the existing military-service GUI. Its 60-day pair
# timer is already set before selection. Rank/government changes do not end
# an existing paid agreement.
eon_advisers_policy_allowed = {
 exists = yes
 NOT = { tag = eon_advisers_policy_provider }
 OR = { has_civil_war = yes has_war = yes }
 NOT = { has_war_with = eon_advisers_policy_provider }
 OR = {
  check_variable = { influence_array^0 = eon_advisers_policy_provider }
  check_variable = { influence_array^1 = eon_advisers_policy_provider }
  check_variable = { influence_array^2 = eon_advisers_policy_provider }
  check_variable = { influence_array^3 = eon_advisers_policy_provider }
  check_variable = { influence_array^4 = eon_advisers_policy_provider }
  check_variable = { influence_array^5 = eon_advisers_policy_provider }
  check_variable = { influence_array^6 = eon_advisers_policy_provider }
 }
 var:eon_advisers_policy_provider = {
  exists = yes
  NOT = { tag = PREV }
  check_variable = { THIS.id = PREV.eon_advisers_policy_provider }
  OR = {
   has_idea = defense_industry
   num_of_military_factories > 10
   has_country_flag = can_sent_military_aid
  }
  if = {
   limit = { is_ai = yes }
   PREV = { has_government = eon_advisers_policy_provider }
  }
  if = {
   limit = { original_tag = ERI has_country_flag = ETH_transitional_government_FLAG }
   NOT = { has_country_leader = { name = "Eritrean Transitional Government" } }
  }
 }
}

eon_advisers_funds_available = {
 check_variable = { treasury >= 1.5 }
 check_variable = { treasury <= 1000000 }
}
# Recipient scope with explicitly named provider. No treasury is reserved.
eon_advisers_requirements_ready = {
 eon_advisers_incoming_slot_ready = yes
 var:eon_advisers_policy_provider = {
  eon_advisers_outgoing_slot_ready = yes
  eon_advisers_funds_available = yes
  NOT = { has_country_flag = military_services_sent_mercenaries }
 }
}
# Existing selection ROOT provider, FROM.FROM recipient.
eon_advisers_selection_ready = {
 exists = yes
 eon_advisers_pending_slot_ready = yes
 eon_advisers_outgoing_slot_ready = yes
 eon_advisers_funds_available = yes
 set_temp_variable = { eon_advisers_selection_provider = THIS }
 FROM.FROM = {
  set_temp_variable = { eon_advisers_policy_provider = PREV.eon_advisers_selection_provider }
  eon_advisers_policy_allowed = yes
  eon_advisers_requirements_ready = yes
  var:eon_advisers_policy_provider = {
   NOT = { has_country_flag = eon_advisers_retired_pair@PREV }
  }
 }
}
