# Persistent project proposals. This file does not change price or building policy.
# Country scope: investor. Different investors can independently propose to one recipient.
eon_investment_offer_valid = {
	exists = yes
	has_variable = eon_investment_offer_partner
	NOT = { check_variable = { eon_investment_offer_partner = 0 } }
	NOT = { check_variable = { eon_investment_offer_state = 0 } }
	check_variable = { eon_investment_offer_type > 0 }
	check_variable = { eon_investment_offer_type < 16 }
	check_variable = { eon_investment_offer_amount > 0 }
	check_variable = { eon_investment_offer_amount < 11 }
	check_variable = { eon_investment_offer_cost > 0 }
	check_variable = { eon_investment_offer_duration > 0 }
	check_variable = { treasury >= eon_investment_offer_cost }
	check_variable = { active_projects < 15 }
	is_in_array = { array = project_array value = 0 }
	NOT = { has_country_flag = disabled_foreign_investment }
	NOT = { has_country_flag = eon_investment_offer_withdrawn }
	var:eon_investment_offer_partner = {
		exists = yes
		NOT = { has_war_with = PREV }
		NOT = { has_country_flag = int_auto_reject_investment_flag }
		check_variable = { treasury >= PREV.eon_investment_offer_cofunding }
		PREV = { NOT = { has_war_with = PREV } NOT = { is_justifying_wargoal_against = PREV } }
	}
	var:eon_investment_offer_state = {
		NOT = { is_controlled_by = PREV }
		CONTROLLER = { exists = yes check_variable = { id = PREV.PREV.eon_investment_offer_partner } }
		check_variable = { projects_in_state^num = project_type_in_state^num }
		NOT = { is_in_array = { array = projects_in_state value = PREV.id } }
		eon_investment_snapshot_building_available = yes
	}
}

# Country-event context: ROOT/current is recipient, FROM is original donor.
eon_investment_response_matches = {
	FROM = { check_variable = { eon_investment_offer_partner = PREV.id } }
}
eon_investment_response_valid = {
	eon_investment_response_matches = yes
	FROM = { eon_investment_offer_valid = yes }
}
