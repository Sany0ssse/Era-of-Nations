# Visegrad Confederation
autonomy_state = {
	id = autonomy_visegrad_subject

	default = no
	is_puppet = no
	use_overlord_color = yes
	min_freedom_level = 0.0
	manpower_influence = 0.1

	rule = {
		can_not_declare_war = yes
		can_decline_call_to_war = no
		units_deployed_to_overlord = yes
		can_send_volunteers = yes
	}

	modifier = {
		cic_to_overlord_factor = 0
		mic_to_overlord_factor = 0.45
		can_master_build_for_us = 1
		autonomy_manpower_share = 0.1
		extra_trade_to_overlord_factor = 0.4
		overlord_trade_cost_factor = -0.4
		research_sharing_per_country_bonus_factor = -0.4
	}

	ai_subject_wants_higher = {
		factor = -999
	}

	ai_overlord_wants_lower = {
		factor = -999
	}

	ai_overlord_wants_garrison = {
		always = yes
	}

	allowed = {
		OVERLORD = {
			OR = {
				original_tag = POL
				original_tag = HUN
				original_tag = SLO
				original_tag = CZE
			}
			has_country_flag = visegrad_leader
		}
		OR = {
			original_tag = POL
			original_tag = HUN
			original_tag = SLO
			original_tag = CZE
		}
	}

	can_take_level = {
		always = no
	}

	can_lose_level = {
		always = no
	}
}
