autonomy_state = {
	id = autonomy_autonomous_state_colored
	use_overlord_color = yes
	default = no
	is_puppet = yes
	min_freedom_level = 0.0
	manpower_influence = 0.1

	rule = {
		can_not_declare_war = yes
		can_decline_call_to_war = no
		units_deployed_to_overlord = yes
	}

	modifier = {
		cic_to_overlord_factor = 0.75
		mic_to_overlord_factor = 0.75
		can_master_build_for_us = 1
		autonomy_manpower_share = 0.1
		extra_trade_to_overlord_factor = 0.8
		overlord_trade_cost_factor = -0.8
		research_sharing_per_country_bonus_factor = -0.8
		master_ideology_drift = 0.22
	}

	ai_subject_wants_higher = {
		factor = 1.0
	}

	ai_overlord_wants_lower = {
		factor = 1.0
	}

	ai_overlord_wants_garrison = {
		always = yes
	}

	allowed = {
		OVERLORD = { NOT = { is_in_array = { ruling_party = 11 } } }
		OVERLORD = { NOT = { has_country_flag = formed_neo_baathist_uar } }
		OVERLORD = { NOT = { has_country_flag = formed_baathist_uar } }
		has_global_flag = GAME_RULE_allow_colored_puppets
		NOT = {
			has_country_flag = GENERIC_autonomy_federation_flag
		}
	}

	# Conditions to "move closer to freedom"
	# THIS = { } is the subject nation
	# OVERLORD = { } is the overlord nation
	can_take_level = {
		if = { limit = { original_tag = PKK }
			OVERLORD = { NOT = { has_country_flag = TUR_PKK_federalized } }
		}
		if = { limit = { original_tag = ERI }
			OVERLORD = { NOT = { has_country_flag = ETH_locked_federalisation_FLAG } }
		}
		if = {
			limit = {
				OR = {
					original_tag = IKR
					original_tag = BLC
					original_tag = ARA
					original_tag = LOR
				}
			}
			OVERLORD = { NOT = { has_country_flag = PER_federalized_republic } }
		}
	}

	# Conditions to Annex via Puppet System
	can_lose_level = {
		OVERLORD = { NOT = { has_country_flag = integration_of_institutions_idea } }
		if = {
			limit = {
				OR = {
					original_tag = IKR
					original_tag = BLC
					original_tag = ARA
					original_tag = LOR
				}
			}
			OVERLORD = { NOT = { has_country_flag = PER_federalized_republic } }
		}
		if = { limit = { original_tag = PKK }
			OVERLORD = { NOT = { has_country_flag = TUR_PKK_federalized } }
		}
		if = {
			limit = { OR = { original_tag = KUR original_tag = PUK } }
			OVERLORD = { NOT = { has_country_flag = IRQ_autonomous_kurdistan } }
		}
		if = { limit = { original_tag = ERI }
			OVERLORD = { NOT = { has_idea = ETH_eritrean_constitution } }
		}
	}
}
