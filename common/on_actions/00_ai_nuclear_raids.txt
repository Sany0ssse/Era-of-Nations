on_startup = {  ##count nuclear potential
	effect = {
		every_country = {
		 limit = { has_capitulated = no }
		 count_nuclear_potential = yes
		}
	}
}

on_startup = { ##adding war_with_nukestate_idea
	effect = {
		every_country = {
			limit = {
				any_country = {
					has_war_with = PREV
					has_capitulated = no
					OR = {
					   check_variable = { num_equipment@nuclear_missile_equipment > 0 }
					   check_variable = { num_equipment@nuclear_ballistic_missile_equipment > 0 }
					}
					OR = {
						any_controlled_state = {
							OR = {
								check_variable = { non_damaged_building_level@rocket_site > 0 }
								AND = {
									check_variable = { non_damaged_building_level@air_base > 0 }
									check_variable = { num_equipment@strategic_bomber > 0 }
								}
							}
						}
						check_variable = { deployed_navy_manpower_k > 0 }
					}
					NOT = {
						has_idea = non_nuclear_power
						has_idea = nuclear_energy
					}
				}
			}
			add_ideas = War_with_nukestate_idea
		}
	}
}

on_daily = {
	effect = {
		# An old unbound target cannot borrow another callback's global shooter.
		if = {
			limit = {
				has_country_flag = sugoma
				OR = {
					has_capitulated = yes
					NOT = { has_global_flag = amogus }
					NOT = { has_variable = eon_nuclear_queued_shooter }
					NOT = { has_variable = eon_nuclear_queued_level }
					NOT = { var:eon_nuclear_queued_shooter = { exists = yes has_capitulated = no eon_nuclear_arsenal_ready = yes } }
				}
			}
			clr_country_flag = sugoma
			clr_country_flag = eon_nuclear_scripted_launch_succeeded
			clear_variable = eon_nuclear_queued_shooter
			clear_variable = eon_nuclear_queued_level
		}
		if = { limit = { has_country_flag = sussy_baka OR = { has_capitulated = yes NOT = { eon_nuclear_arsenal_ready = yes } } } clr_country_flag = sussy_baka }
		if = {
			limit = {
				has_global_flag = amogus
				has_country_flag = sugoma
				has_capitulated = no
				has_variable = eon_nuclear_queued_shooter
				has_variable = eon_nuclear_queued_level
				var:eon_nuclear_queued_shooter = { exists = yes has_capitulated = no eon_nuclear_arsenal_ready = yes }
			}
			# The helpers still take a transient global frame. Restore the incoming
			# frame after the attempt, including an originally absent variable.
			set_temp_variable = { eon_nuclear_daily_previous_has_shooter = 0 }
			set_temp_variable = { eon_nuclear_daily_previous_has_level = 0 }
			if = { limit = { has_variable = global.nuke_striker } set_temp_variable = { eon_nuclear_daily_previous_has_shooter = 1 } }
			if = { limit = { has_variable = global.temp_arctic_nuclear_level } set_temp_variable = { eon_nuclear_daily_previous_has_level = 1 } }
			set_temp_variable = { eon_nuclear_daily_previous_shooter = global.nuke_striker }
			set_temp_variable = { eon_nuclear_daily_previous_level = global.temp_arctic_nuclear_level }
			set_temp_variable = { eon_nuclear_daily_queue_shooter = eon_nuclear_queued_shooter }
			set_variable = { global.nuke_striker = eon_nuclear_queued_shooter }
			set_variable = { global.temp_arctic_nuclear_level = eon_nuclear_queued_level }
			launch_nuclear_ai_country_strike = yes
			if = { limit = { check_variable = { eon_nuclear_daily_previous_has_shooter > 0 } } set_variable = { global.nuke_striker = eon_nuclear_daily_previous_shooter } }
			else = { clear_variable = global.nuke_striker }
			if = { limit = { check_variable = { eon_nuclear_daily_previous_has_level > 0 } } set_variable = { global.temp_arctic_nuclear_level = eon_nuclear_daily_previous_level } }
			else = { clear_variable = global.temp_arctic_nuclear_level }
			if = {
				limit = { has_country_flag = eon_nuclear_scripted_launch_succeeded }
				var:eon_nuclear_daily_queue_shooter = {
					if = {
						limit = { exists = yes has_capitulated = no eon_nuclear_arsenal_ready = yes has_country_flag = sussy_baka }
						set_country_flag = hrucker
						add_named_threat = { threat = 100 name = strat_nuclear_strike_wt_tt }
						# A paid but intercepted attempt still counts. Multiple queued
						# targets report once, on the shooter, after its first payment.
						clr_country_flag = sussy_baka
					}
				}
			}
			clr_country_flag = sugoma
			clear_variable = eon_nuclear_queued_shooter
			clear_variable = eon_nuclear_queued_level
		}
		if = {
			limit = {
				has_global_flag = amogus
				has_country_flag = sussy_baka
				has_capitulated = no
				NOT = { any_country = { has_country_flag = sugoma } }
