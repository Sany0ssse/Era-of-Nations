on_startup = { ##adding war_with_nukestate_idea
	effect = {
		every_country = {
			limit = {
				any_country = {
					has_war_with = PREV
					has_capitulated = no
					OR = {
					   check_variable = { num_equipment@nuclear_missile_equipment > 0 }
					   check_variable = { num_equipment@nuclear_ballistic_missile_equipment > 0 }
					}
					OR = {
						any_controlled_state = {
							OR = {
								check_variable = { non_damaged_building_level@rocket_site > 0 }
								AND = {
									check_variable = { non_damaged_building_level@air_base > 0 }
									check_variable = { num_equipment@strategic_bomber > 0 }
								}
							}
						}
						check_variable = { deployed_navy_manpower_k > 0 }
					}
					NOT = {
						has_idea = non_nuclear_power
						has_idea = nuclear_energy
					}
				}
			}
			add_ideas = War_with_nukestate_idea
		}
	}
}

 effect = {
   if = {

			has_war = yes
			is_ai = yes
			has_capitulated = no
			NOT = {
				has_civil_war = yes
				AND = {
					has_country_flag = Custom_Civil_War
					has_war = yes
				}
			}
			OR = {
			   check_variable = { num_equipment@nuclear_missile_equipment > 9 }
			   check_variable = { num_equipment@nuclear_ballistic_missile_equipment > 9 }
			}
			OR = {
				any_controlled_state = {
					OR = {
						check_variable = { non_damaged_building_level@rocket_site > 0 }
						AND = {
							check_variable = { non_damaged_building_level@air_base > 0 }
							check_variable = { num_equipment@strategic_bomber > 0 }
						}
					}
				}
				check_variable = { deployed_navy_manpower_k > 0 }
			}
			surrender_progress > 0.49
			NOT = {
				   has_idea = non_nuclear_power
				   has_idea = nuclear_energy
				  }
		    }
	set_variable = { global.banner = THIS }
	set_country_flag = sugomusus
	set_country_flag = hrucker
	every_other_country = {
		limit = {
			has_war_with = var:global.banner
			has_capitulated = no
		    check_variable = { nukes_strikes_count = 0 }
			any_controlled_state = {
				NOT = {
					has_dynamic_modifier = {
						modifier = nuclear_fallout
					}
				}
				NOT = {
					has_dynamic_modifier = {
						modifier = nuclear_fallout1
					}
				}
				NOT = {
					has_dynamic_modifier = {
						modifier = nuclear_fallout2
					}
				}
				NOT = {
					has_dynamic_modifier = {
						modifier = nuclear_fallout3
					}
				}
			}
		}
		set_variable = { global.temp_arctic_nuclear_level = PREV.arctic_nuclear_level }
		set_temp_variable = { global.nuke_striker = PREV }
		launch_nuclear_ai_country_strike = yes
		add_to_variable = { nukes_strikes_count = 1 }
	}
	clear_variable = global.banner
                    }
	      }
	         }
