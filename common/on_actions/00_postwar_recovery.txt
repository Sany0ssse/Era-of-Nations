	on_startup = {
		effect = {
			every_country = {
				set_variable = { prev_stability = stability }
				set_variable = { prev_productivity = overall_productivity }
				set_variable = { prev_num_of_factories = industrial_complex_total }
				set_variable = { prev_num_of_damaged_factories = num_of_damaged_civs }
			}
		}

	on_war = {
		effect = {
			if = {
				limit = { has_idea = postwar_restoration }
				remove_ideas = postwar_restoration

			set_variable = { prev_stability = stability }
			set_variable = { prev_productivity = overall_productivity }
			set_variable = { prev_num_of_factories = industrial_complex_total }
			set_variable = { prev_num_of_damaged_factories = num_of_damaged_civs }
		}
	}
	on_peace = {
		effect = {
			set_temp_variable = { stability_postwar_rate = prev_stability }
			divide_temp_variable = { stability_postwar_rate = stability }
			set_temp_variable = { productivity_postwar_rate = prev_productivity }
			divide_temp_variable = { productivity_postwar_rate = overall_productivity }
			set_temp_variable = { factories_postwar_rate = prev_num_of_factories }
			divide_temp_variable = { factories_postwar_rate = industrial_complex_total }
			set_temp_variable = { factories_fixing_postwar_rate = num_of_damaged_civs }
			divide_temp_variable = { factories_fixing_postwar_rate = prev_num_of_damaged_factories }
			if = {
				limit = {
					OR = {
						check_variable = { stability_postwar_rate > 2 }
						check_variable = { productivity_postwar_rate > 2 }
						check_variable = { factories_postwar_rate > 2 }
						check_variable = { factories_fixing_postwar_rate > 2 }
					}
				}
				add_timed_idea = {
					idea = postwar_restoration
					days = 120
				}
				add_offsite_building = {
					type = industrial_complex
					level = 2
				}
			}
		}
	}
