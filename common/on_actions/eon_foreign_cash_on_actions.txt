on_actions = {
 on_daily = { effect = { eon_foreign_cash_daily_update = yes } }
 # Native ROOT annexer, FROM annexed country.
 on_annex = {
  effect = {
   every_country = {
    limit = { exists = yes NOT = { tag = FROM } }
    set_temp_variable = { eon_foreign_cash_annexed_partner = FROM }
    eon_foreign_cash_annex_update = yes
   }
   FROM = { eon_foreign_cash_clear_annexed_country = yes }
  }
 }
 # Native ROOT annexed subject, FROM overlord.
 on_subject_annexed = {
  effect = {
   every_country = {
    limit = { exists = yes NOT = { tag = ROOT } }
    set_temp_variable = { eon_foreign_cash_annexed_partner = ROOT }
    eon_foreign_cash_annex_update = yes
   }
   ROOT = { eon_foreign_cash_clear_annexed_country = yes }
  }
 }
}
