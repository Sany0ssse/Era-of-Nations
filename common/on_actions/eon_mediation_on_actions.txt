on_actions = {
	# Native daily scope is the current country.
	on_daily = { effect = { eon_mediation_daily_cleanup = yes } }

	# ROOT annexer, FROM annexed country.
	on_annex = {
		effect = {
			every_country = {
				limit = { exists = yes NOT = { tag = FROM } }
				set_temp_variable = { eon_mediation_annexed_partner = FROM }
				eon_mediation_cleanup_annexed_pair = yes
			}
			FROM = { eon_mediation_cleanup_annexed_owner = yes }
		}
	}

	# Native subject integration: ROOT annexed subject, FROM its overlord.
	on_subject_annexed = {
		effect = {
			every_country = {
				limit = { exists = yes NOT = { tag = ROOT } }
				set_temp_variable = { eon_mediation_annexed_partner = ROOT }
				eon_mediation_cleanup_annexed_pair = yes
			}
			ROOT = { eon_mediation_cleanup_annexed_owner = yes }
		}
	}
}
