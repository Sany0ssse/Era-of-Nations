							AND = {
								check_variable = { ammo_production_rating < 0 }
								NOT = { check_variable = { ammo_stock_years < 2 } }
							}
						}
					}
					if = {
						limit = { has_idea = heavy_ammo_restrictions_spirit }
						remove_ideas = heavy_ammo_restrictions_spirit
						add_ideas = some_ammo_restrictions_spirit
					}
					else_if = {
						limit = { has_idea = some_ammo_restrictions_spirit }
						remove_ideas = some_ammo_restrictions_spirit
					}
					else_if = {
						limit = {
							NOT = { has_idea = heavy_ammo_restrictions_spirit }
							NOT = { has_idea = some_ammo_restrictions_spirit }
							NOT = { has_idea = some_ammo_additional_consumption_spirit }
							NOT = { has_idea = heavy_ammo_additional_consumption_spirit }
						}
						add_ideas = some_ammo_additional_consumption_spirit
					}
					else = {
						remove_ideas = some_ammo_additional_consumption_spirit
						add_ideas = heavy_ammo_additional_consumption_spirit
					}
				}
				else_if = {
					limit = {
						check_variable = { ammo_production_rating < 0 }
						check_variable = { ammo_stock_years < 2 }
						NOT = { check_variable = { ammo_stock = max_ammo_limit } }
					}
					if = {
						limit = { has_idea = heavy_ammo_additional_consumption_spirit }
						remove_ideas = heavy_ammo_additional_consumption_spirit
						add_ideas = some_ammo_additional_consumption_spirit
					}
					else_if = {
						limit = { has_idea = some_ammo_additional_consumption_spirit }
						remove_ideas = some_ammo_additional_consumption_spirit
					}
					else_if = {
						limit = {
							NOT = { has_idea = heavy_ammo_restrictions_spirit }
							NOT = { has_idea = some_ammo_restrictions_spirit }
							NOT = { has_idea = some_ammo_additional_consumption_spirit }
							NOT = { has_idea = heavy_ammo_additional_consumption_spirit }
						}
						add_ideas = some_ammo_restrictions_spirit
					}
					else = {
						remove_ideas = some_ammo_restrictions_spirit
						add_ideas = heavy_ammo_restrictions_spirit
					}
				}
			}
			if = {
				limit = {
					is_ai = yes
					OR = {
						has_country_flag = EU_candidate
						has_idea = EU_member
					}
				}
				update_eu_dirty_variable = yes
				if = {
					limit = {
						NOT = { has_country_flag = show_offices_screen }
						NOT = { has_country_flag = show_eu_parliament }
						NOT = { has_country_flag = eu_sec_dis_open }
						NOT = { has_country_flag = show_euro_info }
						NOT = { has_country_flag = show_euro_voting }
						NOT = { has_country_flag = show_eu_fiscal }
					}
					set_country_flag = show_offices_screen
				}
				else_if = {
					limit = {
						has_country_flag = show_offices_screen
					}
					clr_country_flag = show_offices_screen
					set_country_flag = show_eu_parliament
					set_country_flag = eu_sec_dis_open
				}
				else_if = {
					limit = {
						has_country_flag = show_eu_parliament
					}
					clr_country_flag = eu_sec_dis_open
					clr_country_flag = show_eu_parliament
					set_country_flag = show_euro_info
				}
				else_if = {
					limit = {
						has_country_flag = show_euro_info
					}
					clr_country_flag = show_euro_info
					set_country_flag = show_euro_voting
				}
				else_if = {
					limit = {
						has_country_flag = show_euro_voting
					}
					clr_country_flag = show_euro_voting
					set_country_flag = show_eu_fiscal
				}
				else_if = {
					limit = {
						has_country_flag = show_eu_fiscal
					}
					clr_country_flag = show_eu_fiscal
					set_country_flag = show_offices_screen
				}
			}
		}
