AI_Rocket_Site = {
	enable = {
		OR = {
			threat > 0.5
			has_global_flag = age_has_begun_4
		}
		num_of_available_civilian_factories > 15
		OR = {
			check_variable = { num_equipment@guided_missile_equipment > 5000 }
			check_variable = { num_equipment@hypersonic_missile_equipment > 5000 }
			check_variable = { num_equipment@ballistic_missile_equipment > 5000 }
			check_variable = { num_equipment@sam_missile_equipment > 5000 }
			AND = {
				check_variable = { num_equipment@anti_ship_missile_equipment > 5000 }
				any_owned_state = { is_coastal = yes }
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = building_target id = rocket_site value = 100 }
}
