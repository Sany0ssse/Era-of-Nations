formable_nations_integration_script = {
    set_country_flag = { flag = integration_of_institutions_idea value = 1 days = 100 }

    set_temp_variable = { tag_index = LIT.id }
	set_temp_variable = { pp_cost = -100 }
	set_temp_variable = { treasury_cost = -3 }
	set_temp_variable = { popularity_change = -0.03 }
	formable_nations_integration_effect_script = yes

    set_temp_variable = { tag_index = LAT.id }
	set_temp_variable = { pp_cost = -100 }
	set_temp_variable = { treasury_cost = -3 }
	set_temp_variable = { popularity_change = -0.03 }
	formable_nations_integration_effect_script = yes

    set_temp_variable = { tag_index = EST.id }
	set_temp_variable = { pp_cost = -100 }
	set_temp_variable = { treasury_cost = -3 }

	formable_nations_integration_effect_script = yes
}

formable_nations_integration_effect_script = {
	if = {
		limit = {
			var:tag_index = {
				influence_higher_80 = yes
				is_ai = yes
				is_subject_of = ROOT
                has_war = no
                has_civil_war = no
			}
			country_exists = var:tag_index
		}
		add_political_power = pp_cost
		set_temp_variable = { treasury_change = treasury_cost }
		modify_treasury_effect = yes
		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = popularity_change }
		set_temp_variable = { temp_outlook_increase = popularity_change }
		add_relative_party_popularity = yes
		var:tag_index = {
			ROOT = {
				annex_country = { target = PREV transfer_troops = yes }
			}
			every_core_state = { add_core_of = ROOT }
		}
		hidden_effect = {
			var:tag_index = {
				every_unit_leader = {
					set_nationality = ROOT
				}
				add_to_variable = { PREV.treasury = treasury }
				add_to_variable = { PREV.debt = debt }
	            clamp_variable = { var = PREV.debt min = 0 max = 1000000 }
				add_to_variable = { PREV.int_investments = int_investments }
			}
            if = {
                limit = {
                    AND = {
                        OR = {
                            original_tag = LIT
                            original_tag = LAT
                            original_tag = EST
                        }
                    LIT = {
                        all_core_state = {
                            is_core_of = ROOT
                        }
                    }
                    LAT = {
                        all_core_state = {
                        is_core_of = ROOT
                        }
                    }
                    EST = {
                        all_core_state = {
                        is_core_of = ROOT
                        }
                    }
                    }
                }
                formable_nations_created_countries = yes
            }
		}
	}
}
