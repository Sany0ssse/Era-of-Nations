eon_eu_recount_office_ballot = {
 # Root helper recomputes actual state-resident population, independent of conscription.
 eon_eu_recount_council_vote = yes
 set_variable = { global.eon_eu_office_voters = 0 }
 set_variable = { global.eon_eu_office_population_k = 0 }
 set_variable = { global.eon_eu_office_yes_population_k = 0 }
 set_variable = { global.eon_eu_office_yes = 0 }
 set_variable = { global.eon_eu_office_no = 0 }
 set_variable = { global.eon_eu_office_ep_total = 0 }
 set_variable = { global.eon_eu_office_ep_yes = 0 }
 set_variable = { global.eon_eu_office_ep_no = 0 }
 for_each_scope_loop = {
  array = global.EU_member
  if = {
   limit = { eon_eu_is_active_member = yes }
   add_to_variable = { global.eon_eu_office_ep_total = THIS.MEP_Total }
   if = { limit = { is_in_array = { array = global.eon_eu_office_supporters value = THIS } }
    add_to_variable = { global.eon_eu_office_ep_yes = THIS.MEP_Total }
   }
   else_if = { limit = { is_in_array = { array = global.eon_eu_office_rejecters value = THIS } }
    add_to_variable = { global.eon_eu_office_ep_no = THIS.MEP_Total }
   }
   if = {
    limit = {
     NOT = { has_idea = EUU_voting_rights_suspended }
     OR = { NOT = { check_variable = { global.eon_eu_office_id = 3 } } has_idea = the_euro }
    }
    add_to_variable = { global.eon_eu_office_voters = 1 }
    add_to_variable = { global.eon_eu_office_population_k = THIS.eon_eu_voting_population_k }
    if = { limit = { is_in_array = { array = global.eon_eu_office_supporters value = THIS } }
     add_to_variable = { global.eon_eu_office_yes = 1 }
     add_to_variable = { global.eon_eu_office_yes_population_k = THIS.eon_eu_voting_population_k }
    }
    else_if = { limit = { is_in_array = { array = global.eon_eu_office_rejecters value = THIS } }
     add_to_variable = { global.eon_eu_office_no = 1 }
    }
   }
  }
 }
 set_temp_variable = { eon_eu_office_state_threshold = global.eon_eu_office_voters }
 multiply_temp_variable = { eon_eu_office_state_threshold = 0.72 }
 set_temp_variable = { eon_eu_office_population_threshold = global.eon_eu_office_population_k }
 multiply_temp_variable = { eon_eu_office_population_threshold = 0.65 }
 set_temp_variable = { eon_eu_office_non_supporters = global.eon_eu_office_voters }
 subtract_from_temp_variable = { eon_eu_office_non_supporters = global.eon_eu_office_yes }
 set_temp_variable = { eon_eu_office_ep_threshold = global.eon_eu_office_ep_total }
 multiply_temp_variable = { eon_eu_office_ep_threshold = 0.5 }
 set_temp_variable = { eon_eu_office_board_total_times_two = global.eon_eu_office_voters }
 multiply_temp_variable = { eon_eu_office_board_total_times_two = 2 }
 set_temp_variable = { eon_eu_office_board_yes_times_three = global.eon_eu_office_yes }
 multiply_temp_variable = { eon_eu_office_board_yes_times_three = 3 }
}
eon_eu_resolve_office_ballot = {
 if = {
  limit = { check_variable = { global.eon_eu_office_id > 0 } NOT = { has_global_flag = eon_eu_office_ballot_deadline } }
  set_temp_variable = { eon_eu_office_requested_id = global.eon_eu_office_id }
  set_temp_variable = { eon_eu_office_ballot_valid = 0 }
  var:global.eon_eu_office_nominee = {
   if = {
    limit = {
     eon_eu_office_nominee_eligible = yes
     eon_eu_office_vacant = yes
     check_variable = { var = eon_eu_office_nominee_generation value = global.eon_eu_office_nominee_generation compare = equals }
    }
    set_temp_variable = { eon_eu_office_ballot_valid = 1 }
   }
  }
  eon_eu_recount_office_ballot = yes
  set_temp_variable = { eon_eu_office_ballot_passed = 0 }
  if = {
   limit = { check_variable = { eon_eu_office_ballot_valid = 1 } }
   if = { limit = { OR = { check_variable = { global.eon_eu_office_id = 2 } AND = { check_variable = { global.eon_eu_office_id = 1 } check_variable = { global.eon_eu_office_phase = 2 } } } eon_eu_office_ep_majority_passed = yes }
    set_temp_variable = { eon_eu_office_ballot_passed = 1 }
   }
   else_if = { limit = { check_variable = { global.eon_eu_office_id = 6 } eon_eu_office_board_passed = yes }
    set_temp_variable = { eon_eu_office_ballot_passed = 1 }
   }
   else_if = { limit = { NOT = { OR = { check_variable = { global.eon_eu_office_id = 2 } check_variable = { global.eon_eu_office_id = 6 } AND = { check_variable = { global.eon_eu_office_id = 1 } check_variable = { global.eon_eu_office_phase = 2 } } } } eon_eu_office_enhanced_qmv_passed = yes }
    set_temp_variable = { eon_eu_office_ballot_passed = 1 }
   }
  }
  if = {
   limit = { check_variable = { eon_eu_office_ballot_passed = 1 } }
   if = { limit = { check_variable = { global.eon_eu_office_id = 1 } check_variable = { global.eon_eu_office_phase = 1 } }
    set_variable = { global.eon_eu_office_phase = 2 }
    clear_array = global.eon_eu_office_supporters
    clear_array = global.eon_eu_office_rejecters
    set_global_flag = { flag = eon_eu_office_ballot_deadline value = 1 days = 14 }
    eon_eu_recount_office_ballot = yes
    eon_eu_notify_office_ballot = yes
   }
   else = { eon_eu_appoint_current_office = yes }
  }
  else = {
   var:global.eon_eu_office_nominee = {
    set_country_flag = { flag = eon_eu_office_nomination_cooldown value = 1 days = 60 }
    if = { limit = { is_ai = no } country_event = { id = eon_eu_institutions.3 } }
   }
   eon_eu_clear_office_ballot = yes
   set_global_flag = { flag = eon_eu_office_appointment_cooldown value = 1 days = 7 }
  }
  update_eu_dirty_variable = yes
 }
}

eon_eu_ai_office_voting = {
 for_each_scope_loop = {
  array = global.EU_member
  if = {
   limit = {
    eon_eu_can_cast_office_ballot = yes is_ai = yes
    NOT = { is_in_array = { array = global.eon_eu_office_supporters value = THIS } }
    NOT = { is_in_array = { array = global.eon_eu_office_rejecters value = THIS } }
   }
   if = {
    limit = {
     check_variable = { eurosceptic < 0.6 }
     # Resolve the nominee, then return to the voter; PREV is now the nominee.
     var:global.eon_eu_office_nominee = { PREV = { has_opinion = { target = PREV value > -25 } } }
    }
    eon_eu_cast_office_yes = yes
   }
   else = { eon_eu_cast_office_no = yes }
  }

eon_eu_initialize_historical_offices = {
 # 2000 historical origins: Prodi(ITA), Fontaine(FRA), Duisenberg(HOL), Solana(SPR).
 # Existing holders are never overwritten; these are origin proxies, not governments' control.
 if = {
  limit = { NOT = { has_global_flag = eon_eu_institutions_initialized } }
  set_global_flag = eon_eu_institutions_initialized
  for_each_scope_loop = { array = global.EU_member if = { limit = { check_variable = { eon_eu_office_nominee_generation < 1 } } set_variable = { eon_eu_office_nominee_generation = 1 } } }
  # Adopt old-save holders without restarting or extending their existing timed ideas.
  for_each_scope_loop = {
   array = global.EU_member
   if = { limit = { has_idea = ECB_president check_variable = { eon_eu_ecb_appointed_generation < 1 } }
    set_variable = { eon_eu_ecb_appointed_generation = eon_eu_office_nominee_generation }
   }
   if = { limit = { has_idea = EU_council_president check_variable = { eon_eu_council_served_generation < 1 } }
    set_variable = { eon_eu_council_served_generation = eon_eu_office_nominee_generation }
    set_variable = { eon_eu_council_nominee_terms = 1 }
   }
   if = { limit = { has_idea = frontex_ExDir check_variable = { eon_eu_frontex_served_generation < 1 } }
    set_variable = { eon_eu_frontex_served_generation = eon_eu_office_nominee_generation }
    set_variable = { eon_eu_frontex_nominee_terms = 1 }
   }
  }
  if = { limit = { date < 2000.2.1 }
   if = { limit = { NOT = { any_of_scopes = { array = global.EU_member has_idea = EU_commission_president } } }
    ITA = { if = { limit = { eon_eu_is_active_member = yes } eon_eu_appoint_office_1 = yes } }
   }
   if = { limit = { NOT = { any_of_scopes = { array = global.EU_member has_idea = EU_parliament_president } } }
    FRA = { if = { limit = { eon_eu_is_active_member = yes } eon_eu_appoint_office_2 = yes } }
   }
   if = { limit = { NOT = { any_of_scopes = { array = global.EU_member has_idea = ECB_president } } }
    HOL = { if = { limit = { eon_eu_is_active_member = yes has_idea = the_euro } eon_eu_appoint_office_3 = yes } }
   }
   if = { limit = { NOT = { any_of_scopes = { array = global.EU_member has_idea = EU_foreign_minister } } }
    SPR = { if = { limit = { eon_eu_is_active_member = yes } eon_eu_appoint_office_5 = yes } }
   }
  }
 }
}
