# Era of Nations: canonical ballots. Country scope tokens are opaque values.
# All counters are rebuilt from live members; cached increments are not receipts.
eon_eu_recount_council_vote = {
	set_variable = { global.eon_eu_voters = 0 }
	set_variable = { global.eon_eu_voting_population_k = 0 }
	set_variable = { global.EU_active_vote_yes = 0 }
	set_variable = { global.EU_active_vote_no = 0 }
	set_variable = { global.EU_pop_ratio_yes = 0 }
	set_variable = { global.EU_pop_ratio_no = 0 }
	set_variable = { global.eon_eu_yes_fraction = 0 }
	set_variable = { global.eon_eu_opposing_members = 0 }
	set_variable = { global.eon_eu_unanswered_members = 0 }
	clear_array = global.eon_eu_counted_supporters
	clear_array = global.eon_eu_counted_rejecters
	clear_array = global.eon_eu_counted_abstainers
	every_country = {
		limit = { eon_eu_can_vote = yes }
		add_to_variable = { global.eon_eu_voters = 1 }
		set_variable = { eon_eu_voting_population_k = 0 }
		every_owned_state = {
			add_to_variable = { PREV.eon_eu_voting_population_k = state_population_k }
		}
		add_to_variable = { global.eon_eu_voting_population_k = eon_eu_voting_population_k }
		if = { limit = { is_in_array = { array = global.EU_active_vote_supporters value = THIS } }
			add_to_array = { array = global.eon_eu_counted_supporters value = THIS }
			add_to_variable = { global.EU_active_vote_yes = 1 }
			add_to_variable = { global.EU_pop_ratio_yes = eon_eu_voting_population_k }
		}
		else_if = { limit = { is_in_array = { array = global.EU_active_vote_rejecters value = THIS } }
			add_to_array = { array = global.eon_eu_counted_rejecters value = THIS }
			add_to_variable = { global.EU_active_vote_no = 1 }
			add_to_variable = { global.EU_pop_ratio_no = eon_eu_voting_population_k }
		}
		else_if = { limit = { is_in_array = { array = global.eon_eu_council_abstainers value = THIS } }
			add_to_array = { array = global.eon_eu_counted_abstainers value = THIS }
		}
		else = { add_to_variable = { global.eon_eu_unanswered_members = 1 } }
	}
	set_variable = { global.eon_eu_opposing_members = global.eon_eu_voters }
	subtract_from_variable = { global.eon_eu_opposing_members = global.EU_active_vote_yes }
	if = { limit = { check_variable = { global.eon_eu_voters > 0 } }
		set_variable = { global.eon_eu_yes_fraction = global.EU_active_vote_yes }
		divide_variable = { global.eon_eu_yes_fraction = global.eon_eu_voters }
	}
	if = { limit = { check_variable = { global.eon_eu_voting_population_k > 0 } }
		divide_variable = { global.EU_pop_ratio_yes = global.eon_eu_voting_population_k }
		divide_variable = { global.EU_pop_ratio_no = global.eon_eu_voting_population_k }
	}
	clear_array = global.EU_active_vote_supporters
	clear_array = global.EU_active_vote_rejecters
	clear_array = global.eon_eu_council_abstainers
	for_each_scope_loop = { array = global.eon_eu_counted_supporters
		add_to_array = { array = global.EU_active_vote_supporters value = THIS }
	}
	for_each_scope_loop = { array = global.eon_eu_counted_rejecters
		add_to_array = { array = global.EU_active_vote_rejecters value = THIS }
	}
	for_each_scope_loop = { array = global.eon_eu_counted_abstainers
		add_to_array = { array = global.eon_eu_council_abstainers value = THIS }
	}
	clear_array = global.eon_eu_counted_supporters
	clear_array = global.eon_eu_counted_rejecters
	clear_array = global.eon_eu_counted_abstainers

 eon_eu_record_open_council_tally = yes
}

eon_eu_support_council_vote = {
	if = { limit = { eon_eu_can_cast_council_vote = yes }
		remove_from_array = { array = global.EU_active_vote_rejecters value = THIS }
		remove_from_array = { array = global.eon_eu_council_abstainers value = THIS }
		if = { limit = { NOT = { is_in_array = { array = global.EU_active_vote_supporters value = THIS } } }
			add_to_array = { array = global.EU_active_vote_supporters value = THIS }
		}
		eon_eu_recount_council_vote = yes
		update_eu_dirty_variable = yes
	}
}

			add_to_array = { array = global.EU_active_vote_rejecters value = THIS }
		}

		update_eu_dirty_variable = yes

eon_eu_start_council_proposal = {
 if = { limit = { eon_eu_council_proposal_ready = yes }
  eon_eu_handoff_snapshot_entry = yes
  if = { limit = { NOT = { eon_eu_council_proposal_free = yes } } add_political_power = -500 }
  set_variable = { global.current_active_vote_disp = v }
  set_variable = { global.EUXXX_council_agenda = v }
  set_variable = { global.eon_eu_council_round_agenda = v }
  add_to_variable = { global.eon_eu_council_vote_instance = 1 }
  set_variable = { global.eon_eu_council_outcome_instance = -1 }
  set_variable = { global.eon_eu_council_vote_owner = global.eucomission }
  set_global_flag = eon_eu_council_vote_clock_started
  set_global_flag = { flag = eon_eu_council_vote_deadline days = 14 value = 1 }
  set_variable = { global.eon_eu_council_vote_deadline_day = global.num_days }
  add_to_variable = { global.eon_eu_council_vote_deadline_day = 14 }
  eon_eu_handoff_claim_entry = yes
  clear_array = global.EU_active_vote_supporters
  clear_array = global.EU_active_vote_rejecters
  clear_array = global.eon_eu_council_abstainers
  every_country = { clr_country_flag = current_vote_influenced }
  eon_eu_mark_current_council_agenda = yes
  for_each_scope_loop = { array = global.EU_member
   if = { limit = { eon_eu_is_active_member = yes }
    set_variable = { eon_eu_council_mission_instance = global.eon_eu_council_vote_instance }
    set_variable = { eon_eu_council_mission_agenda = global.current_active_vote_disp }
    if = { limit = { check_variable = { var = global.eon_eu_council_vote_owner value = THIS.id compare = equals } }
     activate_mission = EU_voting_mission_focus_EUXXX_QMV_result
    }
    else = { activate_mission = EU_voting_mission_focus_EUXXX_QMV_result_timer }
    if = { limit = { is_ai = no } country_event = { id = EUvote.2 } }
   }
  }
  if = { limit = { is_ai = no } set_global_flag = EUU_player_proposed_law }
  eon_eu_council_ai_pulse = yes
  clr_global_flag = EUU_player_proposed_law
  eon_eu_recount_council_vote = yes
  EU_show_display_effects_for_law = yes
  update_eu_dirty_variable = yes
 }
}

eon_eu_reconcile_council_vote_owner = {
 eon_eu_migrate_legacy_council_round = yes
 eon_eu_cleanup_orphan_council_round = yes
 eon_eu_capture_council_deadline_outcome = yes
 if = { limit = { eon_eu_council_round_identity_current = yes }
  if = { limit = { var:global.eucomission = {
   eon_eu_is_active_member = yes has_idea = EU_commission_president
   is_in_array = { array = global.EU_member value = THIS }
  } }
   if = { limit = { NOT = { check_variable = { var = global.eon_eu_council_vote_owner value = global.eucomission compare = equals } } }
    var:global.eon_eu_council_vote_owner = { remove_mission = EU_voting_mission_focus_EUXXX_QMV_result }
    set_variable = { global.eon_eu_council_vote_owner = global.eucomission }
    var:global.eucomission = {
     set_variable = { eon_eu_council_mission_instance = global.eon_eu_council_vote_instance }
     set_variable = { eon_eu_council_mission_agenda = global.current_active_vote_disp }
     remove_mission = EU_voting_mission_focus_EUXXX_QMV_result_timer
     if = { limit = { has_global_flag = eon_eu_council_vote_deadline }
      activate_mission = EU_voting_mission_focus_EUXXX_QMV_result
      # num_days is a documented day counter, not the opaque global.date.
      # The natural positive flag remains authoritative if day rounding differs.
      set_temp_variable = { eon_eu_council_days_remaining = global.eon_eu_council_vote_deadline_day }
      subtract_from_temp_variable = { eon_eu_council_days_remaining = global.num_days }
      clamp_temp_variable = { var = eon_eu_council_days_remaining min = 1 max = 14 }
      subtract_from_temp_variable = { eon_eu_council_days_remaining = 14 }
      add_days_mission_timeout = { mission = EU_voting_mission_focus_EUXXX_QMV_result days = eon_eu_council_days_remaining }
     }
    }
   }
   if = { limit = { NOT = { has_global_flag = eon_eu_council_vote_deadline } }
    var:global.eucomission = { eon_eu_resolve_current_council_agenda = yes }
   }
  }
  # A vacancy retains both round and original deadline. Resume on lawful
  # appointment, never renew the14-day positiveflag or apply from an old holder.
 }

eon_eu_migrate_legacy_council_round = {
 if = { limit = {
  NOT = { has_global_flag = eon_eu_council_vote_clock_started }
  eon_eu_legacy_council_context = yes
 }
  set_variable = { global.eon_eu_legacy_council_clock_witnesses = 0 }
  set_variable = { global.eon_eu_legacy_council_remaining_days = 99999 }
  every_country = { limit = { eon_eu_is_active_member = yes is_in_array = { array = global.EU_member value = THIS } }
   if = { limit = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result }
    add_to_variable = { global.eon_eu_legacy_council_clock_witnesses = 1 }
    set_variable = { eon_eu_legacy_council_days = days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result }
    if = { limit = { check_variable = { var = eon_eu_legacy_council_days value = global.eon_eu_legacy_council_remaining_days compare = less_than } }
     set_variable = { global.eon_eu_legacy_council_remaining_days = eon_eu_legacy_council_days }
    }
   }
   if = { limit = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result_timer }
    add_to_variable = { global.eon_eu_legacy_council_clock_witnesses = 1 }
    set_variable = { eon_eu_legacy_council_days = days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result_timer }
    if = { limit = { check_variable = { var = eon_eu_legacy_council_days value = global.eon_eu_legacy_council_remaining_days compare = less_than } }
     set_variable = { global.eon_eu_legacy_council_remaining_days = eon_eu_legacy_council_days }
    }
   }
  }
  if = { limit = {
   has_global_flag = eon_eu_legacy_council_expired_callback
   check_variable = { var = global.eon_eu_legacy_council_expired_agenda value = global.current_active_vote_disp compare = equals }
   check_variable = { var = global.eon_eu_legacy_council_expired_instance value = global.eon_eu_council_vote_instance compare = equals }
  }
   add_to_variable = { global.eon_eu_legacy_council_clock_witnesses = 1 }
   set_variable = { global.eon_eu_legacy_council_remaining_days = 0 }
  }
  if = { limit = { check_variable = { global.eon_eu_legacy_council_clock_witnesses > 0 } }
   set_temp_variable = { eon_eu_legacy_open_snapshot_valid = 0 }
   if = { limit = {
    check_variable = { var = global.eon_eu_council_open_instance value = global.eon_eu_council_vote_instance compare = equals }
    check_variable = { var = global.eon_eu_council_open_agenda value = global.current_active_vote_disp compare = equals }
   } set_temp_variable = { eon_eu_legacy_open_snapshot_valid = 1 } }
   add_to_variable = { global.eon_eu_council_vote_instance = 1 }
   set_variable = { global.eon_eu_council_round_agenda = global.current_active_vote_disp }
   if = { limit = { check_variable = { eon_eu_legacy_open_snapshot_valid = 1 } }
    set_variable = { global.eon_eu_council_open_instance = global.eon_eu_council_vote_instance }
   }
   else = { set_variable = { global.eon_eu_council_open_instance = -1 } }
   set_variable = { global.eon_eu_council_outcome_instance = -1 }
   clr_global_flag = eon_eu_legacy_council_expired_callback
   set_global_flag = eon_eu_council_vote_clock_started
   set_variable = { global.eon_eu_council_vote_owner = global.eucomission }
   set_variable = { global.eon_eu_council_vote_deadline_day = global.num_days }
   if = { limit = { check_variable = { global.eon_eu_legacy_council_remaining_days > 0 } }
    set_global_flag = { flag = eon_eu_council_vote_deadline days = global.eon_eu_legacy_council_remaining_days value = 1 }
    add_to_variable = { global.eon_eu_council_vote_deadline_day = global.eon_eu_legacy_council_remaining_days }
   }
   else = { clr_global_flag = eon_eu_council_vote_deadline }
   for_each_scope_loop = { array = global.EU_member
    if = { limit = { eon_eu_is_active_member = yes }
     set_variable = { eon_eu_council_mission_instance = global.eon_eu_council_vote_instance }
     set_variable = { eon_eu_council_mission_agenda = global.current_active_vote_disp }
     if = { limit = { NOT = { check_variable = { var = global.eucomission value = THIS.id compare = equals } } }
      remove_mission = EU_voting_mission_focus_EUXXX_QMV_result
     }
    }
   }
   # Reconciliation arms a sole-timer witness on the live Commission without
   # another14day deadline, and preserves existing result mission if present.
   var:global.eucomission = {
    if = { limit = { has_global_flag = eon_eu_council_vote_deadline }
     if = { limit = { has_active_mission = EU_voting_mission_focus_EUXXX_QMV_result }
      set_temp_variable = { eon_eu_council_existing_days = days_mission_timeout@EU_voting_mission_focus_EUXXX_QMV_result }
     }
     else = {
      activate_mission = EU_voting_mission_focus_EUXXX_QMV_result
      set_temp_variable = { eon_eu_council_existing_days = 14 }
     }
     set_temp_variable = { eon_eu_council_days_remaining = global.eon_eu_legacy_council_remaining_days }
     subtract_from_temp_variable = { eon_eu_council_days_remaining = eon_eu_council_existing_days }
     add_days_mission_timeout = { mission = EU_voting_mission_focus_EUXXX_QMV_result days = eon_eu_council_days_remaining }
    }
   }
  }
  else = {
   # No real mission or genuine elapsed callback exists: this is an orphan,
   # not evidence that a new14day clock or an enacted law may be invented.
   eon_eu_cleanup_orphan_council_round = yes
  }
 }

eon_eu_capture_council_deadline_outcome = {
 if = { limit = {
  eon_eu_council_round_identity_current = yes
  NOT = { has_global_flag = eon_eu_council_vote_deadline }
  check_variable = { global.current_active_vote_disp > 0 }
  NOT = { check_variable = { var = global.eon_eu_council_outcome_instance value = global.eon_eu_council_vote_instance compare = equals } }
 }
  set_variable = { global.eon_eu_council_outcome_yes = 0 }
  if = { limit = {
   check_variable = { var = global.eon_eu_council_open_instance value = global.eon_eu_council_vote_instance compare = equals }
   check_variable = { var = global.eon_eu_council_open_agenda value = global.current_active_vote_disp compare = equals }
  }
   # Recounts after actual expiry cannot rewrite the last eligible open tally.
   set_variable = { global.eon_eu_council_outcome_yes = global.eon_eu_council_open_outcome_yes }
   set_variable = { global.eon_eu_council_deadline_yes_count = global.eon_eu_council_open_yes_count }
   set_variable = { global.eon_eu_council_deadline_no_count = global.eon_eu_council_open_no_count }
   set_variable = { global.eon_eu_council_deadline_population_yes = global.eon_eu_council_open_population_yes }
  }
  else = {
   # An older save contains no predeadline snapshot. At the first verified
   # native elapsed witness only, adopt its then-live ballots/eligibility.
   eon_eu_recount_council_vote = yes
   if = { limit = { EU_voting_decision_result_trigger = yes }
    set_variable = { global.eon_eu_council_outcome_yes = 1 }
   }
   set_variable = { global.eon_eu_council_deadline_yes_count = global.EU_active_vote_yes }
   set_variable = { global.eon_eu_council_deadline_no_count = global.EU_active_vote_no }
   set_variable = { global.eon_eu_council_deadline_population_yes = global.EU_pop_ratio_yes }
  }
  set_variable = { global.eon_eu_council_outcome_instance = global.eon_eu_council_vote_instance }
  set_variable = { global.eon_eu_council_outcome_agenda = global.eon_eu_council_round_agenda }
 }
