# Scripted effects for CSTO
CSTO_join = {
	set_temp_variable = { eon_CSTO_new_membership = 0 }
	if = {
		limit = { NOT = { has_idea = CSTO_member } }
		set_temp_variable = { eon_CSTO_new_membership = 1 }
	}
	# Scope the accepted applicant through PREV rather than the event's ROOT.
	random_scope_in_array = {
		array = global.CSTO_member
		limit = {
			has_idea = CSTO_member
			is_faction_leader = yes
			has_faction_template = faction_template_csto
		}
		if = {
			limit = { NOT = { is_in_faction_with = PREV } }
			add_to_faction = PREV
		}
	}
	if = {
		limit = { has_faction_template = faction_template_csto }
		if = {
			limit = { check_variable = { eon_CSTO_new_membership = 1 } }
			hidden_effect = { news_event = { id = CSTO.4 hours = 6 } }
		}
		add_ideas = CSTO_member
		while_loop_effect = {
			limit = { is_in_array = { global.CSTO_member = THIS } }
			remove_from_array = { global.CSTO_member = THIS }
		}
		add_to_array = { global.CSTO_member = THIS }
		add_to_tech_sharing_group = CSTO_Tech_Share
	}
}

CSTO_leave = {
	CSTO_leave_idea = yes
}

CSTO_leave_idea = {
	if = {
		limit = {
			NOT = { has_country_flag = eon_CSTO_membership_cleanup }
			OR = {
				has_idea = CSTO_member
				is_in_array = { global.CSTO_member = THIS }
				has_faction_template = faction_template_csto
			}
		}
		set_country_flag = eon_CSTO_membership_cleanup
		while_loop_effect = {
			limit = { is_in_array = { global.CSTO_member = THIS } }
			remove_from_array = { global.CSTO_member = THIS }
		}
		remove_from_tech_sharing_group = CSTO_Tech_Share
		if = {
			limit = { has_faction_template = faction_template_csto }
			leave_faction = yes
		}
		remove_ideas = CSTO_member
		hidden_effect = { news_event = { id = CSTO.5 hours = 6 } }
		clr_country_flag = eon_CSTO_membership_cleanup
	}
}
