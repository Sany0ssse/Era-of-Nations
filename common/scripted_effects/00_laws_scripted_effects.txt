set_ban_weapons = {
    if = {
        limit = { has_idea = ban_weapons }
        add_political_power = 1500
    }
    else_if = {
        limit = { has_idea = allow_weapons }
        swap_ideas = {
			remove_idea = allow_weapons
			add_idea = ban_weapons
        }
    }
    else = {
        swap_ideas = {
			remove_idea = regulate_weapons
			add_idea = ban_weapons
        }
    }
}
