apply_protest_effects = {
	# monthly cooldown of tensions
	apply_passive_protest_drift = yes
	apply_passive_radicalisation_drift = yes

	set_temp_variable = { protest_drift = 0 }
	calculate_protest_drift = yes

	add_to_variable = { protest_strength = protest_drift }
	clamp_variable = { var = protest_strength min = 0 max = 100 }
	if = {
		limit = {
			check_variable = { THIS.protest_strength > 35 }
		}

		roll_protest_chance = yes
	}
}

apply_passive_protest_drift = {
	random = {
		chance = 50
		add_to_variable = { protest_strength = -2 }
	}

	if = { limit = { NOT = { has_variable = protest_strength } } set_variable = { protest_strength = 0 } }

	clamp_variable = { var = protest_strength min = 0 max = 100 }
}

apply_passive_radicalisation_drift = {
	random = {
		chance = 10
		add_to_variable = { protest_radicalisation = -1 }
	}

	if = { limit = { NOT = { has_variable = protest_radicalisation } } set_variable = { protest_radicalisation = 0 } }

	clamp_variable = {
		var = THIS.protest_radicalisation
		min = 0
		max = 10
	}
}

calculate_protest_drift = {
	if = {
		limit = {
			check_variable = { energy_balance < 0 }
		}

		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_fuel < 1
		}

		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			OR = {
				has_idea = inflation_03
				has_negative_economy_vibe = yes
			}
		}

		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_dynamic_modifier = { modifier = high_unemployment_modifier }
		}
		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_idea = War_with_nukestate_idea
		}

		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_stability < 0.25
		}

		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_offensive_war = yes
			has_war_support < 0.25
			OR = {
				has_idea = AB_partial_mobilization
				has_idea = AB_full_mobilization
				has_idea = partial_draft_army
				has_idea = draft_army
			}
		}
		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			has_war_support < 0.25
			any_other_country = {
				has_volunteers_amount_from = { tag = PREV count > 0 }
			}
		}
		add_to_temp_variable = { protest_drift = 1 }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_welfare_spending > THIS.social_budget }
		}

		set_temp_variable = { buf = THIS.expected_welfare_spending }
		subtract_from_temp_variable = { buf = THIS.social_budget }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_minimal_salary > THIS.Minimal_Salary }
		}

		set_temp_variable = { buf = THIS.expected_minimal_salary }
		subtract_from_temp_variable = { buf = THIS.Minimal_Salary }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_health_spending > THIS.health_budget }
		}

		set_temp_variable = { buf = THIS.expected_health_spending }
		subtract_from_temp_variable = { buf = THIS.health_budget }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_edu_spending > THIS.education_budget }
		}

		set_temp_variable = { buf = THIS.expected_edu_spending }
		subtract_from_temp_variable = { buf = THIS.education_budget }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_police_spending > THIS.crime_fighting }
		}

		set_temp_variable = { buf = THIS.expected_police_spending }
		subtract_from_temp_variable = { buf = THIS.crime_fighting }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_adm_spending > THIS.bureaucracy }
		}

		set_temp_variable = { buf = THIS.expected_adm_spending }
		subtract_from_temp_variable = { buf = THIS.bureaucracy }
		add_to_temp_variable = { protest_drift = buf }
	}
	if = {
		limit = {
			check_variable = { THIS.expected_military_sp > THIS.Military_Spending }
		}

		set_temp_variable = { buf = THIS.expected_military_sp }
		subtract_from_temp_variable = { buf = THIS.Military_Spending }
		add_to_temp_variable = { protest_drift = buf }
	}
	set_temp_variable = { protests_multiplyer = 1 }
	add_to_temp_variable = { protests_multiplyer = modifier@protests_drift_modifier }
	clamp_temp_variable = { var = protests_multiplyer min = 0.01 }
	multiply_temp_variable = { protest_drift = protests_multiplyer }
	round_temp_variable = protest_drift
	clamp_temp_variable = { var = protest_drift min = 0 }
}

roll_protest_chance = {
	random_list = {
		500 = {
			# nothing
		}
		var:protest_strength = {
			if = {
				limit = {
					NOT = { has_idea = intervention_isolation }
					NOT = { has_idea = intervention_local_security }
					NOT = { has_idea = intervention_limited_interventionism }
					has_war_support < 0.25
					any_other_country = {
						has_volunteers_amount_from = { tag = PREV count > 0 }
					}
				}
				random = {
					chance = 10
					country_event = { id = protests.21 random_days = 27 }
				}
			}
			else_if = {
				limit = {
					has_offensive_war = yes
					has_war_support < 0.25
					OR = {
						has_idea = AB_partial_mobilization
						has_idea = AB_full_mobilization
						has_idea = partial_draft_army
						has_idea = draft_army
					}
				}
				random = {
					chance = 10
					country_event = { id = protests.22 random_days = 27 }
				}
			}
			if = {
				limit = {
					has_idea = assembly_01
					NOT = { has_idea = negligible_corruption }
					NOT = { has_idea = slight_corruption }
					NOT = { has_idea = modest_corruption }
				}
				random = {
					chance = 5
					country_event = { id = protests.23 random_days = 27 }
				}
			}
			if = {
				limit = { check_variable = { THIS.protest_radicalisation < 3 } }
				random_list = {
					10 = {
						country_event = { id = protests.1 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.2 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.3 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.4 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.5 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.6 random_days = 27 }
					}
				}
			}
			else_if = {
				limit = { check_variable = { THIS.protest_radicalisation < 7 } }
				random_list = {
					10 = {
						country_event = { id = protests.10 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.11 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.12 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.13 random_days = 27 }
					}
					10 = {
						country_event = { id = protests.14 random_days = 27 }
					}
				}
			}
			else = {
				random_list = {
					90 = {
						random_list = {
							10 = {
								country_event = { id = protests.10 random_days = 27 }
							}
							10 = {
								country_event = { id = protests.11 random_days = 27 }
							}
							10 = {
								country_event = { id = protests.12 random_days = 27 }
							}
							10 = {
								country_event = { id = protests.13 random_days = 27 }
							}
							10 = {
								country_event = { id = protests.14 random_days = 27 }
							}
						}
					}
					10 = {
						country_event = { id = protests.20 random_days = 27 }
					}
				}
			}
		}
	}
}

# arg1ument - buf, used as return value of relative protest chance
calculate_basic_chance_for_protest = {
	if = {
		limit = { has_idea = police_01 }
		add_to_temp_variable = { buf = -35 }
	}
	else_if = {
		limit = { has_idea = police_02 }
		add_to_temp_variable = { buf = -20 }
	}
	else_if = {
		limit = { has_idea = police_04 }
		add_to_temp_variable = { buf = 20 }
	}
	else_if = {
		limit = { has_idea = police_05 }
		add_to_temp_variable = { buf = 35 }
	}
}

# argument - arg1
increase_radicalisation = {
	log = "[GetDateText]: [This.GetName]: radicalisation increased by: [?arg1]"
	add_to_variable = { THIS.protest_radicalisation = arg1 }

	clamp_variable = {
		var = THIS.protest_radicalisation
		min = 0
		max = 10
	}
}

	log = "[GetDateText]: [This.GetName]: protest strength increased by: [?arg1]"
	add_to_variable = { THIS.protest_strength = arg1 }

	clamp_variable = { var = protest_strength min = 0 max = 100 }
}
