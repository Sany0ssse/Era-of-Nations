revoke_sanctions = {
	ROOT = { eon_maritime_sanctions_stop = yes }
	set_variable = { i = 0 }
	set_variable = { i1 = 0 }
	while_loop_effect = {
		limit = { always = yes }
		break = break
		if = {
			limit = { check_variable = { sanctioneers^i = ROOT } }
            remove_from_array = {
				array = sanctioneers
				index = i
			}
			subtract_from_variable = { sum_sanctions_force = sanctions^i }
			remove_from_array = {
				array = sanctions
				index = i
			}
			remove_opinion_modifier = {
				target = ROOT
				modifier = sanctions
			}
			while_loop_effect = {
				limit = { always = yes }
				break = break1
				if = {
					limit = { check_variable = { ROOT.sanctions_targets^i1 = THIS } }
					remove_from_array = {
						array = ROOT.sanctions_targets
						index = i1
					}
					subtract_from_variable = { ROOT.sum_sanctions_feedback_force = ROOT.sanctions_feedback^i1 }
					remove_from_array = {
						array = ROOT.sanctions_feedback
						index = i1
					}
					set_temp_variable = { break1 = 1 }
				}
				add_to_variable = { i1 = 1 }
			}
			set_temp_variable = { break = 1 }
		}
		add_to_variable = { i = 1 }
	}
	clear_variable = i
	clear_variable = i1

	if = {
		limit = {
			NOT = {
				any_country = {
					is_in_array = { PREV.sanctioneers = THIS }
				}
			}
		}
		for_each_scope_loop = {
			array = global.countries
			PREV = {
				remove_opinion_modifier = {
					target = PREV
					modifier = helps_fighting_sanctions
				}
				clear_array = sanctions_evade_helpers
				clear_array = sanctions_evade_helpers_value
			}
		}
		remove_dynamic_modifier = {	modifier = sanctions_dynamic_modifier }
		clear_variable = sum_sanctions_force
		clear_variable = sum_fighting_sanctions_help
		clear_variable = sanctions_modifier_value
		clear_variable = sanctions_modifier_value_minus
		clear_variable = sanctions_modifier_value_first
	}

	if = {
		limit = {
			NOT = {
				any_country = {
					is_in_array = { ROOT.sanctions_targets = THIS }
				}
			}
		}
		ROOT = {
			remove_dynamic_modifier = {	modifier = sanctions_feedback_dynamic_modifier }
			clear_variable = sanctions_feedback_modifier_value
			clear_variable = sanctions_feedback_modifier_value_minus
		}
	}

	ROOT = { calculate_sanctions_feedback_modifier = yes }
}

		clear_variable = sum_fighting_sanctions_help

	else = {
		subtract_from_variable = { sanctions_modifier_value = sum_fighting_sanctions_help }

	if = {
