
## Spawn units
# Blackwater
spawn_light_infantry_blackwater = {
	load_oob = pmc_blackwater_light

	70 = {
		teleport_armies = {
			limit = { tag = ROOT }
		}
	}
}

spawn_mot_infantry_blackwater = {
	if = {
		limit = { has_dlc = "No Step Back" }
		load_oob = pmc_blackwater_mot
		else = {
			load_oob = pmc_blackwater_mot_nonnsb
		}
