	set_temp_variable = { consumer_goods_factor_full = 1 }
	add_to_temp_variable = { consumer_goods_factor_full = modifier@consumer_goods_factor }
	multiply_temp_variable = { actual_consumer_goods_percent = consumer_goods_factor_full }
	set_variable = { var = actual_consumer_goods_percent_display value = actual_consumer_goods_percent }


	hidden_effect = {
		productivity_growth_country_display_effect = yes
		every_controlled_state = {
			                 #counting number of damaged buildings for states and summing states numbers for country
   add_to_variable = { PREV.num_of_damaged_civs = damaged_building_level@industrial_complex }
   add_to_variable = { PREV.num_of_damaged_mils = damaged_building_level@arms_factory }
   add_to_variable = { PREV.num_of_damaged_dockyards = damaged_building_level@dockyard }
   add_to_variable = { PREV.num_of_damaged_offices = damaged_building_level@offices }
   add_to_variable = { PREV.num_of_damaged_rub_ref = damaged_building_level@rubber_refinery }
   add_to_variable = { PREV.num_of_damaged_agriculture_district = damaged_building_level@agriculture_district }
   add_to_variable = { PREV.num_of_damaged_nuclear_reactor = damaged_building_level@nuclear_reactor }
   add_to_variable = { PREV.num_of_damaged_synthetic_refinery = damaged_building_level@synthetic_refinery }
   add_to_variable = { PREV.num_of_synthetic_refinery = building_level@synthetic_refinery }

			add_to_variable = { PREV.num_missile_silo = building_level@rocket_site }

			if = { limit = { check_variable = { building_level@synthetic_refinery > 0 } }
				set_temp_variable = { state_renewable_generation = building_level@synthetic_refinery }

			set_temp_variable = { productivity_temp_state_var = productivity_state_var }
			multiply_temp_variable = { productivity_temp_state_var = 0.001 }
			multiply_temp_variable = { productivity_temp_state_var = temp_pop }
			add_to_variable = { PREV.overall_productivity = productivity_temp_state_var }
			productivity_growth_controlled_display_effect = yes
		
		}
		
  set_variable = { damaged_civil_fac_debuff = num_of_damaged_civs }
