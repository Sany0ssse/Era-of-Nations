    if = {
        limit = {
            check_variable = { global.arctic_controllers^arctic_region_to_aquire = NULL }
        }
        set_variable = { global.arctic_controllers^arctic_region_to_aquire = ROOT }
        set_variable = { global.arctic_military^arctic_region_to_aquire = 1}

        set_variable = { arctic_region_to_build_military_base = arctic_region_to_aquire }

    }
}

select_arctic_attacker_state = {
    set_temp_variable = { arctic_unit_strngth_prev = 0 }
    set_temp_variable = { arctic_unit_org_prev = 0 }
    set_temp_variable = { arctic_unit_xp_prev = 0 }
    set_temp_variable = { arctic_air_strngth_prev = 0 }
    set_temp_variable = { arctic_air_xp_prev = 0 }
    set_temp_variable = { arctic_air_ace_prev = 0 }
    for_each_loop = {
        array = global.arctic_controllers
        value = v
        index = i
        set_temp_variable = { state_to_check_if_neighbor = i }
        if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { global.arctic_unit_strngth^i > arctic_unit_strngth_prev }
            }
            set_temp_variable = { arctic_unit_strngth_prev = global.arctic_unit_strngth^i }
            set_variable = { arctic_attacker_state = i }
        }
        else_if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { global.arctic_unit_org^i > arctic_unit_org_prev }
            }
            set_temp_variable = { arctic_unit_org_prev = global.arctic_unit_org^i }
            set_variable = { arctic_attacker_state = i }
        }
        else_if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { global.arctic_unit_xp^i > arctic_unit_xp_prev }
            }
            set_temp_variable = { arctic_unit_xp_prev = global.arctic_unit_xp^i }
            set_variable = { arctic_attacker_state = i }
        }
        else_if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { global.arctic_air_strngth^i > arctic_air_strngth_prev }
            }
            set_temp_variable = { arctic_air_strngth_prev = global.arctic_air_strngth^i }
            set_variable = { arctic_attacker_state = i }
        }
        else_if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { global.arctic_air_xp^i > arctic_air_xp_prev }
            }
            set_temp_variable = { arctic_air_xp_prev = global.arctic_air_xp^i }
            set_variable = { arctic_attacker_state = i }
        }
        else_if = {
            limit = {
                arctic_state_is_neighbor = yes
                check_variable = { v = ROOT }
                check_variable = { arctic_air_ace_prev = 0 }
                NOT = { check_variable = { global.arctic_air_ace^i = 0 } }
            }
            set_temp_variable = { arctic_air_ace_prev = global.arctic_air_ace^i }
            set_variable = { arctic_attacker_state = i }
        }
    }
}

count_attack_effect = {
    add_to_variable = { divisions_in_offensive_combat_consumption = 10 }
    add_fuel = -10
    set_temp_variable = { attacker_attack = global.arctic_unit_strngth^attacker_state }
    set_temp_variable = { attacker_org = global.arctic_unit_org^attacker_state }
    multiply_temp_variable = { attacker_org = 0.5 }
    add_to_temp_variable = { attacker_attack = attacker_org }
    add_to_temp_variable = { attacker_attack = arctic_warfare_land_buff }
    multiply_temp_variable = { attacker_attack = 0.001 }
    if = {
        limit = { has_idea = recent_arctic_exercises }
        add_to_temp_variable = { attacker_attack = 0.005 }
    }
    if = {
        limit = { check_variable = { global.arctic_air_strngth^attacker_state > 0 } }
        add_to_variable = { air_ammo_consumption = 1 }
        add_fuel = -10
        set_temp_variable = { defender_aa = var:defender.arctic_warfare_aa_buff }
        multiply_temp_variable = { defender_aa = 0.0005 }
        set_temp_variable = { attacker_air = var:attaker.arctic_warfare_air_buff }
        set_temp_variable = { attacker_air_multiplyier = global.arctic_air_strngth^attacker_state }
        multiply_temp_variable = { attacker_air_multiplyier = 0.1 }
        multiply_temp_variable = { attacker_air = attacker_air_multiplyier }
        multiply_temp_variable = { attacker_air = 0.001 }
        set_temp_variable = { max_attacker_air_death_rate = defender_aa }
        if = {
            limit = {
                check_variable = { max_attacker_air_death_rate > 0.49 }
            }
            set_temp_variable = { max_attacker_air_death_rate = 2 }
        }
        else = {
            set_temp_variable = { max_attacker_air_death_rate = 1 }
        }
        subtract_from_temp_variable = { attacker_air = defender_aa }
        add_to_temp_variable = { attacker_attack = attacker_air }
        random = {
            chance = 1
            subtract_from_variable = { global.arctic_air_strngth^attacker_state = max_attacker_air_death_rate }
            set_country_flag = shooted_air
        }
    }
    if = {
        limit = {
            OR = {
                check_variable = { defender_state = 0 }
                check_variable = { defender_state = 1 }
                check_variable = { defender_state = 2 }
                check_variable = { defender_state = 3 }
                check_variable = { defender_state = 4 }
                check_variable = { defender_state = 6 }
                check_variable = { defender_state = 7 }
                check_variable = { defender_state = 13 }
                check_variable = { defender_state = 14 }
                check_variable = { defender_state = 15 }
                check_variable = { defender_state = 16 }
                check_variable = { defender_state = 18 }
            }
        }
        set_temp_variable = { attacker_navy = var:attacker.arctic_warfare_navy_buff }
        multiply_temp_variable = { attacker_navy = var:attacker.alliance_naval_strength_ratio }
        multiply_temp_variable = { attacker_navy = 0.001 }
        add_to_temp_variable = { attacker_attack = attacker_navy }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    var:attacker = { has_country_flag = arctic_analyst_general }
                    var:defender = { has_country_flag = arctic_analyst_general }
