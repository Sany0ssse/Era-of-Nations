eon_eu_refresh_membership = {
 if = {
  limit = { NOT = { has_global_flag = eon_eu_registry_refreshing } }
  set_global_flag = eon_eu_registry_refreshing
  every_country = { limit = { has_global_flag = eon_eu_admission_guard_ready has_idea = EU_member NOT = { has_country_flag = eon_eu_membership_recognized } } eon_eu_reject_unapproved_membership = yes }
  clear_array = global.eon_eu_previous_members
  for_each_scope_loop = { array = global.EU_member add_to_array = { global.eon_eu_previous_members = THIS } }
  clear_array = global.EU_member
  clear_array = global.EU_candidates
  set_variable = { global.EU_member_count = 0 }
  every_country = {
   limit = { eon_eu_is_live_legal_member = yes }
   add_to_array = { global.EU_member = THIS }
   add_to_variable = { global.EU_member_count = 1 }
  }
  if = {
   limit = { check_variable = { global.EU_member_count = 0 } }
   if = { limit = { NOT = { has_global_flag = eon_eu_dormant } }
    eon_eu_reset_institutional_laws = yes
    set_global_flag = eon_eu_archive_cleaned
   }
   set_global_flag = eon_eu_dormant
   every_country = { eon_eu_clear_country_access = yes }
   for_each_scope_loop = {
    array = global.eon_eu_previous_members
    # Legal membership survives temporary subordination while other institutions function.
    # With no representation left, archive the game institution; no automatic resurrection.
    remove_ideas = EU_member
    eon_eu_clear_country_access = yes
   }
   every_country = {
    limit = { OR = { has_idea = EU_member has_country_flag = EU_candidate check_variable = { eon_eu_accession_phase > 0 } } }
    remove_ideas = EU_member
    eon_eu_clear_country_access = yes
   }
   set_variable = { global.var_EUpopTotal = 0 }
   set_variable = { global.var_EUpopTotalDisplay = 0 }
   set_variable = { global.var_europeanism = 0 }
   set_variable = { global.EU_total_qmv_state = 0 }
   set_variable = { global.EU_majority_qmv_state = 0 }
   set_variable = { global.EU_gdp = 0 }
   set_variable = { global.EU_gdp_capita = 0 }
   set_variable = { global.EU_total_budget = 0 }
   set_variable = { global.EU_total_exp = 0 }
  }
  else = {
   # An archived union can only resume via the explicit consenting founding pair.
   if = { limit = { has_global_flag = eon_eu_dormant }
    # Stray script-granted membership cannot revive an archived institution or its bonuses.
    every_country = {
     limit = { OR = { has_idea = EU_member has_country_flag = EU_candidate check_variable = { eon_eu_accession_phase > 0 } } }
     remove_ideas = EU_member
     eon_eu_clear_country_access = yes
    }
    clear_array = global.EU_member
    set_variable = { global.EU_member_count = 0 }
   }
   else = {
    every_country = {
     limit = { eon_eu_is_candidate = yes }
     add_to_array = { global.EU_candidates = THIS }
    }
    EU_update_vars = yes
   }
  }
  if = {
   limit = {
    OR = {
     NOT = { check_variable = { var = global.eon_eu_previous_members^num value = global.EU_member_count compare = equals } }
     any_of_scopes = { array = global.eon_eu_previous_members NOT = { is_in_array = { global.EU_member = THIS } } }
    }
   }
   set_global_flag = eon_eu_membership_changed
  }
  clear_array = global.eon_eu_previous_members
  clr_global_flag = eon_eu_registry_refreshing
  update_eu_dirty_variable = yes
 }
}

# Treaty EP consent is a transparent game abstraction: each country controls its
# national delegation. Its existing elected MEP_Total seats vote as one block.
# It is separate from national governments' Council / ratification decisions.
eon_eu_recount_treaty_parliament = {
 clr_country_flag = eon_eu_accession_ep_consent
 clr_country_flag = eon_eu_withdrawal_ep_consent
 set_temp_variable = { eon_eu_treaty_ep_total = 0 }
 set_temp_variable = { eon_eu_treaty_accession_ep_yes = 0 }
 set_temp_variable = { eon_eu_treaty_withdrawal_ep_yes = 0 }
 set_temp_variable = { eon_eu_treaty_withdrawal_ep_no = 0 }
 for_each_scope_loop = {
  array = global.EU_member
  if = { limit = { eon_eu_is_active_member = yes check_variable = { MEP_Total > 0 } }
   add_to_temp_variable = { eon_eu_treaty_ep_total = MEP_Total }
   if = { limit = { is_in_array = { PREV.eon_eu_accession_ep_approvals = THIS } }
    add_to_temp_variable = { eon_eu_treaty_accession_ep_yes = MEP_Total }
   }
   # The withdrawing state is excluded from Council deliberation, not automatically
