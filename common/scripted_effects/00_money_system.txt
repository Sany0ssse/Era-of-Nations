###Treasury reset### -- Legacy commands. Moved this functionality to a update_display command
#In a couple places I've left a comment saying referring to treasury reset explanation. This is the explanation
#It took me a while to figure out why you add debt_rate to treasury_rate so ill try to explain (Gearz)
#When your debt_rate changes you need to "reset the treasury_rate" so you return
#how much you were taking from the treasury before and then take the new amount you need
#e.g treasury_rate $10B, you take $2B for your current debt_rate. Treasury rate = $8B
#Your new debt_rate is $4B. So you need to "reset the treasury" you give the treasury_rate back the $2B
#So treasury_rate is now back to $10B, now subtract the $4B and now treasury rate is $6B

			consumer_goods_from_corporate_tax = {
				value = corporate_tax_rate
				multiply = 0.01
			}
		}

		set_variable = {
			receiving_investment_cost_form_corporate_tax = {
				value = corporate_tax_rate
				multiply = 0.01
			}
		}

		# Productivity
		set_variable = {
			productivity_growth_modifier_from_corporate_tax = {
				value = corporate_tax_rate
				multiply = -1
				add = 20

			}
		}

			small_businesses_workers_from_corporate_tax = {
				value = corporate_tax_rate
				multiply = -0.001
			}
		}

			stability_factor_from_population_tax = {
				value = population_tax_rate

		#				has_idea = social_05

		#			}
		#		}
		#	}
		#	set_variable = { high_population_tax = population_tax_rate }
		#	divide_variable = { high_population_tax = 25 }
		#}
		#else = {
		#	set_variable = { high_population_tax = 1 }

# Effect: calculate_tax_gain
# Purpose: Calculates the amount of taxes earned based of ROOT's corporate taxes and population taxes
calculate_tax_gain = {
	set_variable = { tax_gain = 0 }
	# Update starting tax rate
	calculate_tax_rate = yes

	# Corporate Tax Expansion
	set_variable = { corporate_tax = 0 }

	# Subsistence Agriculture Tax Gain Income
	set_variable = {
		agriculture_tax_gain = {
			value = gdp_from_agriculture
			multiply = {
				value = modifier@agricolture_productivity_modifier
				multiply = modifier@agriculture_tax_modifier
				add = 1
				multiply = corporate_tax_rate
				multiply = 0.00019
			}
		}
	}
	clamp_variable = { var = agriculture_tax_gain min = 0 }

	#Small Businesses Gain
	set_variable = {
		small_businesses_tax_gain = {
			value = gdp_from_small_businesses
			multiply = {
				value = modifier@small_businesses_productivity_modifier
				multiply = modifier@small_businesses_tax_modifier
				add = 1
				multiply = corporate_tax_rate
				multiply = 0.00019
			}
		}
	}
	clamp_variable = { var = small_businesses_tax_gain min = 0 }

	# Set base building counts (base tax constants will be factored into building-specific unified multipliers)
	# Civilian Factories - base constant 2.5
	# Military Factories - base constant 0.2
	# Naval Dockyards - base constant 0.2
	# Offices - base constant 5
	# Rubber Refineries - base constant 0.2
	# Agriculture Districts - base constant 2.60
	set_variable = { civil_fac_tax = industrial_complex_total }
	subtract_from_variable = { civil_fac_tax = damaged_civil_fac_debuff }
	set_variable = { military_fac_tax = military_fac }
	subtract_from_variable = { military_fac_tax = damaged_military_fac_debuff }
	set_variable = { dockyard_tax = dockyard_fac }
	subtract_from_variable = { dockyard_tax = damaged_dockyard_debuff }
	set_variable = { office_tax = office_park_total }
	subtract_from_variable = { office_tax = damaged_offices_debuff }
	set_variable = { rub_ref_tax = rubber_refinery_total }
	subtract_from_variable = { rub_ref_tax = damaged_rub_ref_debuff }
	set_variable = { agriculture_district_tax = agriculture_district_total }
	subtract_from_variable = { agriculture_district_tax = damaged_agriculture_district_debuff }

	# Calculate unified building tax multiplier (applied once at the end)
	# This combines: control_factor, corporate_tax_rate, tax_gain_multiplier,
	# 7-day display (0.007), corporate_tax_gain_multiplier, and overall_productivity

	# Additional additive modifiers (used for both buildings and population)
	set_temp_variable = {
		tax_gain_multiplier = {
			value = 1
			add = modifier@tax_gain_multiplier_modifier
		}
	}

	##For display
	set_variable = {
		tax_gain_multiplier_display = {
			value = tax_gain_multiplier
			add = -1
		}
	}

	# Build the unified multiplier
	set_temp_variable = {
		unified_building_multiplier = {
			value = @money_control_factor_scaling
			multiply = corporate_tax_rate
			multiply = tax_gain_multiplier
			multiply = 0.007
			multiply = {
				value = 1
				add = modifier@corporate_tax_income_multiplier_modifier
			}
			multiply = {
				value = overall_productivity
				multiply = 0.001
			}
		}
	}

	#Companies Gain
	set_variable = {
		companies_tax_gain = {
			value = gdp_from_companies
			multiply = {
				value = unified_building_multiplier
				multiply = {
					value = 1
					add = modifier@companies_tax_gain_multiplier_modifier
				}
			}
			multiply = 0.25
		}
	}
	if = {
		limit = {
			check_variable = { count_companies > 0 }
		}
		clamp_variable = { var = companies_tax_gain min = 0.001 }
	}
	else = { set_variable = { companies_tax_gain = 0 } }

	#Branches Gain
	set_variable = {
		branches_tax_gain = {
			value = gdp_from_branches
			multiply = {
				value = unified_building_multiplier
				multiply = {
					value = 1
					add = modifier@branches_tax_gain_multiplier_modifier
				}
			}
			multiply = 0.25
		}
	}
	if = {
		limit = {
			check_variable = { count_branches > 0 }
		}
		clamp_variable = { var = branches_tax_gain min = 0.001 }
	}
	else = { set_variable = { branches_tax_gain = 0 } }

	#Abroad Branches Gain
	set_variable = { abroad_branches_tax_gain = 0 }
	set_variable = { i = 0 }
	for_each_scope_loop = {
		array = countries_with_branches
		set_temp_variable = { gdp_normaliser = percentage_of_global_factories }
		clamp_temp_variable = { var = gdp_normaliser min = 0.01 }
		set_temp_variable = {
			branches_gdp_to_add = {
				value = gdp_per_capita
				multiply = gdp_normaliser
			}
		}
		set_temp_variable = {
			abroad_branches_tax_gain_capitalization_multiplyer = {
				value = capitalization
				multiply = 0.015
			}
		}
		multiply_variable = { branches_gdp_to_add = abroad_branches_tax_gain_capitalization_multiplyer }
		divide_temp_variable = { branches_gdp_to_add = count_branches }
		multiply_temp_variable = { branches_gdp_to_add = PREV.countries_with_count_branches^PREV.i }
		add_to_variable = { PREV.abroad_branches_tax_gain = branches_gdp_to_add }
		add_to_variable = { PREV.i = 1 }
	}
	set_temp_variable = {
		abroad_branches_tax_gain_multiplier = {
			value = unified_building_multiplier
			multiply = {
				value = 1
				add = modifier@companies_tax_gain_multiplier_modifier
			}
			multiply = {
				value = capitalization
				multiply = 0.15
			}
		}
	}
	multiply_variable = { abroad_branches_tax_gain = abroad_branches_tax_gain_multiplier }
	if = {
		limit = {
			any_country = {
				is_in_array = {
					array = PREV.countries_with_branches
					value = THIS
				}
			}
		}
		clamp_variable = { var = abroad_branches_tax_gain min = 0.001 }
	}
	else = { set_variable = { abroad_branches_tax_gain = 0 } }
	clear_variable = i

	# Apply building-specific modifiers, productivity, employment, then building-specific unified multiplier
	# Each building: base_count * (1 + tax_modifier) * (1 + productivity_modifier) * employment * building_unified
	# Civilian Factories
	if = { limit = { check_variable = { civil_fac_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@civilian_industry_tax_modifier
				multiply = {
					value = 1
					add = modifier@civilian_factories_productivity
				}
				multiply = civilian_factories_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 2.5
				}
			}
		}
		multiply_variable = { civil_fac_tax = building_final_modifier }
	}

	# Military Factories
	if = { limit = { check_variable = { military_fac_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@military_industry_tax_modifier
				multiply = {
					value = 1
					add = modifier@military_factories_productivity
				}
				multiply = military_factories_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 0.2
				}
			}
		}
		multiply_variable = { military_fac_tax = building_final_modifier }
	}

	# Dockyards
	if = { limit = { check_variable = { dockyard_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@dockyard_income_tax_modifier
				multiply = {
					value = 1
					add = modifier@dockyard_productivity
				}
				multiply = naval_factories_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 0.2
				}
			}
		}
		multiply_variable = { dockyard_tax = building_final_modifier }
	}

	# Offices
	if = { limit = { check_variable = { office_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@office_park_income_tax_modifier
				multiply = {
					value = 1
					add = modifier@offices_productivity
				}
				multiply = offices_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 5
				}
			}
		}
		multiply_variable = { office_tax = building_final_modifier }
	}

	# Agriculture Districts
	if = { limit = { check_variable = { agriculture_district_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@agriculture_district_income_tax_modifier
				multiply = {
					value = 1
					add = modifier@agricolture_productivity_modifier
				}
				multiply = agriculture_district_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 2.6
				}
			}
		}
		multiply_variable = { agriculture_district_tax = building_final_modifier }
	}

	# Rubber Refinery
	if = { limit = { check_variable = { rub_ref_tax > 0 } }
		set_temp_variable = {
			building_final_modifier = {
				value = 1
				add = modifier@rub_ref_income_tax_modifier
				multiply = {
					value = 1
					add = modifier@rub_ref_productivity
				}
				multiply = rub_ref_manpower_fulfillment
				multiply = {
					value = unified_building_multiplier
					multiply = 0.2
				}
			}
		}
		multiply_variable = { rub_ref_tax = building_final_modifier }
	}

	##POP
	set_variable = {
		population_tax = {
			value = population_total_m
			multiply = population_tax_rate
			multiply = 0.45
			multiply = {
				value = gdp_per_capita
				multiply = 0.005
				add = 0.75
				multiply = {
					value = 1
					add = modifier@population_tax_income_multiplier_modifier
				}
				multiply = tax_gain_multiplier
				multiply = 0.007
			}
		}
	}

	# Clamp the component values to 0 so we don't have our government gifting back tax money to the people
	clamp_variable = { var = agriculture_district_tax min = 0 }
	clamp_variable = { var = agriculture_tax_gain min = 0 }
	clamp_variable = { var = civil_fac_tax min = 0 }
	clamp_variable = { var = military_fac_tax min = 0 }
	clamp_variable = { var = dockyard_tax min = 0 }
	clamp_variable = { var = office_tax min = 0 }
	clamp_variable = { var = rub_ref_tax min = 0 }

	# Add Corporate Tax Rates Together
	add_to_variable = { corporate_tax = agriculture_tax_gain }
	add_to_variable = { corporate_tax = small_businesses_tax_gain }
	add_to_variable = { corporate_tax = companies_tax_gain }
	add_to_variable = { corporate_tax = abroad_branches_tax_gain }
	add_to_variable = { corporate_tax = branches_tax_gain }
	add_to_variable = { corporate_tax = agriculture_district_tax }
	add_to_variable = { corporate_tax = civil_fac_tax }
	add_to_variable = { corporate_tax = military_fac_tax }
	add_to_variable = { corporate_tax = dockyard_tax }
	add_to_variable = { corporate_tax = office_tax }
	add_to_variable = { corporate_tax = rub_ref_tax }

	# Clamps Corporate and Population Tax Rates to 0 to prevent negative incomes
	clamp_variable = { var = corporate_tax min = 0 }
	clamp_variable = { var = population_tax min = 0 }

	set_variable = {
		tax_gain = {
			value = corporate_tax
			add = population_tax
		}
	}
}

calculate_additional_income_rate = {
	set_variable = { additional_income_rate = 0 }

	#Factories leasing
	if = {
		limit = { check_variable = { leased_civilian_factories < 0 } }
		set_temp_variable = {
			leased_civilian_factories_income = {
				value = leased_civilian_factories
				multiply = -0.02
				multiply = global.average_frontier_emerging_gdp_per_capita
				multiply = {
					value = 1
					add = modifier@leased_civilian_factories_income_modifier
				}
			}
		}
		set_variable = { lease_civilian_factories_income_display = leased_civilian_factories_income }
		add_to_variable = { additional_income_rate = leased_civilian_factories_income }
	}


	#Sanction evasion help
	clear_variable = sanctions_evasion_income
	for_each_loop = {
		array = global.countries
		value = v
		if = {
			limit = {
				is_in_array = { sanctions_evade_helpers = THIS }
			}
			if = {
				limit = { var:v = { has_idea = non_power } }
				add_to_variable = { sanctions_evasion_income = 0.05 }
			}
			else_if = {
				limit = { var:v = { has_idea = minor_power } }
				add_to_variable = { sanctions_evasion_income = 0.1 }
			}
			else_if = {
				limit = { var:v = { has_idea = regional_power } }
				add_to_variable = { sanctions_evasion_income = 0.5 }
			}
			else_if = {
				limit = { var:v = { has_idea = large_power } }
				add_to_variable = { sanctions_evasion_income = 1 }
			}
			else_if = {
				limit = { var:v = { has_idea = great_power } }
				add_to_variable = { sanctions_evasion_income = 2.5 }
			}
			else = { add_to_variable = { sanctions_evasion_income = 5 } }
		}
	}
	add_to_variable = { additional_income_rate = sanctions_evasion_income }

	#Migrants Agreements
	if = {
		limit = {
			has_dynamic_modifier = { modifier = migrants_agreement_cut }
		}
		set_variable = {
			adding_migrants_income = {
				value = migrants_cut
				multiply = -1

	if = {
		limit = { check_variable = { debt = 0 } }

		set_variable = { interest_rate = 0 }
		set_variable = { debt_rate = 0 }
		set_variable = { debt_ratio = 0 }
	}
	else = {
		# Country has debt - calculate interest rate, debt_rate, and debt_ratio
		#Base value comes from debt/gdp ratio
		set_variable = {
			interest_rate = {
				value = debt
				divide = gdp_total
				multiply = 10
				add = modifier@interest_rate_multiplier_modifier
			}
		}

		#Additional additive modifiers
		set_variable = { interest_rate_multiplier = modifier@interest_rate_multiplier_modifier }

		# Tension affects debt interest rate
        # This Country Tension
		set_variable = {
			country_threat_interest_rate_multiplyer = {
				value = has_added_tension_amount
				multiply = 0.02
			}
		}
		clamp_variable = { var = country_threat_interest_rate_multiplyer max = 1 }
		set_variable = {
			country_threat_interest_rate_multiplyer_display = {
				value = country_threat_interest_rate_multiplyer
				multiply = 0.1
			}
		}
		multiply_variable = { country_threat_interest_rate_multiplyer = 10 }
		#Global Threat Tension
		set_variable = {
			threat_interest_rate_multiplyer = {
				value = global.threat
				subtract = {
					value = has_added_tension_amount
					multiply = 0.01
				}
				multiply = 0.25
			}
		}
		clamp_variable = { var = threat_interest_rate_multiplyer max = 1 }
		set_variable = {
			threat_interest_rate_multiplyer_display = {
				value = threat_interest_rate_multiplyer
				multiply = 0.1
			}
		}
		multiply_variable = { threat_interest_rate_multiplyer = 10 }

        # Adding Tension Modifiers
		add_to_variable = { interest_rate = country_threat_interest_rate_multiplyer }
		add_to_variable = { interest_rate = threat_interest_rate_multiplyer }
		clamp_variable = { var = interest_rate min = 0.8 max = 50 } #Min is 0.8% on debt max is 50%

		#Tension affect debt interest rate mechanic end

		#Calculate debt rate based on interest
		set_variable = { debt_rate = debt }
		multiply_variable = { debt_rate = 0.00019 }	#Weekly interest rate, 52 weeks in a year
		multiply_variable = { debt_rate = interest_rate }	#Yearly interest rate

		#If interest rate is really high, add additional modifier
		if = { limit = { check_variable = { interest_rate > 14.999 } }
			if = {
				limit = {
					NOT = { has_dynamic_modifier = { modifier = very_high_interest_country_modifiers } }
				}
				add_dynamic_modifier = { modifier = very_high_interest_country_modifiers }
			}
			# Construction/factories/dockyard speed
			set_variable = {
				high_interest_penalty_modifier = {
					value = interest_rate
					add = -15
					multiply = -0.05
				}
			}
			# Stability
			set_variable = {
				stability_factor_very_high_interest_modifier = {
					value = interest_rate
					add = -15
					multiply = -0.02
				}
			}
			# Popularity
			##Set all to 0 in case an Outlook change between weeklys
			clear_variable = western_drift_very_high_interest_modifier
			clear_variable = emerging_drift_very_high_interest_modifier
			clear_variable = nonaligned_drift_very_high_interest_modifier
			clear_variable = salafist_drift_very_high_interest_modifier
			clear_variable = nationalist_drift_very_high_interest_modifier
			set_temp_variable = {
				outlook_drift = {
					value = interest_rate
					add = -15
					multiply = -0.01
				}
			}
			if = { limit = { has_government = democratic }
				add_to_variable = { western_drift_very_high_interest_modifier = outlook_drift }
			}
			else_if = { limit = { has_government = communism }
				add_to_variable = { emerging_drift_very_high_interest_modifier = outlook_drift }
			}
			else_if = { limit = { has_government = neutrality }
				add_to_variable = { nonaligned_drift_very_high_interest_modifier = outlook_drift }
			}
			else_if = { limit = { has_government = fascism }
				add_to_variable = { salafist_drift_very_high_interest_modifier = outlook_drift }
			}
			else_if = { limit = { has_government = nationalist }
				add_to_variable = { nationalist_drift_very_high_interest_modifier = outlook_drift }
			}
		}
		# Debt to GDP Ratio
		# Math Explanation:
		# The Debt to GDP Ratio helps dictate the ability of the nation to repay the debts
		# debt / total gdp = debt_ratio
		set_variable = {
			debt_ratio = {
				value = debt
				divide = gdp_total
			}
		}
	}

	# Remove high interest modifier if interest rate dropped below 15%
	if = {
		limit = {
			check_variable = { interest_rate < 15 }
			has_dynamic_modifier = { modifier = very_high_interest_country_modifiers }
		}
		remove_dynamic_modifier = { modifier = very_high_interest_country_modifiers }
		clear_variable = high_interest_penalty_modifier
		clear_variable = stability_factor_very_high_interest_modifier
		clear_variable = western_drift_very_high_interest_modifier
		clear_variable = emerging_drift_very_high_interest_modifier
		clear_variable = nonaligned_drift_very_high_interest_modifier
		clear_variable = salafist_drift_very_high_interest_modifier
		clear_variable = nationalist_drift_very_high_interest_modifier
	}
}

			add = num_equipment@cv_small_plane_airframe

			add = num_equipment@small_plane_strike_airframe
			add = num_equipment@cv_small_plane_strike_airframe

			add = num_deployed_planes_with_type@small_plane_cas_airframe
			add = num_deployed_planes_with_type@small_plane_naval_bomber_airframe
			add = num_deployed_planes_with_type@cv_small_plane_airframe
			add = num_deployed_planes_with_type@cv_small_plane_cas_airframe
			add = num_deployed_planes_with_type@cv_small_plane_naval_bomber_airframe
			add = num_deployed_planes_with_type@small_plane_strike_airframe
			add = num_deployed_planes_with_type@cv_small_plane_strike_airframe
		}
	}

	# Apply cost weights
	multiply_temp_variable = { strategic_airforce = 14 }
	multiply_temp_variable = { expensive_airforce = 5 }
	multiply_temp_variable = { regular_airforce = 3 }
	multiply_temp_variable = { cheap_airforce = 1 }

	# Add missile buildings
	set_temp_variable = {
		num_missile_silo_money = {
			value = num_missile_silo
			multiply = 28
		}
	}
	set_temp_variable = {
		num_anti_air_building_money = {
			value = num_anti_air_building
			multiply = 14
		}
	}
	set_temp_variable = {
		num_radar_stations_money = {
			value = num_radar_stations
			multiply = 5
		}
	}
	set_temp_variable = {
		num_air_bases_money = {
			value = num_air_bases
			multiply = 20
		}
	}

	# Calculate total airforce cost
	set_temp_variable = {
		airforce_count = {
			value = num_air_bases
			add = strategic_airforce
			add = expensive_airforce
			add = regular_airforce
			add = cheap_airforce
			add = num_missile_silo_money
			add = num_anti_air_building_money
			add = num_radar_stations_money
			add = num_air_bases_money
			multiply = 0.071
			multiply = 0.001
			multiply = 0.667
		}

	#Additive personnel cost multiplier
	set_temp_variable = {
		personnel_cost_multiplier = {
			value = 1
			add = modifier@personnel_cost_multiplier_modifier
		}
	}
	clamp_temp_variable = { var = personnel_cost_multiplier min = 0.25 }
	set_temp_variable = {
		army_personnel_cost_multiplier = {
			value = 1
			add = modifier@army_personnel_cost_multiplier_modifier
		}
	}
	clamp_temp_variable = { var = army_personnel_cost_multiplier min = 0.25 }
	set_temp_variable = {
		navy_personnel_cost_multiplier = {
			value = 1
			add = modifier@navy_personnel_cost_multiplier_modifier
		}
	}
	clamp_temp_variable = { var = navy_personnel_cost_multiplier min = 0.25 }
	set_temp_variable = {
		airforce_personnel_cost_multiplier = {
			value = 1
			add = modifier@airforce_personnel_cost_multiplier_modifier

		equipment_cost_multiplier = {

			add = modifier@equipment_cost_multiplier_modifier

	multiply_temp_variable = { land_count = personnel_cost_multiplier }
	multiply_temp_variable = { navy_count = personnel_cost_multiplier }
	multiply_temp_variable = { airforce_count = personnel_cost_multiplier }
	multiply_temp_variable = { land_count = army_personnel_cost_multiplier }
	multiply_temp_variable = { navy_count = navy_personnel_cost_multiplier }
	multiply_temp_variable = { airforce_count = airforce_personnel_cost_multiplier }
	#Modify equipment cost with modifier
	multiply_variable = { equipment_operative_cost = equipment_cost_multiplier }

	#Multiply personnel costs basde on GDP/c, higher GDP makes things more expensive (linear approximation 0.003 * gdp/c + 0.129)
	set_temp_variable = {
		gdp_defence_modifier_personnel = {
			value = gdp_per_capita
			multiply = 0.004
			add = 0.129
		}
	}
	multiply_temp_variable = { land_count = gdp_defence_modifier_personnel }
	multiply_temp_variable = { navy_count = gdp_defence_modifier_personnel }
	multiply_temp_variable = { airforce_count = gdp_defence_modifier_personnel }

	set_temp_variable = {
		gdp_defence_modifier_equipment = {
			value = gdp_per_capita
			multiply = 0.015
			add = 1
		}
	}
	multiply_variable = { equipment_operative_cost = gdp_defence_modifier_equipment }

	set_temp_variable = {
		mobilization_level_modifier = {
			value = 0.25
			multiply = def_index
			add = 0.75
		}
	}
	multiply_temp_variable = { land_count = mobilization_level_modifier }
	multiply_temp_variable = { navy_count = mobilization_level_modifier }
	multiply_temp_variable = { airforce_count = mobilization_level_modifier }

	set_temp_variable = {
		equipment_cost_mobilization_level_modifier = {
			value = 0.25
			multiply = def_index
			add = 0.75
		}
	}
	multiply_variable = { equipment_operative_cost = equipment_cost_mobilization_level_modifier }

	if = { limit = { has_idea = AB_partial_mobilization_idea }
		if = { limit = { is_ai = yes }
			multiply_temp_variable = { land_count = 1.05 }
			multiply_temp_variable = { navy_count = 1.125 }
			multiply_temp_variable = { airforce_count = 1.125 }
		}
		else = {
			multiply_temp_variable = { land_count = 1.1 }
			multiply_temp_variable = { navy_count = 1.25 }
			multiply_temp_variable = { airforce_count = 1.25 }
		}
	}

	if = { limit = { has_idea = AB_full_mobilization_idea }
		if = { limit = { is_ai = yes }
			multiply_temp_variable = { land_count = 1.2 }
			multiply_temp_variable = { navy_count = 1.45 }
			multiply_temp_variable = { airforce_count = 1.45 }
		}
		else = {
			multiply_temp_variable = { land_count = 1.25 }
			multiply_temp_variable = { navy_count = 1.5 }
			multiply_temp_variable = { airforce_count = 1.5 }
		}
	}

	#Add up armed forces

	set_temp_variable = {
		personnel_cost = {
			value = land_count
			add = navy_count
			add = airforce_count
		}
	}

	#Values for breakdown tooltips

	set_variable = {
		defence_breakdown_industry = {
			value = array_defence_spend^def_index
			multiply = 7
		}
	}

	set_variable = {
		defence_breakdown_land = {
			value = land_count
			multiply = 7
			add = equipment_operative_cost
		}
	}

	set_variable = {
		defence_breakdown_navy = {
			value = navy_count
			multiply = 7
		}
	}

	set_variable = {
		defence_breakdown_airforce = {
			value = airforce_count
			multiply = 7
		}
	}

	set_variable = {
		defence_breakdown_industry_gdp_modifier = {
			value = 8.1
			divide = gdp_defence_modifier_industry
			add = -1
		}
	}

	set_variable = {
		defence_breakdown_personnel_gdp_modifier = {
			value = gdp_defence_modifier_personnel
			multiply = 7.752
			add = -1
		}
	}

	set_variable = {
		defence_breakdown_personnel_modifier = {
			value = personnel_cost_multiplier
			add = -1
		}
	}

	#Add military industry and personnel costs together for the total
	for_each_loop = {
		array = array_defence_spend
		index = i
		value = v
		add_to_variable = { array_defence_spend^i = personnel_cost }
		multiply_variable = { array_defence_spend^i = 7 } #Multiply each index by 7 days
		add_to_variable = { array_defence_spend^i = equipment_operative_cost }
		clamp_variable = { var = array_defence_spend^i min = 0 }
	}

	set_variable = { defence_gain = array_defence_spend^def_index }
	clamp_variable = { var = defence_gain min = 0 }
}

			value = 0.04
			multiply = amount_research_slots
			add = -0.2
		}
	}

	#helps balance large countries (china/india/ecc) and represents poor countries setting up infrastructure that western countries already have

		education_cost_gdp_multiplier_display = {
			value = gdp_scaling_modifier
			add = -1
		}
	}

	set_temp_variable = {
		education_cost_multiplier = {
			value = 1
			add = modifier@education_cost_multiplier_modifier
		}
	}

	multiply_temp_variable = { education_cost_multiplier = gdp_scaling_modifier }

#Weekly tick action
automated_debt_taker = { # Checks to see if treasury is negative at the same time as weekly balance adds or removes from treasury, borrowing debt automatically if treasury is negative (to make it positive).
	if = { limit = { check_variable = { treasury < 0 } }
		set_temp_variable = { treasury_temp = treasury }
		multiply_temp_variable = { treasury_temp = -1 }
		multiply_temp_variable = { treasury_temp = 1.25 }
		set_temp_variable = { gdp_adder = gdp_total }
		multiply_temp_variable = { gdp_adder = 0.01 } # 0.1% of GDP
		add_to_temp_variable = { treasury_temp = gdp_adder }
		set_temp_variable = { debt_temp = treasury_temp }
		multiply_temp_variable = { debt_temp = 1.01 } # adding 1% fee to debt autmoatically taken out (manual borrowing also has this)

		add_to_variable = { treasury = treasury_temp }
		clamp_variable = { var = treasury min = -1000000 max = 1000000 }
		add_to_variable = { debt = debt_temp }
	    clamp_variable = { var = debt min = 0 max = 1000000 }
		calculate_interest_rate = yes
	}
}

#Calculate GDP/C
calculate_gdp = {
	# Calculate Last Quarter's GDP
	if = {
		limit = {
			NOT = { has_country_flag = calculated_quarterly_gdp }
			OR = {
				check_variable = { global.month = 3 }
				check_variable = { global.month = 6 }
				check_variable = { global.month = 9 }
				check_variable = { global.month = 12 }
			}
		}
		if = {
			limit = {check_variable = { global.month = 3 }}
			set_variable = { last_quarter_gdp_total = first_gdp_total }
		}
		set_temp_variable = { last_quarter_gdp_total_temp = last_quarter_gdp_total }
		set_temp_variable = { this_quarter_gdp_total_temp = gdp_total }

		if = {
			limit = {
				OR = {
					NOT = { has_variable = last_quarter_gdp_total }
					check_variable = { last_quarter_gdp_total = 0 }
					check_variable = { last_quarter_gdp_total = 0.1 }
				}
			}
			set_variable = { last_quarter_gdp_growth_rate = 0 }
			set_variable = { last_quarter_gdp_total = gdp_total }
			add_to_array = { quarterly_gdp_growth_rate = last_quarter_gdp_growth_rate }
			set_country_flag = { flag = calculated_quarterly_gdp days = 85 value = 1 }
		}
		else = {
			set_variable = { last_quarter_gdp_growth_rate = this_quarter_gdp_total_temp }
			divide_variable = { last_quarter_gdp_growth_rate = last_quarter_gdp_total_temp }
			add_to_variable = { last_quarter_gdp_growth_rate = -1 }
			clamp_variable = { var = last_quarter_gdp_growth_rate min = -1 }
			
			set_temp_variable = { base_inflation_percent_change = last_quarter_gdp_growth_rate }
			multiply_temp_variable = { base_inflation_percent_change = 0.1 }
			multiply_temp_variable = { base_inflation_percent_change = -1 }
			clamp_temp_variable = { var = base_inflation_percent_change min = -0.05 max = 0.05 }
			modify_base_inflation_effect = yes
	
			set_variable = { last_quarter_gdp_total = gdp_total }
			add_to_array = { quarterly_gdp_growth_rate = last_quarter_gdp_growth_rate }
			set_country_flag = { flag = calculated_quarterly_gdp days = 85 value = 1 }
		}
	}

	# Reset the Gross Domestic Payment
	set_variable = { var = gdp_total value = 0 }

	# Calculates Workforce
	set_temp_variable = { workforce_total = population_total_m }
	multiply_temp_variable = { workforce_total = 0.6 }
	set_temp_variable = { workforce_total_multiplier = 1 }
	add_to_temp_variable = { workforce_total_multiplier = modifier@total_workforce_modifier }
	# this means that the maximum available workforce can't exceed 90% of the population
	clamp_temp_variable = { var = workforce_total_multiplier min = 0 max = 1.5 }
	multiply_temp_variable = { workforce_total = workforce_total_multiplier }
	set_temp_variable = { workforce_total_min = population_total_m }
	multiply_temp_variable = { workforce_total_min = 0.25 }
	clamp_temp_variable = { var = workforce_total min = workforce_total_min }
	#Counting Migrant Workers
	set_variable = { migrant_workforce_add = 0 }
	if = {
		limit = { check_variable = { net_immigration_rate > 0 } }
		set_variable = { migrant_workforce_add = net_immigration_rate }
		multiply_variable = { migrant_workforce_add = modifier@monthly_population }
		multiply_variable = { migrant_workforce_add = population_total_m }
		set_temp_variable = { migrant_workforce_add_multiplyer = 1 }
		add_to_temp_variable = { migrant_workforce_add_multiplyer = modifier@migrant_workforce_modifier }
		clamp_temp_variable = {
			var = migrant_workforce_add_multiplyer
			min = 0
		}
		multiply_variable = { migrant_workforce_add = migrant_workforce_add_multiplyer }
	}

	#Counting base unemplemployment
	set_temp_variable = { base_unemployment = workforce_total }
	multiply_temp_variable = { base_unemployment = base_unemployment_percent }
	set_temp_variable = { base_unemployment_multiplyer = 1 }
	add_to_temp_variable = { base_unemployment_multiplyer = modifier@base_unemployment_modifier }
	clamp_temp_variable = { var = base_unemployment_multiplyer min = 0.01 }
	multiply_temp_variable = { base_unemployment = base_unemployment_multiplyer }

	#Subtract base unemplemployment
	subtract_from_temp_variable = { workforce_total = base_unemployment }

	#Adding Migrant Workers
	set_temp_variable = { migrant_workforce_max = @maximum_migrant_workforce_percent_base }
	if = {
		limit = { NOT = { check_variable = { modifier@maximum_migrant_workforce_percent_modifier = 0 } } }
		add_to_temp_variable = { migrant_workforce_max = modifier@maximum_migrant_workforce_percent_modifier }
	}
	multiply_temp_variable = { migrant_workforce_max = workforce_total }
	clamp_temp_variable = {
		var = migrant_workforce_max
		min = 0
	}
	clamp_variable = {
		var = migrant_workforce_add
		min = 0
		max = migrant_workforce_max
	}
	add_to_temp_variable = { workforce_total = migrant_workforce_add }

	# Sets the total workforce displays
	set_variable = { workforce_total_display = workforce_total }
	set_variable = { workforce_total_percentage_display = workforce_total_display }
	divide_variable = { workforce_total_percentage_display = population_total_m }

	# Primary Sector
	# Employment in the Resource Sector
	set_variable = { resource_sector_workers = total_resource_amount }

	add_to_variable = { resource_sector_workers = resource_rights_who_provides_oil }

	add_to_variable = { resource_sector_workers = resource_rights_who_provides_chromium }
	add_to_variable = { resource_sector_workers = resource_rights_who_provides_rubber }
	add_to_variable = { resource_sector_workers = resource_rights_who_provides_aluminium }
	add_to_variable = { resource_sector_workers = resource_rights_who_provides_tungsten }

	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_oil }
	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_steel }
	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_chromium }
	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_rubber }
	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_aluminium }
	subtract_from_variable = { resource_sector_workers = resource_rights_who_takes_tungsten }


	multiply_variable = { resource_sector_workers = 0.01 } #in thousands Number of people who depend from these jobs, so not just direct workers but should include retail, services, food industry
	set_temp_variable = { resource_sector_workers_multiplier = 1 }
	add_to_temp_variable = { resource_sector_workers_multiplier = modifier@resource_sector_workers_modifier }
	multiply_variable = { resource_sector_workers = resource_sector_workers_multiplier }
	if = {
		limit = {
				check_variable = { overall_productivity > 1250 }
		}
		set_temp_variable = { resource_sector_workers_percentage = 0.1 }
    }
    else = {
		set_temp_variable = { productivity_factor = 1 }
		set_temp_variable = { res_p = overall_productivity }
		multiply_temp_variable = { res_p = 0.0008 }
		subtract_from_temp_variable = { productivity_factor = res_p }
		set_temp_variable = { resource_sector_workers_percentage = 0.9 }

	divide_variable = { resource_sector_worker_fulfillment = resource_sector_workers }
	clamp_variable = { var = resource_sector_worker_fulfillment min = 0 max = 1 }
	set_variable = { resource_sector_worker_fulfillment_var = -1 }
	add_to_variable = { resource_sector_worker_fulfillment_var = resource_sector_worker_fulfillment }
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = resources_manpower_fulfillment_modifier } } }

#Calculate modifiers associated with GDP/C
calculate_gdp_modifiers = {
	if = { limit = { NOT = { check_variable = { gdp_total = last_gdp_check } } }
		set_variable = { last_gdp_check = gdp_total }
		#Add the modifier if not existing
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = gdp_per_capita_country_modifiers } } }
			add_dynamic_modifier = { modifier = gdp_per_capita_country_modifiers }
		}

		#Construction speed factor
		#-0.012 * GDP/C + 0.572	- GDP/C c [0,30]
		#-0.001 * GDP/C + 0.230 - GDP/C c [30, 200]
		#Cap at 0
		set_variable = { production_speed_buildings_factor_gdp_modifier = gdp_per_capita }
		if = {
			limit = { check_variable = { production_speed_buildings_factor_gdp_modifier < 30 } }
			multiply_variable = { production_speed_buildings_factor_gdp_modifier = -0.012 }
			add_to_variable = { production_speed_buildings_factor_gdp_modifier = 0.572 }
		}
		else = {
			multiply_variable = { production_speed_buildings_factor_gdp_modifier = -0.001 }
			add_to_variable = { production_speed_buildings_factor_gdp_modifier = 0.230 }
		}
		clamp_variable = { var = production_speed_buildings_factor_gdp_modifier min = 0 }

		#Research speed factor
		# 0.003 * GDP/C - 0.153
		# Break-even at GDP/c = 51, clamped to 0 above
		set_variable = { research_speed_factor_gdp_modifier = gdp_per_capita }
		multiply_variable = { research_speed_factor_gdp_modifier = 0.003 }
		add_to_variable = { research_speed_factor_gdp_modifier = -0.153 }
		clamp_variable = { var = research_speed_factor_gdp_modifier max = 0 }

		#Stability factor
		# 0.001 * GDP/C - 0.100
		# Break-even at GDP/c = 100
		set_variable = { stability_factor_gdp_modifier = gdp_per_capita }
		multiply_variable = { stability_factor_gdp_modifier = 0.001 }
		add_to_variable = { stability_factor_gdp_modifier = -0.1 }

		# max workforce
		# 0.100 - 0.001 * GDP/C
		# Break-even at GDP/c = 100, min clamp (-0.30) at GDP/c = 400
		set_variable = { total_workforce_gdp_c_modifier_var = gdp_per_capita }
		multiply_variable = { total_workforce_gdp_c_modifier_var = -0.001 }
		add_to_variable = { total_workforce_gdp_c_modifier_var = 0.1 }
		clamp_variable = { var = total_workforce_gdp_c_modifier_var min = -0.30 }

		#fossil powerplants construction speed
		# 0.5 - 0.0125 * GDP/C
		# Break-even at GDP/c = 40, min clamp (-0.75) at GDP/c = 100
		set_variable = { fossil_pp_construction_gdp_c_modifier_var = gdp_per_capita }
		multiply_variable = { fossil_pp_construction_gdp_c_modifier_var = -0.0125 }
		add_to_variable = { fossil_pp_construction_gdp_c_modifier_var = 0.5 }
		clamp_variable = { var = fossil_pp_construction_gdp_c_modifier_var min = -0.75 }

		#Influence Defence
		set_variable = { gdpc_influence_defence_modifier = gdp_per_capita }
		divide_variable = { gdpc_influence_defence_modifier = global.average_gdp_per_capita }
		add_to_variable = { gdpc_influence_defence_modifier = -1 }
		clamp_variable = { var = gdpc_influence_defence_modifier min = 0 max = 0.25 }

		#Investment Cost
		set_variable = { gdpc_investment_cost_modifier = gdp_per_capita }
		divide_variable = { gdpc_investment_cost_modifier = global.average_gdp_per_capita }
		add_to_variable = { gdpc_investment_cost_modifier = -1 }
		clamp_variable = { var = gdpc_investment_cost_modifier min = -0.25 max = 0 }
	}
}

calculate_ai_taxes_desire = {
	# Raise Taxes
	if = { limit = { check_variable = { treasury_rate < 0 } }
		set_temp_variable = { mather = treasury_rate } # -10.00
		multiply_temp_variable = { mather = -2 } # 20.00
		if = { limit = { num_of_available_civilian_factories < 15 }
			multiply_temp_variable = { mather = 5 }
		}
		if = { limit = { num_of_available_civilian_factories < 30 }
			multiply_temp_variable = { mather = 5 }
		}
		if = { limit = { has_stability < 0.50 }
			multiply_temp_variable = { mather = 0.75 }
		}
		if = { limit = { has_stability < 0.25 }
			multiply_temp_variable = { mather = 0.75 }
		}
		# do not increase over 40%
		if = { limit = { check_variable = { population_tax_rate > 39 } }
			multiply_temp_variable = { mather = 0 }
		}

		# Custom: These are high pop nation should raise population tax
		if = {
			limit = {
				OR = {
					original_tag = CHI
					original_tag = RAJ
					original_tag = DRC
					original_tag = IND
					original_tag = JAP
					original_tag = NIG
					original_tag = EGY
					original_tag = BRA
				}
			}
			multiply_temp_variable = { mather = 4 }
		}

		set_variable = { ai_change_population_taxes = mather }
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [THIS.GetName]: AI Population Raise Desire [?ai_change_population_taxes] Current Population Tax: [?population_tax_rate] Current Treasury Rate: [?treasury_rate]"
		}
	}
	else = { # Lower Taxes
		set_temp_variable = { mather = treasury_rate } # 5
		multiply_temp_variable = { mather = 1.50 } # 2.5
		if = { limit = { has_stability < 0.50 }
			multiply_temp_variable = { mather = 2 }
		}
		if = { limit = { has_stability < 0.35 }
			multiply_temp_variable = { mather = 2 }
		}
		if = { limit = { has_stability < 0.15 }
			multiply_temp_variable = { mather = 2 }
		}
		if = { limit = { has_stability < 0.05 }
			multiply_temp_variable = { mather = 2 }
		}

		if = { limit = { check_variable = { interest_rate < 3 } }
			multiply_temp_variable = { mather = 1.15 }
		}
		if = { limit = { check_variable = { interest_rate < 5 } }
			multiply_temp_variable = { mather = 1.15 }
		}

		if = { limit = { check_variable = { debt_ratio < 0.15 } }
			multiply_temp_variable = { mather = 1.5 }
		}
		if = { limit = { check_variable = { debt_ratio < 0.25 } }
			multiply_temp_variable = { mather = 1.5 }
		}
		# Do not decrease lower than 20% personal taxes
		# Do not decrease personal if we can decrease corporate
		if = {
			limit = {
				OR = {
					check_variable = { population_tax_rate < 20 }
					check_variable = { corporate_tax_rate > 20 }
				}
			}
			multiply_temp_variable = { mather = 0 }
		}

		set_variable = { ai_change_population_taxes = mather }
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [THIS.GetName]: AI Population Lower Desire [?ai_change_population_taxes] Current Population Tax: [?population_tax_rate] Current Treasury Rate: [?treasury_rate]"
		}
	}

	if = { limit = { check_variable = { treasury_rate < 0 } }
		set_temp_variable = { mather = treasury_rate } # -2.0
		multiply_temp_variable = { mather = -2.5 } # -2.0 * -5 = 10
		if = { limit = { has_stability < 0.50 } # Should force the AI to raise corporate tax on low stability
			multiply_temp_variable = { mather = 2 }
		}
		if = { limit = { check_variable = { office_park_total > 20 } }
			multiply_temp_variable = { mather = 2.5 }
		}
		if = { limit = { check_variable = { office_park_total > 40 } }
			multiply_temp_variable = { mather = 2.5 }
		}
		if = { limit = { check_variable = { office_park_total > 60 } }
			multiply_temp_variable = { mather = 2.5 }
		}
		if = { limit = { num_of_available_civilian_factories < 15 }
			multiply_temp_variable = { mather = 0.05 }
		}
		if = { limit = { num_of_factories > 5 }
			multiply_temp_variable = { mather = 1.25 }
		}
		if = { limit = { num_of_factories > 15 }
			multiply_temp_variable = { mather = 1.75 }
		}
		if = { limit = { num_of_factories > 25 }
			multiply_temp_variable = { mather = 2.50 }
		}

		# do not increase corporate taxes if we can increase personal taxes
		if = {
			limit = {
				check_variable = { population_tax_rate < 30 }
			}

			multiply_temp_variable = { mather = 0 }
		}

		set_variable = { ai_change_corporate_taxes = mather }
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [THIS.GetName]: AI Corporate Raise Desire: [?ai_change_corporate_taxes] Current Corporate Tax: [?corporate_tax_rate] Current Treasury Rate: [?treasury_rate]"
		}
	}
	else = {
		set_temp_variable = { mather = treasury_rate } # 5
		multiply_temp_variable = { mather = 0.50 } # 2.5
		if = { limit = { num_of_available_civilian_factories < 15 }
			multiply_temp_variable = { mather = 3 }
		}
		if = { limit = { num_of_available_civilian_factories < 5 }
			multiply_temp_variable = { mather = 3 }
		}
		if = { limit = { num_of_available_civilian_factories > 30 }
			multiply_temp_variable = { mather = 0.5 }
		}

		if = { limit = { check_variable = { interest_rate < 3 } }
			multiply_temp_variable = { mather = 1.15 }
		}
		if = { limit = { check_variable = { interest_rate < 5 } }
			multiply_temp_variable = { mather = 1.15 }
		}

		if = { limit = { check_variable = { debt_ratio < 0.15 } }
			multiply_temp_variable = { mather = 1.5 }
		}
		if = { limit = { check_variable = { debt_ratio < 0.25 } }
			multiply_temp_variable = { mather = 1.5 }
		}

		# do not decrease lower than 15%
		if = {
			limit = {
				check_variable = { corporate_tax_rate < 16 }
			}
			multiply_temp_variable = { mather = 0 }
		}

		set_variable = { ai_change_corporate_taxes = mather }
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [THIS.GetName]: AI Corporate Lower Desire: [?ai_change_corporate_taxes] Current Corporate Tax: [?corporate_tax_rate] Current Treasury Rate: [?treasury_rate]"
		}
	}

update_employment_productivity_output_dynamic_modifier = {
	# productivity (P) and employment level (E) are supposed to behave as multiplicative factors of factories outputs (O)
	# (Outputs being counstruction speed for civilian factories, military output for mils and dockyard output for docks)
	# So if B is the base output and M the ingame modifier (production_speed_buildings_factor in the case of construction speed, for example)
	# The equation is O = B * (1 + M) * P * E
	# Unfortunately the game code doesn't allow such a direct multiplication to occur
	# Instead it is only possible to change the value of M by adding more bonuses or maluses so M = ( S + X )
	# Where S is the sum of all ingame modifiers (from laws, techs, national content, etc) and X the value we need to determine
	# Therefore the following calcutions exist to determine the value of X that satisfy the following equation:
	# B * (1 + S) * P * E = B * (1 + S + X)
	# B is simplified out so P * E * (1 + S) -1 -S = X
	# We group the -1: P * E * (1 + S) -1 * (1 + S) = X
	# And finally group the (1 + S): X = (1 + S) * (P*E - 1)
	# This value of X we found is then applied to a dynamic modifier
	if = {
		limit = {
			OR = {
				NOT = { check_variable = { overall_productivity = last_productivity_check } }
				NOT = { check_variable = { industrial_complex_total = last_industrial_complex_total } }
			}
		}

		set_variable = { last_productivity_check = overall_productivity }
		set_variable = { last_industrial_complex_total = industrial_complex_total }

		set_temp_variable = { preexisting_construction_speed_output_var = construction_speed_output_var }
		set_temp_variable = { preexisting_military_factories_output_var = military_factories_output_var }
		set_temp_variable = { preexisting_dockyards_output_var = dockyards_output_var }
		set_temp_variable = { preexisting_agriculture_district_output_var = agriculture_district_output_var }


		set_temp_variable = { productivity_calculations_var = overall_productivity }
		multiply_temp_variable = { productivity_calculations_var = 0.001 }

		# Cache expensive modifier reads upfront
		set_temp_variable = { cached_construction_modifier = modifier@production_speed_buildings_factor }
		set_temp_variable = { cached_military_modifier = modifier@industrial_capacity_factory }
		set_temp_variable = { cached_dockyard_modifier = modifier@industrial_capacity_dockyard }
		set_temp_variable = { cached_agriculture_modifier = modifier@production_speed_agriculture_district_factor }

		#this section calculates the employment level assuming that all factories you receive from trade, leasing and other sources are at 100% employment
		set_temp_variable = { civilian_factories_manpower_fulfillment_dyn_mod_var = industrial_complex_total }
		multiply_temp_variable = { civilian_factories_manpower_fulfillment_dyn_mod_var = civilian_factories_manpower_fulfillment }
		set_temp_variable = { factories_from_other_sources = num_of_civilian_factories }
		subtract_from_temp_variable = { factories_from_other_sources = industrial_complex_total }
		add_to_temp_variable = { civilian_factories_manpower_fulfillment_dyn_mod_var = factories_from_other_sources }
		divide_temp_variable = { civilian_factories_manpower_fulfillment_dyn_mod_var = num_of_civilian_factories }

		set_variable = { construction_speed_output_var = productivity_calculations_var }
		multiply_variable = { construction_speed_output_var = civilian_factories_manpower_fulfillment_dyn_mod_var }
		add_to_variable = { construction_speed_output_var = -1 } #to be used as malus in the dynamic modifier
		set_temp_variable = { sum_of_construction_speed_modifiers = cached_construction_modifier }
		subtract_from_temp_variable = { sum_of_construction_speed_modifiers = preexisting_construction_speed_output_var }
		add_to_temp_variable = { sum_of_construction_speed_modifiers = 1 }
		multiply_variable = { construction_speed_output_var = sum_of_construction_speed_modifiers }

		if = {
			limit = { num_of_military_factories > 0 }
			#this section calculates the employment level assuming that all factories you receive from trade, leasing and other sources are at 100% employment
			set_temp_variable = { military_factories_manpower_fulfillment_dyn_mod_var = military_fac }
			multiply_temp_variable = { military_factories_manpower_fulfillment_dyn_mod_var = military_factories_manpower_fulfillment }
			set_temp_variable = { mil_factories_from_other_sources = num_of_military_factories }
			subtract_from_temp_variable = { mil_factories_from_other_sources = military_fac }
			add_to_temp_variable = { military_factories_manpower_fulfillment_dyn_mod_var = mil_factories_from_other_sources }
			divide_temp_variable = { military_factories_manpower_fulfillment_dyn_mod_var = num_of_military_factories }

			set_variable = { military_factories_output_var = productivity_calculations_var }
			multiply_variable = { military_factories_output_var = military_factories_manpower_fulfillment_dyn_mod_var }
			add_to_variable = { military_factories_output_var = -1 } #to be used as malus in the dynamic modifier
			set_temp_variable = { sum_of_military_output_modifiers = cached_military_modifier }
			subtract_from_temp_variable = { sum_of_military_output_modifiers = preexisting_military_factories_output_var }
			add_to_temp_variable = { sum_of_military_output_modifiers = 1 }
			multiply_variable = { military_factories_output_var = sum_of_military_output_modifiers }
		}
		else = {
			set_variable = { military_factories_output_var = 0 }
		}

		if = {
			limit = { num_of_naval_factories > 0 }
			#this section calculates the employment level assuming that all factories you receive from trade, leasing and other sources are at 100% employment
			set_temp_variable = { naval_factories_manpower_fulfillment_dyn_mod_var = dockyard_fac }
			multiply_temp_variable = { naval_factories_manpower_fulfillment_dyn_mod_var = naval_factories_manpower_fulfillment }
			set_temp_variable = { nav_factories_from_other_sources = num_of_naval_factories }
			subtract_from_temp_variable = { nav_factories_from_other_sources = dockyard_fac }
			add_to_temp_variable = { naval_factories_manpower_fulfillment_dyn_mod_var = nav_factories_from_other_sources }
			divide_temp_variable = { naval_factories_manpower_fulfillment_dyn_mod_var = num_of_naval_factories }

			set_variable = { dockyards_output_var = productivity_calculations_var }
			multiply_variable = { dockyards_output_var = naval_factories_manpower_fulfillment_dyn_mod_var }
			add_to_variable = { dockyards_output_var = -1 } #to be used as malus in the dynamic modifier
			set_temp_variable = { sum_of_dockyard_output_modifiers = cached_dockyard_modifier }
			subtract_from_temp_variable = { sum_of_dockyard_output_modifiers = preexisting_dockyards_output_var }
			add_to_temp_variable = { sum_of_dockyard_output_modifiers = 1 }
			multiply_variable = { dockyards_output_var = sum_of_dockyard_output_modifiers }
		}
		else = {
			set_variable = { dockyards_output_var = 0 }
		}

		# Agricultural District Section
		set_variable = { agriculture_district_output_var = productivity_calculations_var }
		add_to_variable = { agriculture_district_output_var = -1 } #to be used as malus in the dynamic modifier
		set_temp_variable = { sum_of_agriculture_district_var_modifiers = cached_agriculture_modifier }
		subtract_from_temp_variable = { sum_of_agriculture_district_var_modifiers = preexisting_agriculture_district_output_var }
		add_to_temp_variable = { sum_of_agriculture_district_var_modifiers = -2 }
		multiply_variable = { agriculture_district_output_var = sum_of_agriculture_district_var_modifiers }
		set_variable = { productivity_tax_effect_display = productivity_calculations_var }
		add_to_variable = { productivity_tax_effect_display = -1 }
	}
}

# Effect: calculate_migration_mechanic_values
# Purpose: Calculates the effects of the migration mechanic.
calculate_migration_mechanic_values = {
	# Calculations
	clear_array = array_border_control_spend

	# Set the base cost of policing (billion $ per 10 000 000 people)
	set_temp_variable = { border_control_base_cost = 0.004 }

	# GDP scaling
	set_temp_variable = { gdp_border_cost_modifier = gdp_per_capita }
	multiply_temp_variable = { gdp_border_cost_modifier = @spending_laws_gdp_scaling }
	add_to_temp_variable = { gdp_border_cost_modifier = 1 }

	#Additional additive modifiers
	set_temp_variable = { border_cost_multiplier = 1 }
	add_to_temp_variable = { border_cost_multiplier = modifier@border_control_multiplier_modifier }
	set_variable = { border_cost_multiplier_display = border_cost_multiplier }
	add_to_variable = { border_cost_multiplier_display = -1 }

	# Recreate the Array
	resize_array = { array = array_border_control_spend size = 5 value = 0 }

	#Calculate the base cost of police levels
	for_each_loop = {
		array = array_border_control_spend
		index = i
		value = v
		add_to_variable = { array_border_control_spend^i = population_total }	#Unit 100 000
		multiply_variable = { array_border_control_spend^i = border_cost_multiplier }
		multiply_variable = { array_border_control_spend^i = gdp_border_cost_modifier }
		multiply_variable = { array_border_control_spend^i = 0.07 } #Multiply each index by 7 days #Unit 10 000 000
		multiply_variable = { array_border_control_spend^i = border_control_base_cost }
		clamp_variable = { var = array_border_control_spend^i min = 0 }
	}

	multiply_variable = { array_border_control_spend^0 = 1 }
	multiply_variable = { array_border_control_spend^1 = 1.6 }
	multiply_variable = { array_border_control_spend^2 = 2.2 }
	multiply_variable = { array_border_control_spend^3 = 2.8 }
	multiply_variable = { array_border_control_spend^4 = 3.4 }

	#Set which spending level the country is at
	if = { limit = { has_idea = open_borders }
		set_temp_variable = { border_control_index = 0 }
	}
	else_if = { limit = { has_idea = slightly_regulated_immigration }
		set_temp_variable = { border_control_index = 1 }
	}
	else_if = { limit = { has_idea = regulated_immigration }
		set_temp_variable = { border_control_index = 2 }
	}
	else_if = { limit = { has_idea = heavily_regulated_immigration }
		set_temp_variable = { border_control_index = 3 }
	}
	else_if = { limit = { has_idea = closed_borders }
		set_temp_variable = { border_control_index = 4 }
	}

	set_variable = { border_control_gain = array_border_control_spend^border_control_index }
	clamp_variable = { var = border_control_gain min = 0 }

	# -------------------- #
	# Cleanup Variables
	set_variable = { net_immigration_rate = 0 }
	set_variable = { max_net_immigration_rate = 0 }
	set_variable = { personal_war_refugees_display = 0 }
	set_variable = { neighbour_war_refugees_display = 0 }
	set_variable = { immigration_government_factor_display = 0 }
	set_variable = { immigration_employment_factor_display = 0 }

	# Add the Dynamic Modifier If you don't have it
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = migration_rate_impact_effect } } }
		add_dynamic_modifier = { modifier = migration_rate_impact_effect }
	}
	# Productivity Calculation
	set_temp_variable = { immigration_productivity_calculations = overall_productivity }
	divide_temp_variable = { immigration_productivity_calculations = global.average_world_productivity }
	add_to_temp_variable = { immigration_productivity_calculations = -1 }
	set_variable = { immigration_productivity_calculations_display = immigration_productivity_calculations }

	# Calculate Unemployment/Worker fulfillment
	if = { limit = { check_variable = { average_worker_fulfillment < 1 } }
		set_temp_variable = { immigration_employment_factor = 1 }
		subtract_from_temp_variable = { immigration_employment_factor = average_worker_fulfillment }
	}
	else = {
		subtract_from_temp_variable = { immigration_employment_factor = total_unemployed_percentage_display }
		multiply_temp_variable = { immigration_employment_factor = -0.75 }
	}
	set_variable = { immigration_employment_factor_display = immigration_employment_factor }

	# Government Section
	set_temp_variable = { immigration_government_factor = 0 }
	if = { limit = { has_liberal_government = yes }
		add_to_temp_variable = { immigration_government_factor = 0.05 }
	}
	if = { limit = { has_conservative_government = yes }
		add_to_temp_variable = { immigration_government_factor = -0.05 }
	}
	set_variable = { immigration_government_factor_display = immigration_government_factor }

	# Calculate Net Immigration Rate
	set_variable = { max_net_immigration_rate = 0 }
	add_to_variable = { max_net_immigration_rate = immigration_productivity_calculations }
	add_to_variable = { max_net_immigration_rate = immigration_employment_factor }
	add_to_variable = { max_net_immigration_rate = immigration_war_support_factor }
	add_to_variable = { max_net_immigration_rate = immigration_government_factor }
	add_to_variable = { max_net_immigration_rate = modifier@maximum_migration_rate_value }

	# War Refugees Section (optimized to avoid nested neighbor checks)
	hidden_effect = {
		set_temp_variable = { neighbouring_war_refugees = 0 }
		every_neighbor_country = {
			limit = { has_war = yes }
			add_to_temp_variable = { PREV.neighbouring_war_refugees = THIS.population_total_m }
			multiply_temp_variable = { PREV.neighbouring_war_refugees = @refugee_percentage_adjuster }
		}
	}
	set_variable = { neighbour_war_refugees_display = neighbouring_war_refugees }
	add_to_variable = { max_net_immigration_rate = neighbouring_war_refugees }

	# Check if you are at war and if you have refugees
	set_temp_variable = { our_war_refugees = 0 }
	if = { limit = { has_war = yes }
		# Calculate Total Potential Refugees
		add_to_temp_variable = { our_war_refugees = population_total_m }
		multiply_temp_variable = { our_war_refugees = @refugee_percentage_adjuster }
		multiply_temp_variable = { our_war_refugees = 0.20 } # Control factor to make it so the number doesn't get very big

		# Reduce War Refugees Based on War Suppport
		set_temp_variable = { total_war_support = 0.50 }
		add_to_temp_variable = { total_war_support = modifier@war_support_factor }
		clamp_temp_variable = { var = total_war_support min = 0 max = 1 }

		set_temp_variable = { war_support_modifier = -1 }
		add_to_temp_variable = { war_support_modifier = total_war_support }
		multiply_temp_variable = { our_war_refugees = total_war_support }

		set_variable = { personal_war_refugees_display = our_war_refugees }
		multiply_variable = { personal_war_refugees_display = -1 }
	}
	subtract_from_variable = { max_net_immigration_rate = our_war_refugees }

	# Calculate Migration Rate
	set_temp_variable = { migration_rate_value = modifier@base_migration_rate_value }

	set_temp_variable = { migration_rate_value_multiplier = 1 }
	add_to_temp_variable = { migration_rate_value_multiplier = modifier@migration_rate_value_factor }
	clamp_temp_variable = { var = migration_rate_value_multiplier min = 0.01 }

	if = { limit = { check_variable = { migration_rate_value > 0 } }
		multiply_temp_variable = { migration_rate_value = migration_rate_value_multiplier }
	}
	else = {
		divide_temp_variable = { migration_rate_value = migration_rate_value_multiplier }
	}

	# Check if there are other countries
	if = { limit = { check_variable = { global.countries^1 = 0 } }
		set_variable = { max_net_immigration_rate = 0 }
		set_variable = { net_immigration_rate = 0 }
	}
	clamp_variable = { var = max_net_immigration_rate max = 2.0 min = -0.9999 }

	set_variable = { net_immigration_rate = max_net_immigration_rate }
	multiply_variable = { net_immigration_rate = migration_rate_value }
	clamp_variable = { var = net_immigration_rate min = -0.9999 max = max_net_immigration_rate }
}

set_frontier_emerging_economy_massive = {
	every_country = {
		if = {
			limit = {
				OR = {
					has_idea = frontier_market
					has_idea = emerging_economy
				}
				NOT = {
					is_in_array = {
						array = global.frontier_and_emerging_economies
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.frontier_and_emerging_economies
				value = THIS
			}
		}
		else_if = {
			limit = {
				has_idea = developed_economy
				is_in_array = {
					array = global.frontier_and_emerging_economies
					value = THIS
				}
			}
			remove_from_array = {
				array = global.frontier_and_emerging_economies
				value = THIS
			}
		}
	}
}
