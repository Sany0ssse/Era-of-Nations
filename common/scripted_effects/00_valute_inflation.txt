count_euro_inflation = {
	set_variable = { global.average_euro_inflation = 0 }
	set_variable = { global.euro_members_count = 0 }
	every_country = {
		limit = { has_idea = the_euro }
		add_to_variable = { global.average_euro_inflation = inflation_percent }
		add_to_variable = { global.euro_members_count = 1 }
	}
	divide_variable = { global.average_euro_inflation = global.euro_members_count }
}

count_euro_personal_inflation = {
	if = {
		limit = {
			check_variable = { inflation_percent < global.average_euro_inflation }
		}
		set_variable = { euro_personal_inflation_multiplier = global.average_euro_inflation }
		subtract_from_variable = { euro_personal_inflation_multiplier = inflation_percent }
	}
	else_if = {
		limit = {
			check_variable = { inflation_percent > global.average_euro_inflation }
		}
		set_variable = { euro_personal_inflation_multiplier = inflation_percent }
		subtract_from_variable = { euro_personal_inflation_multiplier = global.average_euro_inflation }
		multiply_variable = { euro_personal_inflation_multiplier = -1 }
	}
	else = {
		set_variable = { euro_personal_inflation_multiplier = 0 }
	}
	if = {
		limit = { check_variable = { euro_personal_inflation_multiplier < -0.05 } }
		set_variable = { euro_personal_inflation_multiplier = -0.05 }
	}
	if = {
		limit = { check_variable = { euro_personal_inflation_multiplier > 0.05 } }
		set_variable = { euro_personal_inflation_multiplier = 0.05 }
	}
}
