# Era of Nations: bounded lifecycle for existing bilateral energy contracts.
# Amounts retain the original convention: positive imports, negative exports.
# eon_energy_pair_partner is a country-valued temporary variable set by the caller.
eon_energy_read_pair_record = {
	set_temp_variable = { eon_energy_current_count = 0 }
	set_temp_variable = { eon_energy_current_amount = 0 }
	set_temp_variable = { eon_energy_current_price = 0 }
	for_each_loop = {
		array = energy_contractors
		value = eon_energy_record_partner
		index = eon_energy_record_index
		if = {
			limit = { check_variable = { eon_energy_record_partner = eon_energy_pair_partner } }
			add_to_temp_variable = { eon_energy_current_count = 1 }
			set_temp_variable = { eon_energy_current_amount = energy_contracts_ammount^eon_energy_record_index }
			set_temp_variable = { eon_energy_current_price = energy_contracts_price^eon_energy_record_index }
		}
	}
}

# Remove every record for this partner while keeping the three arrays aligned.
eon_energy_remove_pair_record = {
	while_loop_effect = {
		limit = { is_in_array = { array = energy_contractors value = eon_energy_pair_partner } }
		set_temp_variable = { eon_energy_remove_break = 0 }
		for_each_loop = {
			array = energy_contractors
			value = eon_energy_record_partner
			index = eon_energy_record_index
			break = eon_energy_remove_break
			if = {
				limit = { check_variable = { eon_energy_record_partner = eon_energy_pair_partner } }
				set_temp_variable = { eon_energy_remove_index = eon_energy_record_index }
				set_temp_variable = { eon_energy_remove_break = 1 }
			}
		}
		remove_from_array = { array = energy_contractors index = eon_energy_remove_index }
		remove_from_array = { array = energy_contracts_ammount index = eon_energy_remove_index }
		remove_from_array = { array = energy_contracts_price index = eon_energy_remove_index }
	}
}

eon_energy_clear_pending = {
	clr_country_flag = eon_energy_outgoing_offer
	clr_country_flag = eon_energy_incoming_offer
	clr_country_flag = eon_energy_offer_cancelled
	clr_country_flag = question_about_contract
	clear_variable = eon_energy_offer_partner
	clear_variable = eon_energy_offer_amount
	clear_variable = eon_energy_offer_price
	clear_variable = eon_energy_offer_quantity
	clear_variable = eon_energy_offer_weekly
	clear_variable = eon_energy_offer_old_count
	clear_variable = eon_energy_offer_old_amount
	clear_variable = eon_energy_offer_old_price
	clear_variable = eon_energy_offer_old_weekly
	clear_variable = eon_energy_ai_accept_chance
}

# Keep a withdrawn proposal serialized until its existing response is consumed.
# Clearing its lock here would let an old event act on a new same-pair snapshot.
eon_energy_invalidate_pair_pending = {
	if = {
		limit = {
			has_variable = eon_energy_offer_partner
			check_variable = { eon_energy_offer_partner = eon_energy_pair_partner }
		}
		set_country_flag = eon_energy_offer_cancelled
	}
}

# Annexation destroys the partner; no same-pair proposal can be restarted.
eon_energy_clear_pair_pending = {
	if = {
		limit = {
			has_variable = eon_energy_offer_partner
			check_variable = { eon_energy_offer_partner = eon_energy_pair_partner }
		}
		eon_energy_clear_pending = yes
	}
}

# Caller is one contracting country, eon_energy_pair_partner identifies the other.
# The caller owns the notice; no pending lock is created by the notice.
eon_energy_end_pair = {
	eon_energy_remove_pair_record = yes
	eon_energy_invalidate_pair_pending = yes
	var:eon_energy_pair_partner = {
		set_temp_variable = { eon_energy_pair_partner = PREV }
		eon_energy_remove_pair_record = yes
		eon_energy_invalidate_pair_pending = yes
		country_event = energy_selling.5
		ingame_update_setup = yes
	}
	ingame_update_setup = yes
}

# GUI sender: all mutable controls are copied before the receiving event is fired.
eon_energy_send_offer = {
	set_variable = { eon_energy_offer_partner = energy_selling_selected_TAG }
	set_variable = { eon_energy_offer_amount = temp_energy_ammount }
	set_variable = { eon_energy_offer_price = temp_energy_price }
	set_variable = { eon_energy_offer_quantity = eon_energy_offer_amount }
	if = {
		limit = { check_variable = { eon_energy_offer_quantity < 0 } }
		multiply_variable = { eon_energy_offer_quantity = -1 }
	}
	set_variable = { eon_energy_offer_weekly = eon_energy_offer_quantity }
	multiply_variable = { eon_energy_offer_weekly = eon_energy_offer_price }
	set_variable = { eon_energy_offer_old_count = eon_energy_current_count }
	set_variable = { eon_energy_offer_old_amount = eon_energy_current_amount }
	set_variable = { eon_energy_offer_old_price = eon_energy_current_price }
	set_variable = { eon_energy_offer_old_weekly = eon_energy_current_amount }
	if = {
		limit = { check_variable = { eon_energy_offer_old_weekly < 0 } }
		multiply_variable = { eon_energy_offer_old_weekly = -1 }
	}
	multiply_variable = { eon_energy_offer_old_weekly = eon_energy_offer_old_price }
	set_country_flag = eon_energy_outgoing_offer
	set_country_flag = question_about_contract
	if = {
		limit = { is_ai = yes }
		set_country_flag = { flag = energy_recall_calldown@energy_selling_selected_TAG days = 365 value = 1 }
	}
	var:eon_energy_offer_partner = {
		set_temp_variable = { eon_energy_pair_partner = PREV }
		eon_energy_read_pair_record = yes
		set_variable = { eon_energy_offer_old_count = eon_energy_current_count }
		set_variable = { eon_energy_offer_old_amount = eon_energy_current_amount }
		set_variable = { eon_energy_offer_old_price = eon_energy_current_price }
		set_variable = { eon_energy_offer_old_weekly = eon_energy_current_amount }
		if = {
			limit = { check_variable = { eon_energy_offer_old_weekly < 0 } }
			multiply_variable = { eon_energy_offer_old_weekly = -1 }
		}
		multiply_variable = { eon_energy_offer_old_weekly = eon_energy_offer_old_price }
		set_variable = { eon_energy_offer_partner = PREV }
		set_variable = { eon_energy_offer_amount = PREV.eon_energy_offer_amount }
		multiply_variable = { eon_energy_offer_amount = -1 }
		set_variable = { eon_energy_offer_price = PREV.eon_energy_offer_price }
		set_variable = { eon_energy_offer_quantity = PREV.eon_energy_offer_quantity }
		set_variable = { eon_energy_offer_weekly = PREV.eon_energy_offer_weekly }
		set_country_flag = eon_energy_incoming_offer
		set_country_flag = question_about_contract
		if = {
			limit = { check_variable = { eon_energy_offer_amount > 0 } }
			country_event = energy_selling.1
		}
		else = { country_event = energy_selling.4 }
	}
	update_energy_dirty_variable = yes
}

# Recipient event: validate current records against both frozen pre-offer snapshots.
eon_energy_validate_offer = {
	set_temp_variable = { eon_energy_offer_valid = 0 }
	set_temp_variable = { eon_energy_pair_partner = FROM }
	eon_energy_read_pair_record = yes
	if = {
		limit = {
			exists = yes
			has_country_flag = eon_energy_incoming_offer
			NOT = { has_country_flag = eon_energy_offer_cancelled }
			has_country_flag = question_about_contract
			check_variable = { eon_energy_offer_partner = FROM }
			has_country_flag = energy_agreement@FROM
			NOT = { has_war_with = FROM }
			check_variable = { energy_contractors^num = energy_contracts_ammount^num }
			check_variable = { energy_contractors^num = energy_contracts_price^num }
			check_variable = { eon_energy_current_count = eon_energy_offer_old_count }
			check_variable = { eon_energy_current_count < 2 }
			check_variable = { eon_energy_current_amount = eon_energy_offer_old_amount }
			check_variable = { eon_energy_current_price = eon_energy_offer_old_price }
			check_variable = { eon_energy_offer_price > -0.001 }
			NOT = { check_variable = { eon_energy_offer_amount = 0 } }
			FROM = {
				exists = yes
				has_country_flag = eon_energy_outgoing_offer
				NOT = { has_country_flag = eon_energy_offer_cancelled }
				has_country_flag = question_about_contract
				check_variable = { eon_energy_offer_partner = PREV }
				has_country_flag = energy_agreement@PREV
				check_variable = { eon_energy_offer_price = PREV.eon_energy_offer_price }
			}
		}
		set_temp_variable = { eon_energy_mirror_amount = FROM.eon_energy_offer_amount }
		multiply_temp_variable = { eon_energy_mirror_amount = -1 }
		if = {
			limit = { check_variable = { eon_energy_offer_amount = eon_energy_mirror_amount } }
			FROM = {
				set_temp_variable = { eon_energy_pair_partner = PREV }
				eon_energy_read_pair_record = yes
				if = {
					limit = {
						check_variable = { energy_contractors^num = energy_contracts_ammount^num }
						check_variable = { energy_contractors^num = energy_contracts_price^num }
						check_variable = { eon_energy_current_count = eon_energy_offer_old_count }
						check_variable = { eon_energy_current_count < 2 }
						check_variable = { eon_energy_current_amount = eon_energy_offer_old_amount }
						check_variable = { eon_energy_current_price = eon_energy_offer_old_price }
					}
					PREV = { set_temp_variable = { eon_energy_offer_valid = 1 } }
				}
			}
		}
	}
}

# Recipient event. Cleanup is immediate; acknowledgement events cannot unlock later offers.
eon_energy_finish_response = {
	if = {
		limit = {
			has_country_flag = eon_energy_incoming_offer
			check_variable = { eon_energy_offer_partner = FROM }
		}
		FROM = {
			if = {
				limit = {
					has_variable = eon_energy_offer_partner
					check_variable = { eon_energy_offer_partner = PREV }
				}
				eon_energy_clear_pending = yes
			}
		}
		eon_energy_clear_pending = yes
		update_energy_dirty_variable = yes
	}
}

eon_energy_accept_offer = {
	eon_energy_validate_offer = yes
	if = {
		limit = { check_variable = { eon_energy_offer_valid = 1 } }
		set_temp_variable = { eon_energy_pair_partner = FROM }
		eon_energy_remove_pair_record = yes
		add_to_array = { array = energy_contractors value = FROM }
		add_to_array = { array = energy_contracts_ammount value = eon_energy_offer_amount }
		add_to_array = { array = energy_contracts_price value = eon_energy_offer_price }
		FROM = {
			set_temp_variable = { eon_energy_pair_partner = PREV }
			eon_energy_remove_pair_record = yes
			add_to_array = { array = energy_contractors value = PREV }
			add_to_array = { array = energy_contracts_ammount value = eon_energy_offer_amount }
			add_to_array = { array = energy_contracts_price value = eon_energy_offer_price }
			country_event = energy_selling.2
		}
		eon_energy_finish_response = yes
		ingame_update_setup = yes
		FROM = { ingame_update_setup = yes }
	}
	else = {
		custom_effect_tooltip = eon_energy_offer_invalid_tt
		eon_energy_finish_response = yes
	}
}
