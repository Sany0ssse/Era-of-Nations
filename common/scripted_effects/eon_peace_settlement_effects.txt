# Direct native pair settlement; no iteration over other belligerents.
eon_peace_native_statusquo = {
 set_country_flag = eon_consensual_peace@ROOT
 ROOT = { set_country_flag = eon_consensual_peace@PREV }
 white_peace = { tag = ROOT message = eon_peace_native_message }
}

eon_peace_execute_statusquo = {
 if = { limit = { eon_peace_accept_statusquo_ready = yes }
 eon_peace_reject_statusquo = yes
 eon_peace_native_statusquo = yes
 if = { limit = { NOT = { has_war_with = ROOT } }
  set_variable = { eon_peace_last_kind = 1 }
  ROOT = { set_variable = { eon_peace_last_kind = 1 } }
  country_event = { id = eon_peace_settlement.2 }
  ROOT = { country_event = { id = eon_peace_settlement.2 } }
  log = "[EON_PEACE_EXECUTED] sender=[ROOT.GetTag] recipient=[THIS.GetTag] kind=1"
 }
 }
}
