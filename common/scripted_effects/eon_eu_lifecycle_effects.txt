# Reprice current participants and office allocations without transferring treasury.
# Role changes must not trigger a new Parliament election or renew its mandate.
eon_eu_update_budget = {
 if = { limit = {
  eon_eu_union_active = yes
  OR = { has_global_flag = eon_eu_membership_changed has_global_flag = eon_eu_budget_changed NOT = { has_global_flag = eon_eu_budget_year } }
 }
  set_EU_gdp = yes
  update_EU_budget_call_rate = yes
  set_variable = { global.EU_total_exp = global.EU_total_budget }
  set_temp_variable = { eon_eu_budget_ratio_sum = global.EU_growth_exp_ratio }
  add_to_temp_variable = { eon_eu_budget_ratio_sum = global.EU_natural_exp_ratio }
  add_to_temp_variable = { eon_eu_budget_ratio_sum = global.EU_global_exp_ratio }
  add_to_temp_variable = { eon_eu_budget_ratio_sum = global.EU_admin_exp_ratio }
  if = { limit = { check_variable = { eon_eu_budget_ratio_sum > 0 } }
   update_EU_expenditure = yes
  }
  else = { set_EU_expenditure = yes }
  distribute_EU_budget = yes
  EU_global_click_cost = yes
  # Membership and office changes do not postpone the annual recalculation.
  if = { limit = { NOT = { has_global_flag = eon_eu_budget_year } }
   set_global_flag = { flag = eon_eu_budget_year days = 365 value = 1 }
  }
  clr_global_flag = eon_eu_budget_changed
 }
}

# One world clock: the EU does not depend on an immortal utility country or GUI.
eon_eu_update_parliament = {
	if = { limit = { eon_eu_union_active = yes }
		eon_eu_update_budget = yes
		if = { limit = {
			OR = { has_global_flag = eon_eu_membership_changed NOT = { has_global_flag = eon_eu_parliament_term } }
		}
			clear_EU_Parliament = yes
			election_EU_Parliament = yes
			# A changed delegation does not renew the whole Parliament's mandate.
			if = { limit = { NOT = { has_global_flag = eon_eu_parliament_term } }
				if = { limit = { date < 2000.2.1 }
					set_global_flag = { flag = eon_eu_parliament_term days = 1625 value = 1 }
				}
				else = { set_global_flag = { flag = eon_eu_parliament_term days = 1825 value = 1 } }
			}
			every_country = {
				remove_mission = EU_update_vars_mission
				remove_mission = EU_clear_voting_mission
				remove_mission = EU_election_2004
				remove_mission = EU_election_5_years
			}
			clr_global_flag = eon_eu_membership_changed
		}
	}
}
