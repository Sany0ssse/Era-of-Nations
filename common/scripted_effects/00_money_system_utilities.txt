# INFO: The file is a series of other effects to interact with the money system

set_static_state_modifier_on_game_start = {
	every_owned_state = {
		limit = { check_variable = { THIS.state_renewable_capacity_factor_modifier_var > 0 } }
		meta_effect = {
			text = {
				add_state_modifier = { modifier = { state_renewable_capacity_factor_modifier = [NUM] } }
			}
			NUM = "[?state_renewable_capacity_factor_modifier_var]"
		}
	}
}
