count_current_consumption = {
	set_variable = { divisions_in_offensive_combat_consumption = 0 }
	set_variable = { divisions_in_defensive_combat_consumption = 0 }
    set_variable = { land_ammo_consumption = 0 }
    set_variable = { air_ammo_consumption = 0 }
    set_variable = { fleet_ammo_consumption = 0 }
    set_variable = { ammo_consumption = 0 }

    #Count Fleet Consumption
    #Пометка для MSа: Да, я понимаю что тут чекается каждый регион, но только с портами но только для стран с флотом
    # и только во время войны
    # и можно сделать технически чтобы оно чекалось раз в неделю / месяц, подумаю еще
    if = {
        limit = {
            has_war = yes
            check_variable = { num_ships > 0 }
        }
        set_temp_variable = { number_of_ships_in_combat = num_ships }
        every_state = {
            limit = { naval_base > 0 }
            PREV = {
                if = {
                    limit = {
                        ships_in_state_ports = {
                            size  > 0
                            state = PREV
                        }
                    }
                    set_temp_variable = { i = 0 }
                    while_loop_effect = {
                        limit = {
                            check_variable = { i < 100 }
                            ships_in_state_ports = {
                                size  > i
                                state = PREV
                            }
                        }
                        add_to_temp_variable = { i = 1 }
                    }
                    subtract_from_temp_variable = {
                        number_of_ships_in_combat = i
                    }
                }
            }
        }
        set_variable = { fleet_ammo_consumption = number_of_ships_in_combat }
        divide_variable = { fleet_ammo_consumption = num_ships }
        multiply_variable = { fleet_ammo_consumption = fleet_maximum_ammo_consumption }
        round_variable = fleet_ammo_consumption
    }

    #Count Air Consumption
    #Пометка для MSа: Тут уже полегче, в общем то просто если есть война используется макс значение потребления авиации
    # если же ты воюешь добровольцами то процент сколько эти добровольцы от армии составляют ну оч примерно
    # из минусов - приходится чекать страны
    # Ну и с периодичностью тоже можно подумать
    if = {
        limit = {
            has_war = yes
            check_variable = { num_deployed_planes > 0 }
        }
        set_variable = { air_ammo_consumption = air_maximum_ammo_consumption }
        if = {
            limit = {
                check_variable = { alliance_strength_ratio > 1 }
                NOT = {
                    any_enemy_country = { is_major = yes }
                }
            }
            divide_variable = { air_ammo_consumption = alliance_strength_ratio }
            round_variable = air_ammo_consumption
        }
    }
    else_if = {
        limit = {
            check_variable = { num_deployed_planes > 0 }
            any_other_country = {
                has_volunteers_amount_from = {
                    tag = PREV
                    count > 0
                }
            }
        }
        set_temp_variable = { number_of_planes_in_combat = 0 }
        every_other_country = {
            limit = {
                has_volunteers_amount_from = {
                    tag = PREV
                    count > 0
                }
            }
            add_to_temp_variable = { PREV.number_of_planes_in_combat = 1 }
        }
        set_variable = { air_ammo_consumption = number_of_planes_in_combat }
        divide_variable = { air_ammo_consumption = num_deployed_planes }
        multiply_variable = { air_ammo_consumption = air_maximum_ammo_consumption }
        round_variable = air_ammo_consumption
    }

    #Count Land Consumption
    #Пометка для MSа: Тут мы считаем у каждого генерала сколько у него дивизий атакуют или защищаются
    # да, я понимаю что можно арбузить не юзая генералов вообще, но тут ничего не поделаешь +
    # этим вероятно будет заниматься только Хаски и то вряд ли
    # все же играть без генералов кринж
    if = {
        limit = {
            has_army_size = {
                size > 0
            }
            OR = {
                has_war = yes
                any_other_country = {
                    has_volunteers_amount_from = {
                        tag = PREV
                        count > 0
                    }
                }
            }
        }

        set_temp_variable = { one_land_unit_consumption = land_maximum_ammo_consumption }
        divide_temp_variable = { one_land_unit_consumption = num_battalions }

		every_army_leader = {
			limit = {
				is_leading_army_group = no
			}
			add_to_variable = { PREV.divisions_in_offensive_combat_consumption = num_units_offensive_combats }
			add_to_variable = { PREV.divisions_in_defensive_combat_consumption = num_units_defensive_combats }
		}

		#### offensive units consumption
		multiply_variable = { divisions_in_offensive_combat_consumption = one_land_unit_consumption }
		multiply_variable = { divisions_in_offensive_combat_consumption = 2 }
        round_variable = divisions_in_offensive_combat_consumption

		#### defensive units consumption
		multiply_variable = { divisions_in_defensive_combat_consumption = one_land_unit_consumption }
        round_variable = divisions_in_defensive_combat_consumption

        #Sum land units consumption
        add_to_variable = { land_ammo_consumption = divisions_in_offensive_combat_consumption }
        add_to_variable = { land_ammo_consumption = divisions_in_defensive_combat_consumption }
        round_variable = land_ammo_consumption
    }

    #Count Maximum Ammo Consumption
    add_to_variable = { ammo_consumption = fleet_ammo_consumption }
    add_to_variable = { ammo_consumption = air_ammo_consumption }
    add_to_variable = { ammo_consumption = land_ammo_consumption }
    set_temp_variable = { ammo_consumption_multiplyer_modifier = 1 }
    add_to_temp_variable = { ammo_consumption_multiplyer_modifier = modifier@ammo_consumption_modifier }
    multiply_variable = { ammo_consumption = ammo_consumption_multiplyer_modifier }
    round_variable = ammo_consumption
    clamp_variable = { var = ammo_consumption min = 0 max = 1000000000 }
}

count_consumption_rating = {
    set_variable = { ammo_production_rating = ammo_production }
    subtract_from_variable = { ammo_production_rating = ammo_consumption }
    set_variable = { ammo_production_rating_debuff = 0 }
    if = {
        limit = {
            check_variable = { ammo_production_rating < 0 }
            check_variable = { ammo_stock = 0 }
        }
        set_variable = { ammo_production_rating_debuff = ammo_production_rating }
        multiply_variable = { ammo_production_rating_debuff = 0.00001 }
        clamp_variable = { var = ammo_production_rating_debuff min = -0.5 }
        if = {
            limit = {
                NOT = { has_dynamic_modifier = { modifier = ammo_deficit_debuff } }
            }
            add_dynamic_modifier = { modifier = ammo_deficit_debuff }
        }
    }
    else_if = {
        limit = { has_dynamic_modifier = { modifier = ammo_deficit_debuff } }
        remove_dynamic_modifier = { modifier = ammo_deficit_debuff }
    }
    if = {
        limit = {
            check_variable = { ammo_production_rating > 0 }
            check_variable = { ammo_stock < max_ammo_limit }
        }
        set_variable = { Ammo_topbar_icon_index = 2 }
    }
    else_if = {
        limit = { check_variable = { ammo_production_rating < 0 } }
        set_variable = { Ammo_topbar_icon_index = 3 }
    }
    else = {
        set_variable = { Ammo_topbar_icon_index = 1 }
    }
}

        add_to_variable = { summary_ammunition_effect = -0.1 }
    }
    else_if = {
        limit = { has_idea = some_ammo_restrictions_spirit }
        add_to_variable = { summary_ammunition_effect = -0.05 }
    }
    else_if = {
        limit = { has_idea = some_ammo_additional_consumption_spirit }
        add_to_variable = { summary_ammunition_effect = 0.05 }
    }
    else_if = {
        limit = { has_idea = heavy_ammo_additional_consumption_spirit }
        add_to_variable = { summary_ammunition_effect = 0.1 }
    }
}

    if = {
        limit = {
            check_variable = { ammo_stock < max_ammo_limit }
			NOT = { has_active_mission = bankruptcy_incoming_collapse }
            NOT = { has_country_flag = recently_bought_ammo }
        }
	    set_temp_variable = { adding_ammo = 50000 }
        add_ammo = yes
	    set_temp_variable = { treasury_change = -0.1 }
	    modify_treasury_effect = yes
		set_country_flag = { flag = recently_bought_ammo value = 1 days = 28 }
    }
}
